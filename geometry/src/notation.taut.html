8// `geo!(`add, zero, neg, mul, one; AB ∥ DC ∧ ¬(AB ∥ AD)`)`: statements about
9// points of the plane over a ring, written with the ring's operations so that
10// `ring_norm!` reads them. A point is a capital letter naming a variable of
11// type `Point(K)`, and a segment is two of them, `AB` from A to B.
12//
13// - `AB ∥ CD`: the cross product of B - A and D - C is 0.
14// - `AB ≅ CD`: the two squared lengths are the same (no square roots in a ring).
15// - `¬ R`: R is false. `R ∧ S`: both.
16syntax!(geo_notation, ```par
17%start Expr
18%title "plane geometry"
19%%
20Expr: Header Comma Header Comma Header Comma Header Comma Header Semi Formula;
21Header: Name | Embed;
22Formula: Clause { Conj Clause };
23Clause: Negation | Relation | Group;
24Negation: Not Clause;
25Group: LParen Formula RParen;
26Relation: Segment RelOp Segment;
27RelOp: Parallel | Congruent;
28Embed: TautoExpr;
29Name: /[a-z_][_\p{L}\p{N}']*(::[_\p{L}][_\p{L}\p{N}']*)*/;
30Segment: /[A-Z][A-Z]/;
31Parallel: '∥';
32Congruent: '≅';
33Not: '¬';
34Conj: '∧';
35Comma: ',';
36Semi: ';';
37LParen: '(';
38RParen: ')';
39```, with(tokens: geo_tokens)) => struct Expr, enum Header, struct Formula, struct Formula_Rep1, enum Clause, struct Negation, struct Group, struct Relation, enum RelOp, fn parse

41pub(open) struct GeoSyn { add: SynExpr, zero: SynExpr, neg: SynExpr, mul: SynExpr, one: SynExpr }

43pub fn geo_header(h: geo_notation::Header): SynExpr {
44  match h {
45    Name(t) => ExVar(Syn::ident(t))
46    Embed(e) => e
47  }
48}

50// The `i`-th letter of a segment, as the variable of that point.
51pub fn geo_point(t: SynToken, i: u64): SynExpr ! {macro_error} {
52  match Text::slice(t.text, i, i + 1) {
53    None => Syn::error(t.span, "geo!: a segment is two capital letters")
54    Some(c) => ExVar(Syn::ident(SynToken { text: c, span: t.span, ctx: t.ctx }))
55  }
56}

58// `a - b` in the ring.
59pub fn geo_sub(o: GeoSyn, a: SynExpr, b: SynExpr): SynExpr {
60  let add := o.add
61  let neg := o.neg
62  quote { $add($a, $neg($b)) }
63}
64
65// The two coordinates of the vector of a segment, from its first point to its second.
66pub fn geo_vector(o: GeoSyn, t: SynToken): SynExpr × SynExpr ! {macro_error} {
67  let p := geo_point(t, 0)
68  let q := geo_point(t, 1)
69  #(geo_sub(o, quote { px($q) }, quote { px($p) }), geo_sub(o, quote { py($q) }, quote { py($p) }))
70}

72pub fn geo_relation(o: GeoSyn, r: geo_notation::Relation): SynExpr ! {macro_error} {
73  match geo_vector(o, r.segment) {
74    #(u1, u2) => match geo_vector(o, r.segment_2) {
75      #(v1, v2) => {
76        let add := o.add
77        let neg := o.neg
78        let mul := o.mul
79        let zero := o.zero
80        match r.rel_op {
81          Parallel(t) => quote { $add($mul($u1, $v2), $neg($mul($u2, $v1))) = $zero }
82          Congruent(t) => quote {
83            $add($mul($u1, $u1), $mul($u2, $u2)) = $add($mul($v1, $v1), $mul($v2, $v2))
84          }
85        }
86      }
87    }
88  }
89}

91pub fn geo_clause(o: GeoSyn, c: geo_notation::Clause): SynExpr ! {macro_error, Div} {
92  match c {
93    Negation(n) => {
94      let p := geo_clause(o, n.clause)
95      quote { ($p) -> False }
96    }
97    Group(g) => geo_formula(o, g.formula)
98    Relation(r) => geo_relation(o, r)
99  }
100}

102pub fn geo_conjuncts(
103  o: GeoSyn,
104  acc: SynExpr,
105  xs: List(geo_notation::Formula_Rep1)
106): SynExpr ! {macro_error, Div} {
107  match xs {
108    Nil => acc
109    Cons(x, rest) => {
110      let b := geo_clause(o, x.clause)
111      geo_conjuncts(o, quote { $acc ∧ $b }, rest)
112    }
113  }
114}
115
116pub fn geo_formula(o: GeoSyn, f: geo_notation::Formula): SynExpr ! {macro_error, Div} {
117  geo_conjuncts(o, geo_clause(o, f.clause), f.rep)
118}

120pub macro fn geo(e: geo_notation::Expr): SynExpr ! {macro_error, Div} {
121  let o := GeoSyn {
122    add: geo_header(e.header),
123    zero: geo_header(e.header_2),
124    neg: geo_header(e.header_3),
125    mul: geo_header(e.header_4),
126    one: geo_header(e.header_5)
127  }
128  geo_formula(o, e.formula)
129}
