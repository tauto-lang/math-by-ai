9// The key identity of the parallelogram, one coordinate at a time. With
10// u = B - A, v = D - A and e = (C - D) - u, which is also (C - B) - v:
11//
12//   e * (u × v) = (u × (C - D)) * v - (v × (C - B)) * u
13//
14// so when AB ∥ DC and AD ∥ BC the right side is 0, and e is 0 when u × v is not.
15pub theorem parallelogram_key_x<K>(
16  add: (K, K) -> K,
17  zero: K,
18  neg: (K) -> K,
19  mul: (K, K) -> K,
20  one: K,
21  r: CommRing(K, add, zero, neg, mul, one),
22  A: Point(K),
23  B: Point(K),
24  C: Point(K),
25  D: Point(K)
26): ring!(```
27    add, zero, neg, mul, one;
28    ((px(C) - px(D)) - (px(B) - px(A))) * ((px(B) - px(A)) * (py(D) - py(A)) - (py(B) - py(A)) * (px(D) - px(A)))
29    = ((px(B) - px(A)) * (py(C) - py(D)) - (py(B) - py(A)) * (px(C) - px(D))) * (px(D) - px(A))
30    - ((px(D) - px(A)) * (py(C) - py(B)) - (py(D) - py(A)) * (px(C) - px(B))) * (px(B) - px(A))
31  ```) {
32  by ring_norm!(r): ring!(```
33    add, zero, neg, mul, one;
34      ((px(C) - px(D)) - (px(B) - px(A))) * ((px(B) - px(A)) * (py(D) - py(A)) - (py(B) - py(A)) * (px(D) - px(A)))
35      = ((px(B) - px(A)) * (py(C) - py(D)) - (py(B) - py(A)) * (px(C) - px(D))) * (px(D) - px(A))
36      - ((px(D) - px(A)) * (py(C) - py(B)) - (py(D) - py(A)) * (px(C) - px(B))) * (px(B) - px(A))
37    ```) {}
38}

40pub theorem parallelogram_key_y<K>(
41  add: (K, K) -> K,
42  zero: K,
43  neg: (K) -> K,
44  mul: (K, K) -> K,
45  one: K,
46  r: CommRing(K, add, zero, neg, mul, one),
47  A: Point(K),
48  B: Point(K),
49  C: Point(K),
50  D: Point(K)
51): ring!(```
52    add, zero, neg, mul, one;
53    ((py(C) - py(D)) - (py(B) - py(A))) * ((px(B) - px(A)) * (py(D) - py(A)) - (py(B) - py(A)) * (px(D) - px(A)))
54    = ((px(B) - px(A)) * (py(C) - py(D)) - (py(B) - py(A)) * (px(C) - px(D))) * (py(D) - py(A))
55    - ((px(D) - px(A)) * (py(C) - py(B)) - (py(D) - py(A)) * (px(C) - px(B))) * (py(B) - py(A))
56  ```) {
57  by ring_norm!(r): ring!(```
58    add, zero, neg, mul, one;
59      ((py(C) - py(D)) - (py(B) - py(A))) * ((px(B) - px(A)) * (py(D) - py(A)) - (py(B) - py(A)) * (px(D) - px(A)))
60      = ((px(B) - px(A)) * (py(C) - py(D)) - (py(B) - py(A)) * (px(C) - px(D))) * (py(D) - py(A))
61      - ((px(D) - px(A)) * (py(C) - py(B)) - (py(D) - py(A)) * (px(C) - px(B))) * (py(B) - py(A))
62    ```) {}
63}

65// `x = 0` and `y = 0` make `x a - y b = 0`; so `e k = x a - y b` with `k ≠ 0` makes `e = 0`.
66pub theorem cancel_combination<K, e: K, k: K, x: K, y: K, a: K, b: K>(
67  add: (K, K) -> K,
68  zero: K,
69  neg: (K) -> K,
70  mul: (K, K) -> K,
71  one: K,
72  d: IntegralDomain(K, add, zero, neg, mul, one),
73  nonzero: (k = zero) -> False,
74  key: mul(e, k) = add(mul(x, a), neg(mul(y, b))),
75  hx: x = zero,
76  hy: y = zero
77): e = zero {
78  let r := domain_ring(add, zero, neg, mul, one, d)
79  domain_cancel(
80    add,
81    zero,
82    neg,
83    mul,
84    one,
85    d,
86    e,
87    zero,
88    k,
89    nonzero,
90    Eq::trans(
91      key,
92      Eq::trans(
93        Eq::congr((z: K) => add(mul(z, a), neg(mul(y, b))), hx),
94        Eq::trans(
95          Eq::congr((z: K) => add(mul(zero, a), neg(mul(z, b))), hy),
96          by ring_norm!(r): add(mul(zero, a), neg(mul(zero, b))) = mul(zero, k) {}
97        )
98      )
99    )
100  )
101}

103// `(c - b) - (d - a) = (c - d) - (b - a)`: the same difference of the sides.
104pub theorem swap_differences<K>(
105  add: (K, K) -> K,
106  zero: K,
107  neg: (K) -> K,
108  mul: (K, K) -> K,
109  one: K,
110  r: CommRing(K, add, zero, neg, mul, one),
111  a: K,
112  b: K,
113  c: K,
114  dd: K
115): ring!(`add, zero, neg, mul, one; (c - b) - (dd - a) = (c - dd) - (b - a)`) {
116  by ring_norm!(r): ring!(`add, zero, neg, mul, one; (c - b) - (dd - a) = (c - dd) - (b - a)`) {}
117}

119pub theorem eq_of_difference<K, a: K, b: K>(
120  add: (K, K) -> K,
121  zero: K,
122  neg: (K) -> K,
123  mul: (K, K) -> K,
124  one: K,
125  r: CommRing(K, add, zero, neg, mul, one),
126  h: add(a, neg(b)) = zero
127): a = b {
128  ring_eq_of_sub(add, zero, neg, mul, one, r, a, b, h)

131// Two vectors with the same coordinates have the same squared length.
132pub theorem same_length<K, p1: K, p2: K, u1: K, u2: K>(
133  add: (K, K) -> K,
134  mul: (K, K) -> K,
135  h1: p1 = u1,
136  h2: p2 = u2
137): add(mul(u1, u1), mul(u2, u2)) = add(mul(p1, p1), mul(p2, p2)) {
138  Eq::sym(
139    Eq::trans(
140      Eq::congr((z: K) => add(mul(z, z), mul(p2, p2)), h1),
141      Eq::congr((z: K) => add(mul(u1, u1), mul(z, z)), h2)
142    )
143  )
144}

146// A quadrilateral ABCD with AB ∥ DC and AD ∥ BC, where A, B, D are not on a
147// line, has its opposite sides of the same length: a parallelogram. In the
148// plane over any integral domain; lengths are squared, as a ring has no roots.
149pub theorem parallelogram_opposite_sides<K>(
150  add: (K, K) -> K,
151  zero: K,
152  neg: (K) -> K,
153  mul: (K, K) -> K,
154  one: K,
155  d: IntegralDomain(K, add, zero, neg, mul, one),
156  A: Point(K),
157  B: Point(K),
158  C: Point(K),
159  D: Point(K),
160  ab_dc: geo!(`add, zero, neg, mul, one; AB ∥ DC`),
161  ad_bc: geo!(`add, zero, neg, mul, one; AD ∥ BC`),
162  proper: geo!(`add, zero, neg, mul, one; ¬(AB ∥ AD)`)
163): geo!(`add, zero, neg, mul, one; AB ≅ DC ∧ AD ≅ BC`) {
164  let r := domain_ring(add, zero, neg, mul, one, d)
165  // (C - D) - (B - A) = 0, each coordinate.
166  let ex := cancel_combination(
167    add,
168    zero,
169    neg,
170    mul,
171    one,
172    d,
173    proper,
174    parallelogram_key_x(add, zero, neg, mul, one, r, A, B, C, D),
175    ab_dc,
176    ad_bc
177  )
178  let ey := cancel_combination(
179    add,
180    zero,
181    neg,
182    mul,
183    one,
184    d,
185    proper,
186    parallelogram_key_y(add, zero, neg, mul, one, r, A, B, C, D),
187    ab_dc,
188    ad_bc
189  )
190  // So also (C - B) - (D - A) = 0.
191  let fx := Eq::trans(swap_differences(add, zero, neg, mul, one, r, px(A), px(B), px(C), px(D)), ex)
192  let fy := Eq::trans(swap_differences(add, zero, neg, mul, one, r, py(A), py(B), py(C), py(D)), ey)
193  AndIntro(
194    same_length(
195      add,
196      mul,
197      eq_of_difference(add, zero, neg, mul, one, r, ex),
198      eq_of_difference(add, zero, neg, mul, one, r, ey)
199    ),
200    same_length(
201      add,
202      mul,
203      eq_of_difference(add, zero, neg, mul, one, r, fx),
204      eq_of_difference(add, zero, neg, mul, one, r, fy)
205    )
206  )
