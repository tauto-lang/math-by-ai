6// A commutative ring: an abelian group under `add`, a commutative monoid under
7// `mul`, and `mul` distributing over `add`. The other laws (`a * 0 = 0`,
8// `(-a) * b = -(a * b)`, distribution on the right) follow; they are below.
9pub(open) inductive CommRing(
10  A: Type,
11  add: (A, A) -> A,
12  zero: A,
13  neg: (A) -> A,
14  mul: (A, A) -> A,
15  one: A
16): Prop {
17  Mk(
18    additive: Group(A, add, zero, neg),
19    add_comm: (a: A, b: A) -> add(a, b) = add(b, a),
20    multiplicative: Monoid(A, mul, one),
21    mul_comm: (a: A, b: A) -> mul(a, b) = mul(b, a),
22    distrib: (a: A, b: A, c: A) -> mul(a, add(b, c)) = add(mul(a, b), mul(a, c))
23  ): CommRing(A, add, zero, neg, mul, one)
24}

26// A commutative ring where `1 ≠ 0` and a product is 0 only when a factor is.
27pub(open) inductive IntegralDomain(
28  A: Type,
29  add: (A, A) -> A,
30  zero: A,
31  neg: (A) -> A,
32  mul: (A, A) -> A,
33  one: A
34): Prop {
35  Mk(
36    ring: CommRing(A, add, zero, neg, mul, one),
37    nontrivial: (one = zero) -> False,
38    no_zero_divisors: (a: A, b: A, h: mul(a, b) = zero) -> a = zero ∨ b = zero
39  ): IntegralDomain(A, add, zero, neg, mul, one)
40}

42pub theorem domain_ring<A>(
43  add: (A, A) -> A,
44  zero: A,
45  neg: (A) -> A,
46  mul: (A, A) -> A,
47  one: A,
48  d: IntegralDomain(A, add, zero, neg, mul, one)
49): CommRing(A, add, zero, neg, mul, one) {
50  ⊢ CommRing(A, add, zero, neg, mul, one) from match d { Mk(r, n, z) => r }
51}

53  add: (A, A) -> A,

55  neg: (A) -> A,
56  mul: (A, A) -> A,
57  one: A,
58  d: IntegralDomain(A, add, zero, neg, mul, one)
59): (one = zero) -> False {
60  ⊢ (one = zero) -> False from match d { Mk(r, n, z) => n }

62pub theorem domain_no_zero_divisors<A>(
63  add: (A, A) -> A,
64  zero: A,
65  neg: (A) -> A,
66  mul: (A, A) -> A,
67  one: A,
68  d: IntegralDomain(A, add, zero, neg, mul, one),
69  a: A,
70  b: A,
71  h: mul(a, b) = zero
72): a = zero ∨ b = zero {
73  ⊢ a = zero ∨ b = zero from match d { Mk(r, n, z) => z(a, b, h) }

120pub theorem ring_distrib_left<A>(

122  zero: A,

134pub theorem ring_add_monoid<A>(
135  add: (A, A) -> A,
136  zero: A,
137  neg: (A) -> A,
138  mul: (A, A) -> A,
139  one: A,
140  r: CommRing(A, add, zero, neg, mul, one)
141): Monoid(A, add, zero) {
142  group_monoid(add, zero, neg, ring_group(add, zero, neg, mul, one, r))
143}
144pub theorem ring_add_comm_monoid<A>(
145  add: (A, A) -> A,
146  zero: A,
147  neg: (A) -> A,
148  mul: (A, A) -> A,
149  one: A,
150  r: CommRing(A, add, zero, neg, mul, one)
151): CommMonoid(A, add, zero) {
152  CommMonoid::Mk(
153    ring_add_monoid(add, zero, neg, mul, one, r),
154    (a: A, b: A) => ring_add_comm(add, zero, neg, mul, one, r, a, b)
155  )
156}
157pub theorem ring_mul_comm_monoid<A>(
158  add: (A, A) -> A,
159  zero: A,
160  neg: (A) -> A,
161  mul: (A, A) -> A,
162  one: A,
163  r: CommRing(A, add, zero, neg, mul, one)
164): CommMonoid(A, mul, one) {
165  CommMonoid::Mk(
166    ring_mul_monoid(add, zero, neg, mul, one, r),
167    (a: A, b: A) => ring_mul_comm(add, zero, neg, mul, one, r, a, b)
168  )

171pub theorem ring_distrib_right<A>(
172  add: (A, A) -> A,
173  zero: A,
174  neg: (A) -> A,
175  mul: (A, A) -> A,
176  one: A,
177  r: CommRing(A, add, zero, neg, mul, one),
178  a: A,
179  b: A,
180  c: A
181): mul(add(a, b), c) = add(mul(a, c), mul(b, c)) {
182  Eq::trans(
183    ring_mul_comm(add, zero, neg, mul, one, r, add(a, b), c),
184    Eq::trans(
185      ring_distrib_left(add, zero, neg, mul, one, r, c, a, b),
186      Eq::trans(
187        Eq::congr(⟦add(□, mul(c, b))⟧, ring_mul_comm(add, zero, neg, mul, one, r, c, a)),
188        Eq::congr(⟦add(mul(a, c), □)⟧, ring_mul_comm(add, zero, neg, mul, one, r, c, b))
189      )
190    )
191  )
192}

194pub theorem ring_mul_zero<A>(
195  add: (A, A) -> A,
196  zero: A,
197  neg: (A) -> A,
198  mul: (A, A) -> A,
199  one: A,
200  r: CommRing(A, add, zero, neg, mul, one),
201  a: A
202): mul(a, zero) = zero {
203  let g := ring_group(add, zero, neg, mul, one, r)
204  let m := ring_add_monoid(add, zero, neg, mul, one, r)
205  // a * 0 + a * 0 = a * (0 + 0) = a * 0 = a * 0 + 0
206  cancel_left(
207    add,
208    zero,
209    neg,
210    g,
211    mul(a, zero),
212    mul(a, zero),
213    zero,
214    Eq::trans(
215      Eq::sym(ring_distrib_left(add, zero, neg, mul, one, r, a, zero, zero)),
216      Eq::trans(
217        Eq::congr(⟦mul(a, □)⟧, left_unit(add, zero, m, zero)),
218        Eq::sym(right_unit(add, zero, m, mul(a, zero)))
219      )
220    )
221  )
222}
223pub theorem ring_zero_mul<A>(
224  add: (A, A) -> A,
225  zero: A,
226  neg: (A) -> A,
227  mul: (A, A) -> A,
228  one: A,
229  r: CommRing(A, add, zero, neg, mul, one),
230  a: A
231): mul(zero, a) = zero {
232  Eq::trans(
233    ring_mul_comm(add, zero, neg, mul, one, r, zero, a),
234    ring_mul_zero(add, zero, neg, mul, one, r, a)
235  )
236}

238// `(-a) * b = -(a * b)`: `a * b + (-a) * b = (a + -a) * b = 0 * b = 0`.
239pub theorem ring_neg_mul<A>(
240  add: (A, A) -> A,
241  zero: A,
242  neg: (A) -> A,
243  mul: (A, A) -> A,
244  one: A,
245  r: CommRing(A, add, zero, neg, mul, one),
246  a: A,
247  b: A
248): mul(neg(a), b) = neg(mul(a, b)) {
249  let g := ring_group(add, zero, neg, mul, one, r)
250  inverse_unique(
251    add,
252    zero,
253    neg,
254    g,
255    mul(a, b),
256    mul(neg(a), b),
257    Eq::trans(
258      Eq::sym(ring_distrib_right(add, zero, neg, mul, one, r, a, neg(a), b)),
259      Eq::trans(
260        Eq::congr(⟦mul(□, b)⟧, inverse_right(add, zero, neg, g, a)),
261        ring_zero_mul(add, zero, neg, mul, one, r, b)
262      )
263    )
264  )
265}
266pub theorem ring_mul_neg<A>(
267  add: (A, A) -> A,
268  zero: A,
269  neg: (A) -> A,
270  mul: (A, A) -> A,
271  one: A,
272  r: CommRing(A, add, zero, neg, mul, one),
273  a: A,
274  b: A
275): mul(a, neg(b)) = neg(mul(a, b)) {
276  Eq::trans(
277    ring_mul_comm(add, zero, neg, mul, one, r, a, neg(b)),
278    Eq::trans(
279      ring_neg_mul(add, zero, neg, mul, one, r, b, a),
280      Eq::congr(⟦neg(□)⟧, ring_mul_comm(add, zero, neg, mul, one, r, b, a))
281    )
282  )
283}
284pub theorem ring_neg_neg<A>(
285  add: (A, A) -> A,
286  zero: A,
287  neg: (A) -> A,
288  mul: (A, A) -> A,
289  one: A,
290  r: CommRing(A, add, zero, neg, mul, one),
291  a: A
292): neg(neg(a)) = a {
293  inverse_involution(add, zero, neg, ring_group(add, zero, neg, mul, one, r), a)

295pub theorem ring_neg_zero<A>(
296  add: (A, A) -> A,

298  neg: (A) -> A,
299  mul: (A, A) -> A,
300  one: A,
301  r: CommRing(A, add, zero, neg, mul, one)

305pub theorem ring_neg_add<A>(
306  add: (A, A) -> A,
307  zero: A,
308  neg: (A) -> A,
309  mul: (A, A) -> A,
310  one: A,
311  r: CommRing(A, add, zero, neg, mul, one),
312  a: A,
313  b: A
314): neg(add(a, b)) = add(neg(a), neg(b)) {
315  Eq::trans(
316    inverse_product(add, zero, neg, ring_group(add, zero, neg, mul, one, r), a, b),
317    ring_add_comm(add, zero, neg, mul, one, r, neg(b), neg(a))

321// `a + -b = 0` makes `a = b`.
322pub theorem ring_eq_of_sub<A>(
323  add: (A, A) -> A,
324  zero: A,
325  neg: (A) -> A,
326  mul: (A, A) -> A,
327  one: A,
328  r: CommRing(A, add, zero, neg, mul, one),
329  a: A,
330  b: A,
331  h: add(a, neg(b)) = zero
332): a = b {
333  let g := ring_group(add, zero, neg, mul, one, r)
334  // -b = -a, so a = --a = --b = b.
335  Eq::trans(
336    Eq::sym(ring_neg_neg(add, zero, neg, mul, one, r, a)),
337    Eq::trans(
338      Eq::congr(⟦neg(□)⟧, Eq::sym(inverse_unique(add, zero, neg, g, a, neg(b), h))),
339      ring_neg_neg(add, zero, neg, mul, one, r, b)
340    )
341  )
342}

344// In a domain, a nonzero factor cancels: `a * k = b * k` and `k ≠ 0` make `a = b`.
345pub theorem domain_cancel<A>(
346  add: (A, A) -> A,
347  zero: A,
348  neg: (A) -> A,
349  mul: (A, A) -> A,
350  one: A,
351  d: IntegralDomain(A, add, zero, neg, mul, one),
352  a: A,
353  b: A,
354  k: A,
355  nonzero: (k = zero) -> False,
356  h: mul(a, k) = mul(b, k)
357): a = b {
358  let r := domain_ring(add, zero, neg, mul, one, d)
359  let g := ring_group(add, zero, neg, mul, one, r)
360  // (a + -b) * k = a * k + -(b * k) = b * k + -(b * k) = 0
361  let product: mul(
362    add(a, neg(b)),
363    k
364  ) = zero := Eq::trans(
365    ring_distrib_right(add, zero, neg, mul, one, r, a, neg(b), k),
366    Eq::trans(
367      Eq::congr(⟦add(mul(a, k), □)⟧, ring_neg_mul(add, zero, neg, mul, one, r, b, k)),
368      Eq::trans(Eq::congr(⟦add(□, neg(mul(b, k)))⟧, h), inverse_right(add, zero, neg, g, mul(b, k)))
369    )
370  )
371  ⊢ a = b from match domain_no_zero_divisors(add, zero, neg, mul, one, d, add(a, neg(b)), k, product) {
372    OrInl(diff) => ring_eq_of_sub(add, zero, neg, mul, one, r, a, b, diff)
373    OrInr(kz) => False::elim(nonzero(kz))
374  }
