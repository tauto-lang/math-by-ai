1import std::Base
2import std::prelude::Macro { Judgement, ProofStep, Syn, SynExpr, SynSpan, Elab, elab, macro_error }
3import std::Rewrite { Tac }
4import "ring_norm.taut"
5
6// The operations of a ring, as the checker writes them.
7pub(open) struct RingOps { add: SynExpr, zero: SynExpr, neg: SynExpr, mul: SynExpr, one: SynExpr }
8
9module RingTac {
10  // Where `e` is among the variables met so far.
11  pub fn index(vars: List(SynExpr), e: SynExpr, i: u64): Option(u64) ! {Div} {
12    match vars {
13      Nil => None
14      Cons(v, rest) => if Syn::same(v, e) { Some(i) } else { RingTac::index(rest, e, i + 1) }
15    }
16  }
17
18  pub fn length(vars: List(SynExpr)): u64 ! {Div} {
19    match vars {
20      Nil => 0
21      Cons(_, rest) => 1 + RingTac::length(rest)
22    }
23  }
24
25  pub fn snoc(vars: List(SynExpr), e: SynExpr): List(SynExpr) ! {Div} {
26    match vars {
27      Nil => Cons(e, Nil)
28      Cons(v, rest) => Cons(v, RingTac::snoc(rest, e))
29    }
30  }
31
32  // `e` as a variable: its place, after the ones met so far when it is new.
33  pub fn variable(e: SynExpr, vars: List(SynExpr), at: SynSpan): SynExpr × List(SynExpr) ! {Div} {
34    match RingTac::index(vars, e, 0) {
35      Some(i) => {
36        let k := ExNat(Text::of_u64(i), at)
37        #(quote { ReVar($k) }, vars)
38      }
39      None => {
40        let k := ExNat(Text::of_u64(RingTac::length(vars)), at)
41        #(quote { ReVar($k) }, RingTac::snoc(vars, e))
42      }
43    }
44  }
45
46  pub fn two(
47    ops: RingOps,
48    c: SynExpr,
49    a: SynExpr,
50    b: SynExpr,
51    vars: List(SynExpr),
52    at: SynSpan
53  ): SynExpr × List(SynExpr) ! {Div} {
54    let x := RingTac::read(ops, a, vars, at)
55    let y := RingTac::read(ops, b, (x: List(SynExpr)), at)
56    let u := (x: SynExpr)
57    let v := (y: SynExpr)
58    #(quote { $c($u, $v) }, (y: List(SynExpr)))
59  }
60
61  // The `RingExpr` of `e`: the ring's operations and constants, and anything
62  // else a variable.
63  pub fn read(
64    ops: RingOps,
65    e: SynExpr,
66    vars: List(SynExpr),
67    at: SynSpan
68  ): SynExpr × List(SynExpr) ! {Div} {
69    if Syn::same(e, ops.zero) {
70      #(quote { ReZero }, vars)
71    } else if Syn::same(e, ops.one) {
72      #(quote { ReOne }, vars)
73    } else {
74      match Syn::app_parts(e) {
75        None => RingTac::variable(e, vars, at)
76        Some(p) => {
77          let f := (p: SynExpr)
78          match (p: List(SynExpr)) {
79            Nil => RingTac::variable(e, vars, at)
80            Cons(a, rest) => match rest {
81              Nil => if Syn::same(f, ops.neg) {
82                let x := RingTac::read(ops, a, vars, at)
83                let u := (x: SynExpr)
84                #(quote { ReNeg($u) }, (x: List(SynExpr)))
85              } else {
86                RingTac::variable(e, vars, at)
87              }
88              Cons(b, more) => match more {
89                Nil => if Syn::same(f, ops.add) {
90                  RingTac::two(ops, quote { ReAdd }, a, b, vars, at)
91                } else if Syn::same(f, ops.mul) {
92                  RingTac::two(ops, quote { ReMul }, a, b, vars, at)
93                } else {
94                  RingTac::variable(e, vars, at)
95                }
96                Cons(_, _) => RingTac::variable(e, vars, at)
97              }
98            }
99          }
100        }
101      }
102    }
103  }
104
105  pub fn list(vars: List(SynExpr)): SynExpr ! {Div} {
106    match vars {
107      Nil => quote { Nil }
108      Cons(v, rest) => {
109        let tail := RingTac::list(rest)
110        quote { Cons($v, $tail) }
111      }
112    }
113  }
114
115  // The operations from the type of the proof that the structure is a ring,
116  // `CommRing(A, add, zero, neg, mul, one)`.
117  pub fn ops(ty: SynExpr): Option(RingOps) {
118    match Syn::app_parts(ty) {
119      None => None
120      Some(p) => match (p: List(SynExpr)) {
121        Cons(_, Cons(add, Cons(zero, Cons(neg, Cons(mul, Cons(one, Nil)))))) => Some(
122          RingOps { add: add, zero: zero, neg: neg, mul: mul, one: one }
123        )
124        _ => None
125      }
126    }
127  }
128}

130// `ring_norm!(r)`, for a goal `l = r` in a commutative ring: `r` proves
131// `CommRing(A, add, zero, neg, mul, one)`, and the sides are made of those
132// operations and anything else (a variable of the equation). Reflection: the
133// term is `ring_norm_eq(…, Refl)`, whose `Refl` the kernel accepts only when
134// `l + -r` normalises to no term. No goal is left: write no arm.
135pub macro fn ring_norm(s: ProofStep): SynExpr ! {macro_error, elab, Div} {
136  match s.after {
137    Cons(_, _) => Syn::error(s.span, "ring_norm! leaves no goal: write no arm")
138    Nil => match Elab::check(s.arg, s.before, None) {
139      Err(d) => Syn::error(s.span, d.message)
140      Ok(j) => match RingTac::ops(Judgement::term(Judgement::type_of(j))) {
141        None => Syn::error(
142          s.span,
143          "ring_norm!: the argument does not prove CommRing(A, add, zero, neg, mul, one)"
144        )
145        Some(ops) => match Tac::eq_goal(s.before) {
146          None => Syn::error(s.span, "ring_norm!: the goal is not an equation")
147          Some(#(@lhs l, @rhs r, @ty ty)) => {
148            let at := s.span
149            let x := RingTac::read(ops, l, Nil, at)
150            let y := RingTac::read(ops, r, (x: List(SynExpr)), at)
151            let env := RingTac::list((y: List(SynExpr)))
152            let a := (x: SynExpr)
153            let b := (y: SynExpr)
154            let add := ops.add
155            let zero := ops.zero
156            let neg := ops.neg
157            let mul := ops.mul
158            let one := ops.one
159            let proof := s.arg
160            quote { ring_norm_eq($add, $zero, $neg, $mul, $one, $proof, $env, $a, $b, Refl) }
161          }
162        }
163      }
164    }
165  }
