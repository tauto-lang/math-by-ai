39// The operations, in the order of the header.
40pub(open) struct RingSyn { add: SynExpr, zero: SynExpr, neg: SynExpr, mul: SynExpr, one: SynExpr }
41
42pub fn ring_header(h: ring_notation::Header): SynExpr {
43  match h {
44    Name(t) => ExVar(Syn::ident(t))
45    Embed(e) => e
46  }
47}
48pub fn ring_atom(t: ring_notation::Atom, o: RingSyn): SynExpr ! {macro_error, Div} {
49  match t {
50    ZeroLit(tok) => o.zero
51    OneLit(tok) => o.one
52    Embed(e) => e
53    Paren(p) => ring_sum(p.sum, o)
54    Reference(r) => match r.arguments {
55      None => ExVar(Syn::ident(r.name))
56      Some(args) => ExApp(
57        ExVar(Syn::ident(r.name)),
58        Cons(ring_sum(args.sum, o), ring_args(args.rep, o)),
59        r.span
60      )
61    }
62  }
63}
64pub fn ring_args(
65  xs: List(ring_notation::Arguments_Rep1),
66  o: RingSyn
67): List(SynExpr) ! {macro_error, Div} {
68  match xs {
69    Nil => Nil
70    Cons(x, rest) => Cons(ring_sum(x.sum, o), ring_args(rest, o))
71  }
72}
73pub fn ring_unary(u: ring_notation::Unary, o: RingSyn): SynExpr ! {macro_error, Div} {
74  match u {
75    Negated(n) => {
76      let neg := o.neg
77      let a := ring_unary(n.unary, o)
78      quote { $neg($a) }
79    }
80    Atom(a) => ring_atom(a, o)
81  }
82}
83pub fn ring_factors(
84  acc: SynExpr,
85  xs: List(ring_notation::Prod_Rep1),
86  o: RingSyn
87): SynExpr ! {macro_error, Div} {
88  match xs {
89    Nil => acc
90    Cons(x, rest) => {
91      let mul := o.mul
92      let b := ring_unary(x.unary, o)
93      ring_factors(quote { $mul($acc, $b) }, rest, o)
94    }
95  }
96}
97pub fn ring_product(p: ring_notation::Prod, o: RingSyn): SynExpr ! {macro_error, Div} {
98  ring_factors(ring_unary(p.unary, o), p.rep, o)
99}
100pub fn ring_summands(
101  acc: SynExpr,
102  xs: List(ring_notation::Sum_Rep1),
103  o: RingSyn
104): SynExpr ! {macro_error, Div} {
105  match xs {
106    Nil => acc
107    Cons(x, rest) => {
108      let add := o.add
109      let neg := o.neg
110      let b := ring_product(x.prod, o)
111      let next := match x.add_op {
112        Plus(t) => quote { $add($acc, $b) }
113        Minus(t) => quote { $add($acc, $neg($b)) }
114      }
115      ring_summands(next, rest, o)
116    }
117  }
118}
119pub fn ring_sum(s: ring_notation::Sum, o: RingSyn): SynExpr ! {macro_error, Div} {
120  ring_summands(ring_product(s.prod, o), s.rep, o)
121}
122pub macro fn ring(e: ring_notation::Expr): SynExpr ! {macro_error, Div} {
123  let o := RingSyn {
124    add: ring_header(e.header),
125    zero: ring_header(e.header_2),
126    neg: ring_header(e.header_3),
127    mul: ring_header(e.header_4),
128    one: ring_header(e.header_5)
129  }
130  let left := ring_sum(e.formula.sum, o)
131  match e.formula.opt {
132    None => left
133    Some(r) => {
134      let right := ring_sum(r.sum, o)
135      quote { $left = $right }
136    }
137  }
