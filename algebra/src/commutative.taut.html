9pub(open) inductive CommMonoid(A: Type, op: (A, A) -> A, unit: A): Prop {
10  Mk(
11    monoid: Monoid(A, op, unit),
12    comm: (a: A, b: A) -> alg!(`op, unit; a ⋆ b = b ⋆ a`)
13  ): CommMonoid(A, op, unit)
14}
15pub theorem comm_monoid<A>(
16  op: (A, A) -> A,
17  unit: A,
18  m: CommMonoid(A, op, unit)
19): Monoid(A, op, unit) {
20  ⊢ Monoid(A, op, unit) from match m { Mk(base, comm) => base }
21}
22pub theorem commute<A>(
23  op: (A, A) -> A,
24  unit: A,
25  m: CommMonoid(A, op, unit),
26  a: A,
27  b: A
28): alg!(`op, unit; a ⋆ b = b ⋆ a`) {
29  ⊢ alg!(`op, unit; a ⋆ b = b ⋆ a`) from match m { Mk(base, comm) => comm(a, b) }
30}
31pub theorem left_commute<A>(
32  op: (A, A) -> A,
33  unit: A,
34  m: CommMonoid(A, op, unit),
35  a: A,
36  b: A,
37  c: A
38): alg!(`op, unit; a ⋆ (b ⋆ c) = b ⋆ (a ⋆ c)`) {
39  let base := comm_monoid(op, unit, m)
40  calc!(```
41    $alg!(`op, unit; a ⋆ (b ⋆ c)`)
42    = $alg!(`op, unit; a ⋆ b ⋆ c`) from $Eq::sym(assoc(op, unit, base, a, b, c))
43    = $alg!(`op, unit; b ⋆ a ⋆ c`) from $Eq::congr(
44      (x: A) => alg!(`op, unit; x ⋆ c`),
45      commute(op, unit, m, a, b)
46    )
47    = $alg!(`op, unit; b ⋆ (a ⋆ c)`) from $assoc(op, unit, base, b, a, c)
48    ```)
49}
50pub theorem interchange<A>(
51  op: (A, A) -> A,
52  unit: A,
53  m: CommMonoid(A, op, unit),
54  a: A,
55  b: A,
56  c: A,
57  d: A
58): alg!(`op, unit; a ⋆ b ⋆ (c ⋆ d) = a ⋆ c ⋆ (b ⋆ d)`) {
59  proof!(```
60      We have $alg!(`op, unit; a ⋆ b ⋆ (c ⋆ d)`)
61        = $alg!(`op, unit; a ⋆ (b ⋆ (c ⋆ d))`) by assoc(op, unit, comm_monoid(op, unit, m), a, b, op(c, d))
62        = $alg!(`op, unit; a ⋆ (c ⋆ (b ⋆ d))`) by left_commute(op, unit, m, b, c, d)
63        = $alg!(`op, unit; a ⋆ c ⋆ (b ⋆ d)`) by assoc(op, unit, comm_monoid(op, unit, m), a, c, op(b, d)).
64    ```)
65}

67// Generated by adjacent swaps, compatible heads, and transitivity.
68pub(open) inductive Permutation(A: Type): (List(A)) -> (List(A)) -> Prop {
69  Empty: Permutation(A, Nil, Nil)
70  Keep(
71    x: A,
72    xs: List(A),
73    ys: List(A),
74    p: Permutation(A, xs, ys)
75  ): Permutation(A, Cons(x, xs), Cons(x, ys))
76  Swap(x: A, y: A, xs: List(A)): Permutation(A, Cons(x, Cons(y, xs)), Cons(y, Cons(x, xs)))
77  Chain(
78    xs: List(A),
79    ys: List(A),
80    zs: List(A),
81    p: Permutation(A, xs, ys),
82    q: Permutation(A, ys, zs)
83  ): Permutation(A, xs, zs)
84}
85pub theorem product_permutation<A>(
86  op: (A, A) -> A,
87  unit: A,
88  m: CommMonoid(A, op, unit),
89  xs: List(A),
90  ys: List(A),
91  p: Permutation(A, xs, ys)
92): alg!(`op, unit; product(op, ε, xs) = product(op, ε, ys)`) {
93  ⊢ alg!(`op, unit; product(op, ε, xs) = product(op, ε, ys)`) from match p {
94    Empty => Refl
95    Keep(x, as, bs, h) => Eq::congr(
96      (v: A) => alg!(`op, unit; x ⋆ v`),
97      product_permutation(op, unit, m, as, bs, h)
98    )
99    Swap(x, y, rest) => left_commute(op, unit, m, x, y, product(op, unit, rest))
100    Chain(as, bs, cs, p2, q2) => calc!(```
101      $alg!(`op, unit; product(op, ε, as)`)
102      = $alg!(`op, unit; product(op, ε, bs)`) from $product_permutation(op, unit, m, as, bs, p2)
103      = $alg!(`op, unit; product(op, ε, cs)`) from $product_permutation(op, unit, m, bs, cs, q2)
104      ```)
105  }
106}
107pub theorem power_product<A>(
108  op: (A, A) -> A,
109  unit: A,
110  m: CommMonoid(A, op, unit),
111  a: A,
112  b: A,
113  n: Nat
114): alg!(`op, unit; power(op, ε, a ⋆ b, n) = power(op, ε, a, n) ⋆ power(op, ε, b, n)`) {
115  let base := comm_monoid(op, unit, m)
116  proof!(```
117      By induction on n.
118        [zero] Case Zero. By Eq::sym.
119        Case Succ(k). We have $alg!(`op, unit; power(op, ε, a ⋆ b, k) ⋆ (a ⋆ b)`)
120        = $alg!(`op, unit; (power(op, ε, a, k) ⋆ power(op, ε, b, k)) ⋆ (a ⋆ b)`) by the induction hypothesis
121        = $alg!(`op, unit; power(op, ε, a, k) ⋆ a ⋆ (power(op, ε, b, k) ⋆ b)`) by interchange(op, unit, m, power(op, unit, a, k), power(op, unit, b, k), a, b).
122      %%
123      [zero] left_unit(op, unit, base, unit)
124    ```)
125}

127pub theorem permutation_refl<A>(xs: List(A)): Permutation(A, xs, xs) {
128  ⊢ Permutation(A, xs, xs) from match xs {
129    Nil => Empty
130    Cons(x, rest) => Keep(x, rest, rest, permutation_refl(rest))
131  }
132}
133pub theorem permutation_sym<A>(
134  xs: List(A),
135  ys: List(A),
136  p: Permutation(A, xs, ys)
137): Permutation(A, ys, xs) {
138  ⊢ Permutation(A, ys, xs) from match p {
139    Empty => Empty
140    Keep(x, as, bs, h) => Keep(x, bs, as, permutation_sym(as, bs, h))
141    Swap(x, y, rest) => Swap(y, x, rest)
142    Chain(as, bs, cs, h, k) => Chain(
143      cs,
144      bs,
145      as,
146      permutation_sym(bs, cs, k),
147      permutation_sym(as, bs, h)
148    )
149  }
