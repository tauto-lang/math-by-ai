9// No commutativity assumption: all theorems apply to nonabelian groups.
10pub(open) inductive Group(A: Type, op: (A, A) -> A, unit: A, inv: (A) -> A): Prop {
11  Mk(
12    monoid: Monoid(A, op, unit),
13    left_inverse: (a: A) -> alg!(`op, unit; inv(a) ⋆ a = ε`),
14    right_inverse: (a: A) -> alg!(`op, unit; a ⋆ inv(a) = ε`)
15  ): Group(A, op, unit, inv)
16}
17pub theorem group_monoid<A>(
18  op: (A, A) -> A,
19  unit: A,
20  inv: (A) -> A,
21  g: Group(A, op, unit, inv)
22): Monoid(A, op, unit) {
23  ⊢ Monoid(A, op, unit) from match g { Mk(m, l, r) => m }
24}
25pub theorem inverse_left<A>(
26  op: (A, A) -> A,
27  unit: A,
28  inv: (A) -> A,
29  g: Group(A, op, unit, inv),
30  a: A
31): alg!(`op, unit; inv(a) ⋆ a = ε`) {
32  ⊢ alg!(`op, unit; inv(a) ⋆ a = ε`) from match g { Mk(m, l, r) => l(a) }
33}
34pub theorem inverse_right<A>(
35  op: (A, A) -> A,
36  unit: A,
37  inv: (A) -> A,
38  g: Group(A, op, unit, inv),
39  a: A
40): alg!(`op, unit; a ⋆ inv(a) = ε`) {
41  ⊢ alg!(`op, unit; a ⋆ inv(a) = ε`) from match g { Mk(m, l, r) => r(a) }
42}
43pub theorem undo_left<A>(
44  op: (A, A) -> A,
45  unit: A,
46  inv: (A) -> A,
47  g: Group(A, op, unit, inv),
48  a: A,
49  b: A
50): alg!(`op, unit; inv(a) ⋆ (a ⋆ b) = b`) {
51  let m := group_monoid(op, unit, inv, g)
52  calc!(```
53    $alg!(`op, unit; inv(a) ⋆ (a ⋆ b)`)
54    = $alg!(`op, unit; inv(a) ⋆ a ⋆ b`) from $Eq::sym(assoc(op, unit, m, inv(a), a, b))
55    = $alg!(`op, unit; ε ⋆ b`) from $Eq::congr(
56      (x: A) => alg!(`op, unit; x ⋆ b`),
57      inverse_left(op, unit, inv, g, a)
58    )
59    = $b from $left_unit(op, unit, m, b)
60    ```)
61}
62pub theorem undo_right<A>(
63  op: (A, A) -> A,
64  unit: A,
65  inv: (A) -> A,
66  g: Group(A, op, unit, inv),
67  a: A,
68  b: A
69): alg!(`op, unit; b ⋆ a ⋆ inv(a) = b`) {
70  let m := group_monoid(op, unit, inv, g)
71  calc!(```
72    $alg!(`op, unit; b ⋆ a ⋆ inv(a)`)
73    = $alg!(`op, unit; b ⋆ (a ⋆ inv(a))`) from $assoc(op, unit, m, b, a, inv(a))
74    = $alg!(`op, unit; b ⋆ ε`) from $Eq::congr(
75      (x: A) => alg!(`op, unit; b ⋆ x`),
76      inverse_right(op, unit, inv, g, a)
77    )
78    = $b from $right_unit(op, unit, m, b)
79    ```)
80}
81pub theorem cancel_left<A>(
82  op: (A, A) -> A,
83  unit: A,
84  inv: (A) -> A,
85  g: Group(A, op, unit, inv),
86  a: A,
87  b: A,
88  c: A,
89  h: alg!(`op, unit; a ⋆ b = a ⋆ c`)
90): b = c {
91  proof!(```
92      [1] We have b
93        = $alg!(`op, unit; inv(a) ⋆ (a ⋆ b)`) by undo_left(op, unit, inv, g, a, b)
94        = $alg!(`op, unit; inv(a) ⋆ (a ⋆ c)`) by h
95        = c by undo_left(op, unit, inv, g, a, c).
96    ```)
97}
98pub theorem cancel_right<A>(
99  op: (A, A) -> A,
100  unit: A,
101  inv: (A) -> A,
102  g: Group(A, op, unit, inv),
103  a: A,
104  b: A,
105  c: A,
106  h: alg!(`op, unit; b ⋆ a = c ⋆ a`)
107): b = c {
108  proof!(```
109      [1] We have b
110        = $alg!(`op, unit; b ⋆ a ⋆ inv(a)`) by undo_right(op, unit, inv, g, a, b)
111        = $alg!(`op, unit; c ⋆ a ⋆ inv(a)`) by h
112        = c by undo_right(op, unit, inv, g, a, c).
113    ```)
114}
115pub theorem inverse_unique<A>(
116  op: (A, A) -> A,
117  unit: A,
118  inv: (A) -> A,
119  g: Group(A, op, unit, inv),
120  a: A,
121  b: A,
122  h: alg!(`op, unit; a ⋆ b = ε`)
123): b = inv(a) {
124  proof!(```
125      [1] We have k: $alg!(`op, unit; a ⋆ b`)
126        = $alg!(`op, unit; ε`) by h
127        = $alg!(`op, unit; a ⋆ inv(a)`) by inverse_right(op, unit, inv, g, a). By cancel_left(op, unit, inv, g, a, b, inv(a), k).
128    ```)
129}
130pub theorem inverse_involution<A>(
131  op: (A, A) -> A,
132  unit: A,
133  inv: (A) -> A,
134  g: Group(A, op, unit, inv),
135  a: A
136): logic!(`inv(inv(a)) = a`) {
137  proof!(```
138      [1] By Eq::sym.
139      %%
140      [1] inverse_unique(op, unit, inv, g, inv(a), a, inverse_left(op, unit, inv, g, a))
141    ```)
142}
143pub theorem inverse_unit<A>(
144  op: (A, A) -> A,
145  unit: A,
146  inv: (A) -> A,
147  g: Group(A, op, unit, inv)
148): logic!(`inv(unit) = unit`) {
149  proof!(```
150      [1] By Eq::sym.
151      %%
152      [1] inverse_unique(op, unit, inv, g, unit, unit, left_unit(op, unit, group_monoid(op, unit, inv, g), unit))
153    ```)
154}
155pub theorem inverse_product<A>(
156  op: (A, A) -> A,
157  unit: A,
158  inv: (A) -> A,
159  g: Group(A, op, unit, inv),
160  a: A,
161  b: A
162): alg!(`op, unit; inv(a ⋆ b) = inv(b) ⋆ inv(a)`) {
163  proof!(```
164      [main] We have h: $alg!(`op, unit; (a ⋆ b) ⋆ (inv(b) ⋆ inv(a))`)
165        = $alg!(`op, unit; a ⋆ b ⋆ inv(b) ⋆ inv(a)`) by assoc(op, unit, group_monoid(op, unit, inv, g), op(a, b), inv(b), inv(a))
166        = $alg!(`op, unit; a ⋆ inv(a)`) by undo_right(op, unit, inv, g, b, a)
167        = $alg!(`op, unit; ε`) by inverse_right(op, unit, inv, g, a). By Eq::sym.
168      %%
169      [main] inverse_unique(op, unit, inv, g, op(a, b), op(inv(b), inv(a)), h)
170    ```)
171}

173pub theorem solve_left<A>(
174  op: (A, A) -> A,
175  unit: A,
176  inv: (A) -> A,
177  g: Group(A, op, unit, inv),
178  a: A,
179  b: A
180): alg!(`op, unit; a ⋆ (inv(a) ⋆ b) = b`) {
181  proof!(```
182      [1] We have $alg!(`op, unit; a ⋆ (inv(a) ⋆ b)`)
183        = $alg!(`op, unit; a ⋆ inv(a) ⋆ b`) by assoc(op, unit, group_monoid(op, unit, inv, g), a, inv(a), b)
184        = $alg!(`op, unit; ε ⋆ b`) by inverse_right(op, unit, inv, g, a)
185        = b by left_unit(op, unit, group_monoid(op, unit, inv, g), b).
186    ```)

188pub theorem solution_unique<A>(
189  op: (A, A) -> A,
190  unit: A,
191  inv: (A) -> A,
192  g: Group(A, op, unit, inv),
193  a: A,
194  b: A,
195  x: A,
196  h: alg!(`op, unit; a ⋆ x = b`)
197): alg!(`op, unit; x = inv(a) ⋆ b`) {
198  proof!(```
199      [1] We have x
200        = $alg!(`op, unit; inv(a) ⋆ (a ⋆ x)`) by undo_left(op, unit, inv, g, a, x)
201        = $alg!(`op, unit; inv(a) ⋆ b`) by h.
202    ```)
