8// Reflection for the equations of a commutative ring. An expression of `add`,
9// `neg`, `mul`, `zero`, `one` and variables is read as a `RingExpr`, and its
10// normal form is a list of signed monomials (`RingTerm`), each a sorted list of
11// variables (`Nm::ins`, `Nm::mul`, as for `nat_norm!`). The list is sorted by
12// monomial (`Nm::cmp`), and a term put in next to the same monomial with the
13// other sign cancels it, so the terms of one monomial all have the same sign:
14// the normal form is the polynomial with integer coefficients, written out.
15// `l = r` holds when `l + -r` has no term (`ring_norm_eq`); the theorem is the
16// whole trust, and `ring_norm!` only finds the expressions.
17
18pub(open) inductive RingExpr: Type {
19  ReVar(i: Nat): RingExpr
20  ReZero: RingExpr
21  ReOne: RingExpr
22  ReAdd(a: RingExpr, b: RingExpr): RingExpr
23  ReMul(a: RingExpr, b: RingExpr): RingExpr
24  ReNeg(a: RingExpr): RingExpr
25}
26
27// `m`, or `-m` when `minus`.
28pub(open) inductive RingTerm: Type {
29  Rt(minus: Bool, m: List(Nat)): RingTerm
30}

32module Rg {
33  // The value of variable `i`; the variables not in the list are 0.
34  pub fn get<A>(zero: A, env: List(A), i: Nat): A {
35    ⊢ A from match env {
36      Nil => zero
37      Cons(h, t) => ⊢ A from match i {
38        Zero => h
39        Succ(k) => Rg::get(zero, t, k)
40      }
41    }
42  }
43
44  pub fn eval<A>(
45    add: (A, A) -> A,
46    zero: A,
47    neg: (A) -> A,
48    mul: (A, A) -> A,
49    one: A,
50    env: List(A),
51    e: RingExpr
52  ): A {
53    ⊢ A from match e {
54      ReVar(i) => Rg::get(zero, env, i)
55      ReZero => zero
56      ReOne => one
57      ReAdd(a, b) => add(
58        Rg::eval(add, zero, neg, mul, one, env, a),
59        Rg::eval(add, zero, neg, mul, one, env, b)
60      )
61      ReMul(a, b) => mul(
62        Rg::eval(add, zero, neg, mul, one, env, a),
63        Rg::eval(add, zero, neg, mul, one, env, b)
64      )
65      ReNeg(a) => neg(Rg::eval(add, zero, neg, mul, one, env, a))
66    }
67  }
68
69  // The product of the variables of a monomial.
70  pub fn mono<A>(zero: A, mul: (A, A) -> A, one: A, env: List(A), m: List(Nat)): A {
71    ⊢ A from match m {
72      Nil => one
73      Cons(v, r) => mul(Rg::get(zero, env, v), Rg::mono(zero, mul, one, env, r))
74    }
75  }
76
77  pub fn sign<A>(neg: (A) -> A, s: Bool, x: A): A {
78    ⊢ A from match s {
79      false => x
80      true => neg(x)
81    }
82  }
83
84  pub fn poly<A>(
85    add: (A, A) -> A,
86    zero: A,
87    neg: (A) -> A,
88    mul: (A, A) -> A,
89    one: A,
90    env: List(A),
91    p: List(RingTerm)
92  ): A {
93    ⊢ A from match p {
94      Nil => zero
95      Cons(t, r) => ⊢ A from match t {
96        Rt(s, m) => add(
97          Rg::sign(neg, s, Rg::mono(zero, mul, one, env, m)),
98          Rg::poly(add, zero, neg, mul, one, env, r)
99        )
100      }
101    }
102  }
103
104  pub fn flip(s: Bool): Bool {
105    ⊢ Bool from match s { false => true, true => false }
106  }
107
108  pub fn xor(s: Bool, d: Bool): Bool {
109    ⊢ Bool from match s {
110      false => d
111      true => Rg::flip(d)
112    }
113  }
114
115  // The same monomial twice: the two cancel when the signs differ.
116  pub fn ins_eq(s: Bool, d: Bool, m: List(Nat), n: List(Nat), p2: List(RingTerm)): List(RingTerm) {
117    ⊢ List(RingTerm) from match s {
118      false => ⊢ List(RingTerm) from match d {
119        false => Cons(Rt(false, m), Cons(Rt(false, n), p2))
120        true => p2
121      }
122      true => ⊢ List(RingTerm) from match d {
123        false => p2
124        true => Cons(Rt(true, m), Cons(Rt(true, n), p2))
125      }
126    }
127  }
128
129  pub fn ins_at(
130    o: Ordering,
131    s: Bool,
132    m: List(Nat),
133    d: Bool,
134    n: List(Nat),
135    p2: List(RingTerm),
136    rest: List(RingTerm)
137  ): List(RingTerm) {
138    ⊢ List(RingTerm) from match o {
139      Less => Cons(Rt(s, m), Cons(Rt(d, n), p2))
140      Equal => Rg::ins_eq(s, d, m, n, p2)
141      Greater => Cons(Rt(d, n), rest)
142    }
143  }
144
145  // The term `Rt(s, m)` put in, before the first term whose monomial is not before `m`.
146  pub fn ins(s: Bool, m: List(Nat), p: List(RingTerm)): List(RingTerm) {
147    ⊢ List(RingTerm) from match p {
148      Nil => Cons(Rt(s, m), Nil)
149      Cons(u, p2) => ⊢ List(RingTerm) from match u {
150        Rt(d, n) => Rg::ins_at(Nm::cmp(m, n), s, m, d, n, p2, Rg::ins(s, m, p2))
151      }
152    }
153  }
154
155  pub fn add(p: List(RingTerm), q: List(RingTerm)): List(RingTerm) {
156    ⊢ List(RingTerm) from match p {
157      Nil => q
158      Cons(t, r) => ⊢ List(RingTerm) from match t {
159        Rt(s, m) => Rg::ins(s, m, Rg::add(r, q))
160      }
161    }
162  }
163
164  pub fn neg(p: List(RingTerm)): List(RingTerm) {
165    ⊢ List(RingTerm) from match p {
166      Nil => Nil
167      Cons(t, r) => ⊢ List(RingTerm) from match t {
168        Rt(s, m) => Cons(Rt(Rg::flip(s), m), Rg::neg(r))
169      }
170    }
171  }
172
173  pub fn mul_mono(s: Bool, m: List(Nat), q: List(RingTerm)): List(RingTerm) {
174    ⊢ List(RingTerm) from match q {
175      Nil => Nil
176      Cons(u, r) => ⊢ List(RingTerm) from match u {
177        Rt(d, n) => Rg::ins(Rg::xor(s, d), Nm::mul(m, n), Rg::mul_mono(s, m, r))
178      }
179    }
180  }
181
182  pub fn mul(p: List(RingTerm), q: List(RingTerm)): List(RingTerm) {
183    ⊢ List(RingTerm) from match p {
184      Nil => Nil
185      Cons(t, r) => ⊢ List(RingTerm) from match t {
186        Rt(s, m) => Rg::add(Rg::mul_mono(s, m, q), Rg::mul(r, q))
187      }
188    }
189  }
190
191  pub fn norm(e: RingExpr): List(RingTerm) {
192    ⊢ List(RingTerm) from match e {
193      ReVar(i) => Cons(Rt(false, Cons(i, Nil)), Nil)
194      ReZero => Nil
195      ReOne => Cons(Rt(false, Nil), Nil)
196      ReAdd(a, b) => Rg::add(Rg::norm(a), Rg::norm(b))
197      ReMul(a, b) => Rg::mul(Rg::norm(a), Rg::norm(b))
198      ReNeg(a) => Rg::neg(Rg::norm(a))
199    }
200  }

205  pub theorem mono_ins_at<A>(
206    add: (A, A) -> A,
207    zero: A,
208    neg: (A) -> A,
209    mul: (A, A) -> A,
210    one: A,
211    r: CommRing(A, add, zero, neg, mul, one),
212    env: List(A),
213    y: Bool,
214    v: Nat,
215    w: Nat,
216    m2: List(Nat),
217    rest: List(Nat),
218    h: Rg::mono(
219      zero,
220      mul,
221      one,
222      env,
223      rest
224    ) = mul(Rg::get(zero, env, v), Rg::mono(zero, mul, one, env, m2))
225  ): Rg::mono(
226    zero,
227    mul,
228    one,
229    env,
230    Nm::ins_at(y, v, w, m2, rest)
231  ) = mul(Rg::get(zero, env, v), mul(Rg::get(zero, env, w), Rg::mono(zero, mul, one, env, m2))) {
232    ⊢ Rg::mono(
233      zero,
234      mul,
235      one,
236      env,
237      Nm::ins_at(y, v, w, m2, rest)
238    ) = mul(Rg::get(zero, env, v), mul(Rg::get(zero, env, w), Rg::mono(zero, mul, one, env, m2))) from match y {
239      false => Eq::trans(
240        Eq::congr(⟦mul(Rg::get(zero, env, w), □)⟧, h),
241        left_commute(
242          mul,
243          one,
244          ring_mul_comm_monoid(add, zero, neg, mul, one, r),
245          Rg::get(zero, env, w),
246          Rg::get(zero, env, v),
247          Rg::mono(zero, mul, one, env, m2)
248        )
249      )
250      true => Refl
251    }
252  }

255    add: (A, A) -> A,
256    zero: A,
257    neg: (A) -> A,
258    mul: (A, A) -> A,
259    one: A,
260    r: CommRing(A, add, zero, neg, mul, one),
261    env: List(A),
262    v: Nat,
263    m: List(Nat)
264  ): Rg::mono(
265    zero,
266    mul,
267    one,
268    env,
269    Nm::ins(v, m)
270  ) = mul(Rg::get(zero, env, v), Rg::mono(zero, mul, one, env, m)) {
271    ⊢ Rg::mono(
272      zero,
273      mul,
274      one,
275      env,
276      Nm::ins(v, m)
277    ) = mul(Rg::get(zero, env, v), Rg::mono(zero, mul, one, env, m)) from match m {

279      Cons(w, m2) => Rg::mono_ins_at(
280        add,
281        zero,

283        mul,
284        one,
285        r,
286        env,
287        v <= w,
288        v,
289        w,
290        m2,
291        Nm::ins(v, m2),
292        Rg::mono_ins(add, zero, neg, mul, one, r, env, v, m2)
293      )
294    }

297  pub theorem mono_mul<A>(
298    add: (A, A) -> A,
299    zero: A,
300    neg: (A) -> A,
301    mul: (A, A) -> A,
302    one: A,
303    r: CommRing(A, add, zero, neg, mul, one),
304    env: List(A),
305    a: List(Nat),
306    b: List(Nat)
307  ): Rg::mono(
308    zero,
309    mul,
310    one,
311    env,
312    Nm::mul(a, b)
313  ) = mul(Rg::mono(zero, mul, one, env, a), Rg::mono(zero, mul, one, env, b)) {
314    ⊢ Rg::mono(
315      zero,
316      mul,
317      one,
318      env,
319      Nm::mul(a, b)
320    ) = mul(Rg::mono(zero, mul, one, env, a), Rg::mono(zero, mul, one, env, b)) from match a {
321      Nil => Eq::sym(
322        left_unit(
323          mul,
324          one,
325          ring_mul_monoid(add, zero, neg, mul, one, r),
326          Rg::mono(zero, mul, one, env, b)
327        )
328      )
329      Cons(v, rest) => Eq::trans(
330        Rg::mono_ins(add, zero, neg, mul, one, r, env, v, Nm::mul(rest, b)),
331        Eq::trans(
332          Eq::congr(
333            ⟦mul(Rg::get(zero, env, v), □)⟧,
334            Rg::mono_mul(add, zero, neg, mul, one, r, env, rest, b)
335          ),
336          Eq::sym(
337            assoc(
338              mul,
339              one,
340              ring_mul_monoid(add, zero, neg, mul, one, r),
341              Rg::get(zero, env, v),
342              Rg::mono(zero, mul, one, env, rest),
343              Rg::mono(zero, mul, one, env, b)
344            )
345          )
346        )
347      )
348    }

368    add: (A, A) -> A,
369    zero: A,
370    neg: (A) -> A,
371    mul: (A, A) -> A,
372    one: A,
373    r: CommRing(A, add, zero, neg, mul, one),
374    s: Bool,
375    d: Bool,
376    x: A,
377    y: A
378  ): Rg::sign(neg, Rg::xor(s, d), mul(x, y)) = mul(Rg::sign(neg, s, x), Rg::sign(neg, d, y)) {
379    ⊢ Rg::sign(
380      neg,
381      Rg::xor(s, d),
382      mul(x, y)
383    ) = mul(Rg::sign(neg, s, x), Rg::sign(neg, d, y)) from match s {
384      false => ⊢ Rg::sign(neg, Rg::xor(false, d), mul(x, y)) = mul(x, Rg::sign(neg, d, y)) from match d {
385        false => Refl
386        true => Eq::sym(ring_mul_neg(add, zero, neg, mul, one, r, x, y))
387      }
388      true => ⊢ Rg::sign(
389        neg,
390        Rg::xor(true, d),
391        mul(x, y)
392      ) = mul(neg(x), Rg::sign(neg, d, y)) from match d {
393        false => Eq::sym(ring_neg_mul(add, zero, neg, mul, one, r, x, y))
394        true => Eq::sym(
395          Eq::trans(
396            ring_neg_mul(add, zero, neg, mul, one, r, x, neg(y)),
397            Eq::trans(
398              Eq::congr(⟦neg(□)⟧, ring_mul_neg(add, zero, neg, mul, one, r, x, y)),
399              ring_neg_neg(add, zero, neg, mul, one, r, mul(x, y))
400            )
401          )
402        )
403      }
404    }

409  // The same monomial twice: cancelling the two when the signs differ keeps the value.
410  pub theorem ins_eq_sound<A>(
411    add: (A, A) -> A,
412    zero: A,
413    neg: (A) -> A,
414    mul: (A, A) -> A,
415    one: A,
416    r: CommRing(A, add, zero, neg, mul, one),
417    env: List(A),
418    s: Bool,
419    d: Bool,
420    m: List(Nat),
421    n: List(Nat),
422    p2: List(RingTerm),
423    h: m = n
424  ): Rg::poly(
425    add,
426    zero,
427    neg,
428    mul,
429    one,
430    env,
431    Rg::ins_eq(s, d, m, n, p2)
432  ) = add(
433    Rg::sign(neg, s, Rg::mono(zero, mul, one, env, m)),
434    add(
435      Rg::sign(neg, d, Rg::mono(zero, mul, one, env, n)),
436      Rg::poly(add, zero, neg, mul, one, env, p2)
437    )
438  ) {
439    let g := ring_group(add, zero, neg, mul, one, r)
440    ⊢ Rg::poly(
441      add,
442      zero,
443      neg,
444      mul,
445      one,
446      env,
447      Rg::ins_eq(s, d, m, n, p2)
448    ) = add(
449      Rg::sign(neg, s, Rg::mono(zero, mul, one, env, m)),
450      add(
451        Rg::sign(neg, d, Rg::mono(zero, mul, one, env, n)),
452        Rg::poly(add, zero, neg, mul, one, env, p2)
453      )
454    ) from match s {
455      false => ⊢ Rg::poly(
456        add,
457        zero,
458        neg,
459        mul,
460        one,
461        env,
462        Rg::ins_eq(false, d, m, n, p2)
463      ) = add(
464        Rg::mono(zero, mul, one, env, m),
465        add(
466          Rg::sign(neg, d, Rg::mono(zero, mul, one, env, n)),
467          Rg::poly(add, zero, neg, mul, one, env, p2)
468        )
469      ) from match d {
470        false => Refl
471        true => Eq::trans(
472          Eq::sym(
473            solve_left(
474              add,
475              zero,
476              neg,
477              g,
478              Rg::mono(zero, mul, one, env, m),
479              Rg::poly(add, zero, neg, mul, one, env, p2)
480            )
481          ),
482          Eq::congr(
483            ⟦add(
484              Rg::mono(zero, mul, one, env, m),
485              add(
486                neg(Rg::mono(zero, mul, one, env, □)),
487                Rg::poly(add, zero, neg, mul, one, env, p2)
488              )
489            )⟧,
490            h
491          )
492        )
493      }
494      true => ⊢ Rg::poly(
495        add,
496        zero,
497        neg,
498        mul,
499        one,
500        env,
501        Rg::ins_eq(true, d, m, n, p2)
502      ) = add(
503        neg(Rg::mono(zero, mul, one, env, m)),
504        add(
505          Rg::sign(neg, d, Rg::mono(zero, mul, one, env, n)),
506          Rg::poly(add, zero, neg, mul, one, env, p2)
507        )
508      ) from match d {
509        false => Eq::trans(
510          Eq::sym(
511            undo_left(
512              add,
513              zero,
514              neg,
515              g,
516              Rg::mono(zero, mul, one, env, m),
517              Rg::poly(add, zero, neg, mul, one, env, p2)
518            )
519          ),
520          Eq::congr(
521            ⟦add(
522              neg(Rg::mono(zero, mul, one, env, m)),
523              add(Rg::mono(zero, mul, one, env, □), Rg::poly(add, zero, neg, mul, one, env, p2))
524            )⟧,
525            h
526          )
527        )
528        true => Refl
529      }
530    }
531  }

533  pub theorem ins_at_sound<A>(
534    add: (A, A) -> A,
535    zero: A,
536    neg: (A) -> A,
537    mul: (A, A) -> A,
538    one: A,
539    r: CommRing(A, add, zero, neg, mul, one),
540    env: List(A),
541    o: Ordering,
542    s: Bool,
543    m: List(Nat),
544    d: Bool,
545    n: List(Nat),
546    p2: List(RingTerm),
547    rest: List(RingTerm),
548    ho: Nm::cmp(m, n) = o,
549    h: Rg::poly(
550      add,
551      zero,
552      neg,
553      mul,
554      one,
555      env,
556      rest
557    ) = add(
558      Rg::sign(neg, s, Rg::mono(zero, mul, one, env, m)),

611        left_commute(
612          add,
613          zero,

615          Rg::sign(neg, d, Rg::mono(zero, mul, one, env, n)),

617          Rg::poly(add, zero, neg, mul, one, env, p2)
618        )
619      )
620    }
621    k(ho)
622  }

624  pub theorem ins_sound<A>(
625    add: (A, A) -> A,
626    zero: A,
627    neg: (A) -> A,
628    mul: (A, A) -> A,
629    one: A,
630    r: CommRing(A, add, zero, neg, mul, one),
631    env: List(A),
632    s: Bool,
633    m: List(Nat),
634    p: List(RingTerm)
635  ): Rg::poly(
636    add,
637    zero,
638    neg,
639    mul,
640    one,
641    env,
642    Rg::ins(s, m, p)
643  ) = add(
644    Rg::sign(neg, s, Rg::mono(zero, mul, one, env, m)),
645    Rg::poly(add, zero, neg, mul, one, env, p)
646  ) {
647    ⊢ Rg::poly(
648      add,
649      zero,
650      neg,
651      mul,
652      one,
653      env,
654      Rg::ins(s, m, p)
655    ) = add(
656      Rg::sign(neg, s, Rg::mono(zero, mul, one, env, m)),
657      Rg::poly(add, zero, neg, mul, one, env, p)
658    ) from match p {
659      Nil => Refl
660      Cons(u, p2) => ⊢ Rg::poly(
661        add,
662        zero,
663        neg,
664        mul,
665        one,
666        env,
667        Rg::ins(s, m, Cons(u, p2))
668      ) = add(
669        Rg::sign(neg, s, Rg::mono(zero, mul, one, env, m)),
670        Rg::poly(add, zero, neg, mul, one, env, Cons(u, p2))
671      ) from match u {
672        Rt(d, n) => Rg::ins_at_sound(
673          add,
674          zero,
675          neg,
676          mul,
677          one,
678          r,
679          env,
680          Nm::cmp(m, n),
681          s,
682          m,
683          d,
684          n,
685          p2,
686          Rg::ins(s, m, p2),
687          Refl,
688          Rg::ins_sound(add, zero, neg, mul, one, r, env, s, m, p2)
689        )
690      }
691    }

698    zero: A,
699    neg: (A) -> A,
700    mul: (A, A) -> A,
701    one: A,
702    r: CommRing(A, add, zero, neg, mul, one),
703    env: List(A),
704    p: List(RingTerm),
705    q: List(RingTerm)
706  ): Rg::poly(
707    add,
708    zero,
709    neg,
710    mul,
711    one,
712    env,
713    Rg::add(p, q)
714  ) = add(Rg::poly(add, zero, neg, mul, one, env, p), Rg::poly(add, zero, neg, mul, one, env, q)) {
715    ⊢ Rg::poly(
716      add,
717      zero,
718      neg,
719      mul,
720      one,
721      env,
722      Rg::add(p, q)
723    ) = add(Rg::poly(add, zero, neg, mul, one, env, p), Rg::poly(add, zero, neg, mul, one, env, q)) from match p {
724      Nil => Eq::sym(
725        left_unit(
726          add,
727          zero,
728          ring_add_monoid(add, zero, neg, mul, one, r),
729          Rg::poly(add, zero, neg, mul, one, env, q)
730        )
731      )
732      Cons(t, rest) => ⊢ Rg::poly(
733        add,
734        zero,
735        neg,
736        mul,
737        one,
738        env,
739        Rg::add(Cons(t, rest), q)
740      ) = add(
741        Rg::poly(add, zero, neg, mul, one, env, Cons(t, rest)),
742        Rg::poly(add, zero, neg, mul, one, env, q)
743      ) from match t {
744        Rt(s, m) => Eq::trans(
745          Rg::ins_sound(add, zero, neg, mul, one, r, env, s, m, Rg::add(rest, q)),
746          Eq::trans(
747            Eq::congr(
748              ⟦add(Rg::sign(neg, s, Rg::mono(zero, mul, one, env, m)), □)⟧,

751            Eq::sym(
752              assoc(
753                add,
754                zero,
755                ring_add_monoid(add, zero, neg, mul, one, r),
756                Rg::sign(neg, s, Rg::mono(zero, mul, one, env, m)),
757                Rg::poly(add, zero, neg, mul, one, env, rest),
758                Rg::poly(add, zero, neg, mul, one, env, q)
759              )
760            )
761          )
762        )
763      }

832  pub theorem mul_mono_sound<A>(
833    add: (A, A) -> A,
834    zero: A,
835    neg: (A) -> A,
836    mul: (A, A) -> A,
837    one: A,
838    r: CommRing(A, add, zero, neg, mul, one),
839    env: List(A),
840    s: Bool,
841    m: List(Nat),
842    q: List(RingTerm)
843  ): Rg::poly(
844    add,
845    zero,
846    neg,
847    mul,
848    one,
849    env,
850    Rg::mul_mono(s, m, q)
851  ) = mul(
852    Rg::sign(neg, s, Rg::mono(zero, mul, one, env, m)),
853    Rg::poly(add, zero, neg, mul, one, env, q)
854  ) {
855    ⊢ Rg::poly(
856      add,
857      zero,
858      neg,
859      mul,
860      one,
861      env,
862      Rg::mul_mono(s, m, q)
863    ) = mul(
864      Rg::sign(neg, s, Rg::mono(zero, mul, one, env, m)),
865      Rg::poly(add, zero, neg, mul, one, env, q)
866    ) from match q {
867      Nil => Eq::sym(
868        ring_mul_zero(
869          add,
870          zero,
871          neg,
872          mul,
873          one,
874          r,
875          Rg::sign(neg, s, Rg::mono(zero, mul, one, env, m))
876        )
877      )
878      Cons(u, rest) => ⊢ Rg::poly(
879        add,
880        zero,
881        neg,
882        mul,
883        one,
884        env,
885        Rg::mul_mono(s, m, Cons(u, rest))
886      ) = mul(
887        Rg::sign(neg, s, Rg::mono(zero, mul, one, env, m)),
888        Rg::poly(add, zero, neg, mul, one, env, Cons(u, rest))
889      ) from match u {
890        Rt(d, n) => Eq::trans(
891          Rg::ins_sound(
892            add,
893            zero,
894            neg,
895            mul,
896            one,
897            r,
898            env,
899            Rg::xor(s, d),
900            Nm::mul(m, n),
901            Rg::mul_mono(s, m, rest)
902          ),
903          Eq::trans(
904            Eq::congr(
905              ⟦add(
906                Rg::sign(neg, Rg::xor(s, d), □),
907                Rg::poly(add, zero, neg, mul, one, env, Rg::mul_mono(s, m, rest))
908              )⟧,
909              Rg::mono_mul(add, zero, neg, mul, one, r, env, m, n)
910            ),
911            Eq::trans(
912              Eq::congr(
913                ⟦add(□, Rg::poly(add, zero, neg, mul, one, env, Rg::mul_mono(s, m, rest)))⟧,
914                Rg::sign_mul(
915                  add,
916                  zero,
917                  neg,
918                  mul,
919                  one,
920                  r,
921                  s,
922                  d,
923                  Rg::mono(zero, mul, one, env, m),
924                  Rg::mono(zero, mul, one, env, n)
925                )
926              ),
927              Eq::trans(
928                Eq::congr(
929                  ⟦add(
930                    mul(
931                      Rg::sign(neg, s, Rg::mono(zero, mul, one, env, m)),
932                      Rg::sign(neg, d, Rg::mono(zero, mul, one, env, n))
933                    ),
934                    □
935                  )⟧,
936                  Rg::mul_mono_sound(add, zero, neg, mul, one, r, env, s, m, rest)

938                Eq::sym(
939                  ring_distrib_left(
940                    add,
941                    zero,
942                    neg,
943                    mul,
944                    one,
945                    r,
946                    Rg::sign(neg, s, Rg::mono(zero, mul, one, env, m)),

950                )
951              )
952            )
953          )

1040  pub theorem norm_sound<A>(
1041    add: (A, A) -> A,
1042    zero: A,
1043    neg: (A) -> A,
1044    mul: (A, A) -> A,
1045    one: A,
1046    r: CommRing(A, add, zero, neg, mul, one),
1047    env: List(A),
1048    e: RingExpr
1049  ): Rg::eval(
1050    add,
1051    zero,
1052    neg,
1053    mul,
1054    one,
1055    env,
1056    e
1057  ) = Rg::poly(add, zero, neg, mul, one, env, Rg::norm(e)) {
1058    ⊢ Rg::eval(
1059      add,
1060      zero,
1061      neg,
1062      mul,
1063      one,
1064      env,
1065      e
1066    ) = Rg::poly(add, zero, neg, mul, one, env, Rg::norm(e)) from match e {
1067      ReVar(i) => Eq::sym(
1068        Eq::trans(
1069          right_unit(
1070            add,
1071            zero,
1072            ring_add_monoid(add, zero, neg, mul, one, r),
1073            mul(Rg::get(zero, env, i), one)
1074          ),
1075          right_unit(mul, one, ring_mul_monoid(add, zero, neg, mul, one, r), Rg::get(zero, env, i))
1076        )
1077      )
1078      ReZero => Refl
1079      ReOne => Eq::sym(right_unit(add, zero, ring_add_monoid(add, zero, neg, mul, one, r), one))
1080      ReAdd(a, b) => Eq::trans(
1081        Eq::congr(
1082          ⟦add(□, Rg::eval(add, zero, neg, mul, one, env, b))⟧,
1083          Rg::norm_sound(add, zero, neg, mul, one, r, env, a)
1084        ),
1085        Eq::trans(
1086          Eq::congr(
1087            ⟦add(Rg::poly(add, zero, neg, mul, one, env, Rg::norm(a)), □)⟧,
1088            Rg::norm_sound(add, zero, neg, mul, one, r, env, b)
1089          ),
1090          Eq::sym(Rg::add_sound(add, zero, neg, mul, one, r, env, Rg::norm(a), Rg::norm(b)))
1091        )
1092      )
1093      ReMul(a, b) => Eq::trans(
1094        Eq::congr(
1095          ⟦mul(□, Rg::eval(add, zero, neg, mul, one, env, b))⟧,
1096          Rg::norm_sound(add, zero, neg, mul, one, r, env, a)
1097        ),
1098        Eq::trans(
1099          Eq::congr(
1100            ⟦mul(Rg::poly(add, zero, neg, mul, one, env, Rg::norm(a)), □)⟧,
1101            Rg::norm_sound(add, zero, neg, mul, one, r, env, b)
1102          ),
1103          Eq::sym(Rg::mul_sound(add, zero, neg, mul, one, r, env, Rg::norm(a), Rg::norm(b)))
1104        )
1105      )
1106      ReNeg(a) => Eq::trans(
1107        Eq::congr(⟦neg(□)⟧, Rg::norm_sound(add, zero, neg, mul, one, r, env, a)),
1108        Eq::sym(Rg::neg_sound(add, zero, neg, mul, one, r, env, Rg::norm(a)))
1109      )
1110    }
1111  }

1118  neg: (A) -> A,
