40pub fn semiring_atom(
41  t: semiring_notation::Atom,
42  add: SynExpr,
43  zero: SynExpr,
44  mul: SynExpr,
45  one: SynExpr
46): SynExpr ! {macro_error, Div} {
47  match t {
48    ZeroLit(tok) => zero
49    OneLit(tok) => one
50    Embed(e) => e
51    Paren(p) => semiring_sum(p.sum, add, zero, mul, one)
52    Reference(r) => match r.arguments {
53      None => semiring_name(r.name)
54      Some(args) => ExApp(
55        semiring_name(r.name),
56        Cons(
57          semiring_sum(args.sum, add, zero, mul, one),
58          semiring_args(args.rep, add, zero, mul, one)
59        ),
60        r.span
61      )
62    }
63  }
64}

122pub macro fn semiring(e: semiring_notation::Expr): SynExpr ! {macro_error, Div} {
123  let add := semiring_header(e.header)
124  let zero := semiring_header(e.header_2)
125  let mul := semiring_header(e.header_3)
126  let one := semiring_header(e.header_4)
127  let left := semiring_sum(e.formula.sum, add, zero, mul, one)
128  match e.formula.opt { None => left
129    Some(r) => { let right := semiring_sum(r.sum, add, zero, mul, one)
130      quote { $left = $right } }
131  }
