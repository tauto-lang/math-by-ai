35pub fn algebra_header(a: monoid_notation::Header): SynExpr {
36  match a {
37    Name(t) => algebra_name(t)
38    Num(t) => ExNat(t.text, t.span)
39    Embed(e) => e
40  }
41}
42pub fn algebra_atom(
43  a: monoid_notation::Atom,
44  op: SynExpr,
45  unit: SynExpr
46): SynExpr ! {macro_error, Div} {
47  match a {
48    Num(t) => ExNat(t.text, t.span)
49    Identity(t) => unit
50    Embed(e) => e
51    Paren(p) => algebra_product(p.product, op, unit)
52    Reference(r) => match r.arguments {
53      None => if r.name.text == "ε" { unit } else { algebra_name(r.name) }
54      Some(args) => ExApp(
55        algebra_name(r.name),
56        Cons(algebra_product(args.product, op, unit), algebra_args(args.rep, op, unit)),
57        r.span
58      )
59    }
60  }
61}
62pub fn algebra_args(
63  xs: List(monoid_notation::Arguments_Rep1),
64  op: SynExpr,
65  unit: SynExpr
66): List(SynExpr) ! {macro_error, Div} {
67  match xs { Nil => Nil
68    Cons(x, rest) => Cons(algebra_product(x.product, op, unit), algebra_args(rest, op, unit)) }
69}
70pub fn algebra_rest(
71  acc: SynExpr,
72  xs: List(monoid_notation::Product_Rep1),
73  op: SynExpr,
74  unit: SynExpr
75): SynExpr ! {macro_error, Div} {
76  match xs { Nil => acc
77    Cons(x, rest) => { let b := algebra_atom(x.atom, op, unit)
78      algebra_rest(quote { $op($acc, $b) }, rest, op, unit) }
79  }
80}
81pub fn algebra_product(
82  p: monoid_notation::Product,
83  op: SynExpr,
84  unit: SynExpr
85): SynExpr ! {macro_error, Div} {
86  algebra_rest(algebra_atom(p.atom, op, unit), p.rep, op, unit)
87}
88pub fn algebra_expression(e: monoid_notation::Expr): SynExpr ! {macro_error, Div} {
89  let op := algebra_header(e.header)
90  let unit := algebra_header(e.header_2)
91  let left := algebra_product(e.formula.product, op, unit)
92  match e.formula.opt {
93    None => left
94    Some(r) => { let right := algebra_product(r.product, op, unit)
95      quote { $left = $right } }
96  }
97}
98pub macro fn alg(e: monoid_notation::Expr): SynExpr ! {macro_error, Div} { algebra_expression(e) }
