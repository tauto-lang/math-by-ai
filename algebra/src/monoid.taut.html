8// The operation is a parameter: one carrier can have several monoids.
9pub(open) inductive Monoid(A: Type, op: (A, A) -> A, unit: A): Prop {
10  Mk(
11    assoc: (a: A, b: A, c: A) -> alg!(`op, unit; a ⋆ b ⋆ c = a ⋆ (b ⋆ c)`),
12    left_unit: (a: A) -> alg!(`op, unit; ε ⋆ a = a`),
13    right_unit: (a: A) -> alg!(`op, unit; a ⋆ ε = a`)
14  ): Monoid(A, op, unit)
15}

17pub theorem assoc<A>(
18  op: (A, A) -> A,
19  unit: A,
20  m: Monoid(A, op, unit),
21  a: A,
22  b: A,
23  c: A
24): alg!(`op, unit; a ⋆ b ⋆ c = a ⋆ (b ⋆ c)`) {
25  ⊢ alg!(`op, unit; a ⋆ b ⋆ c = a ⋆ (b ⋆ c)`) from match m {
26    Mk(h, l, r) => h(a, b, c)
27  }
28}
29pub theorem left_unit<A>(
30  op: (A, A) -> A,
31  unit: A,
32  m: Monoid(A, op, unit),
33  a: A
34): alg!(`op, unit; ε ⋆ a = a`) {
35  ⊢ alg!(`op, unit; ε ⋆ a = a`) from match m { Mk(h, l, r) => l(a) }
36}
37pub theorem right_unit<A>(
38  op: (A, A) -> A,
39  unit: A,
40  m: Monoid(A, op, unit),
41  a: A
42): alg!(`op, unit; a ⋆ ε = a`) {
43  ⊢ alg!(`op, unit; a ⋆ ε = a`) from match m { Mk(h, l, r) => r(a) }
44}

46pub fn product<A>(op: (A, A) -> A, unit: A, xs: List(A)): A {
47  ⊢ A from match xs { Nil => unit
48    Cons(x, rest) => alg!(`op, unit; x ⋆ product(op, ε, rest)`) }
49}

51pub theorem product_append<A>(
52  op: (A, A) -> A,
53  unit: A,
54  m: Monoid(A, op, unit),
55  xs: List(A),
56  ys: List(A)
57): alg!(`op, unit; product(op, ε, List::append(xs, ys)) = product(op, ε, xs) ⋆ product(op, ε, ys)`) {
58  proof!(```
59      By induction on xs.
60        [base] Case Nil. By Eq::sym.
61        Case Cons(x, rest). We have $alg!(`op, unit; x ⋆ product(op, ε, List::append(rest, ys))`)
62        = $alg!(`op, unit; x ⋆ (product(op, ε, rest) ⋆ product(op, ε, ys))`) by the induction hypothesis
63        = $alg!(`op, unit; x ⋆ product(op, ε, rest) ⋆ product(op, ε, ys)`) by assoc(op, unit, m, x, product(op, unit, rest), product(op, unit, ys)).
64      %%
65      [base] left_unit(op, unit, m, product(op, unit, ys))
66    ```)
67}

69pub theorem unit_unique<A>(
70  op: (A, A) -> A,
71  unit: A,
72  m: Monoid(A, op, unit),
73  other: A,
74  other_left: (a: A) -> alg!(`op, unit; other ⋆ a = a`)
75): other = unit {
76  calc!(```
77    $other
78    = $alg!(`op, unit; other ⋆ ε`) from $Eq::sym(right_unit(op, unit, m, other))
79    = $unit from $other_left(unit)
80    ```)
81}

83pub fn power<A>(op: (A, A) -> A, unit: A, a: A, n: Nat): A {
84  ⊢ A from match n {
85    Zero => unit
86    Succ(k) => op(power(op, unit, a, k), a)
87  }
88}

90pub theorem power_add<A>(
91  op: (A, A) -> A,
92  unit: A,
93  m: Monoid(A, op, unit),
94  a: A,
95  n: Nat,
96  k: Nat
97): alg!(`op, unit; power(op, ε, a, $(n + k)) = power(op, ε, a, n) ⋆ power(op, ε, a, k)`) {
98  proof!(```
99      By induction on k.
100        [base] Case Zero. By Eq::sym.
101        Case Succ(j). We have $alg!(`op, unit; power(op, ε, a, $(n + j)) ⋆ a`)
102        = $alg!(`op, unit; (power(op, ε, a, n) ⋆ power(op, ε, a, j)) ⋆ a`) by the induction hypothesis
103        = $alg!(`op, unit; power(op, ε, a, n) ⋆ (power(op, ε, a, j) ⋆ a)`) by assoc(op, unit, m, power(op, unit, a, n), power(op, unit, a, j), a).
104      %%
105      [base] right_unit(op, unit, m, power(op, unit, a, n))
106    ```)
107}

109// A map preserving the operation and unit preserves every finite product.
110pub fn map_list<A, B>(f: (A) -> B, xs: List(A)): List(B) {
111  ⊢ List(B) from match xs {
112    Nil => Nil
113    Cons(x, rest) => Cons(f(x), map_list(f, rest))
114  }
115}

117pub theorem map_product<A, B>(
118  op: (A, A) -> A,
119  unit: A,
120  target: (B, B) -> B,
121  target_unit: B,
122  f: (A) -> B,
123  preserves_unit: f(unit) = target_unit,
124  preserves_op: (
125    a: A,
126    b: A
127  ) -> f(alg!(`op, unit; a ⋆ b`)) = alg!(`target, target_unit; f(a) ⋆ f(b)`),
128  xs: List(A)
129): f(product(op, unit, xs)) = product(target, target_unit, map_list(f, xs)) {
130  proof!(```
131      By induction on xs.
132        Case Nil. By preserves_unit.
133        Case Cons(x, rest). We have $f(alg!(`op, unit; x ⋆ product(op, ε, rest)`))
134        = $alg!(`target, target_unit; f(x) ⋆ f(product(op, unit, rest))`) by preserves_op(x, product(op, unit, rest))
135        = $alg!(`target, target_unit; f(x) ⋆ product(target, ε, map_list(f, rest))`) by the induction hypothesis.
136    ```)
137}

139pub theorem product_cons_congr<A>(
140  op: (A, A) -> A,
141  unit: A,
142  a: A,
143  b: A,
144  xs: List(A),
145  ys: List(A),
146  heads: a = b,
147  tails: alg!(`op, unit; product(op, ε, xs) = product(op, ε, ys)`)
148): alg!(`op, unit; product(op, ε, Cons(a, xs)) = product(op, ε, Cons(b, ys))`) {
149  congr2(op, a, b, product(op, unit, xs), product(op, unit, ys), heads, tails)
