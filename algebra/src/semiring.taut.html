7// Multiplication need not commute (e.g. matrices).
8pub(open) inductive Semiring(A: Type, add: (A, A) -> A, zero: A, mul: (A, A) -> A, one: A): Prop {
9  Mk(
10    additive: CommMonoid(A, add, zero),
11    multiplicative: Monoid(A, mul, one),
12    distrib_left: (
13      a: A,
14      b: A,
15      c: A
16    ) -> semiring!(`add, zero, mul, one; a * (b + c) = a * b + a * c`),
17    distrib_right: (
18      a: A,
19      b: A,
20      c: A
21    ) -> semiring!(`add, zero, mul, one; (a + b) * c = a * c + b * c`),
22    zero_left: (a: A) -> semiring!(`add, zero, mul, one; 0 * a = 0`),
23    zero_right: (a: A) -> semiring!(`add, zero, mul, one; a * 0 = 0`)
24  ): Semiring(A, add, zero, mul, one)
25}
26pub theorem semiring_additive<A>(
27  add: (A, A) -> A,
28  zero: A,
29  mul: (A, A) -> A,
30  one: A,
31  s: Semiring(A, add, zero, mul, one)
32): CommMonoid(A, add, zero) {
33  ⊢ CommMonoid(A, add, zero) from match s { Mk(a, m, l, r, zl, zr) => a }
34}
35pub theorem semiring_multiplicative<A>(
36  add: (A, A) -> A,
37  zero: A,
38  mul: (A, A) -> A,
39  one: A,
40  s: Semiring(A, add, zero, mul, one)
41): Monoid(A, mul, one) {
42  ⊢ Monoid(A, mul, one) from match s { Mk(a, m, l, r, zl, zr) => m }
43}
44pub theorem distribute_left<A>(
45  add: (A, A) -> A,
46  zero: A,
47  mul: (A, A) -> A,
48  one: A,
49  s: Semiring(A, add, zero, mul, one),
50  a: A,
51  b: A,
52  c: A
53): semiring!(`add, zero, mul, one; a * (b + c) = a * b + a * c`) {
54  ⊢ semiring!(`add, zero, mul, one; a * (b + c) = a * b + a * c`) from match s {
55    Mk(am, mm, l, r, zl, zr) => l(a, b, c)
56  }
57}
58pub theorem distribute_right<A>(
59  add: (A, A) -> A,
60  zero: A,
61  mul: (A, A) -> A,
62  one: A,
63  s: Semiring(A, add, zero, mul, one),
64  a: A,
65  b: A,
66  c: A
67): semiring!(`add, zero, mul, one; (a + b) * c = a * c + b * c`) {
68  ⊢ semiring!(`add, zero, mul, one; (a + b) * c = a * c + b * c`) from match s {
69    Mk(am, mm, l, r, zl, zr) => r(a, b, c)
70  }
71}
72pub theorem annihilate_left<A>(
73  add: (A, A) -> A,
74  zero: A,
75  mul: (A, A) -> A,
76  one: A,
77  s: Semiring(A, add, zero, mul, one),
78  a: A
79): semiring!(`add, zero, mul, one; 0 * a = 0`) {
80  ⊢ semiring!(`add, zero, mul, one; 0 * a = 0`) from match s { Mk(am, mm, l, r, zl, zr) => zl(a) }
81}
82pub theorem annihilate_right<A>(
83  add: (A, A) -> A,
84  zero: A,
85  mul: (A, A) -> A,
86  one: A,
87  s: Semiring(A, add, zero, mul, one),
88  a: A
89): semiring!(`add, zero, mul, one; a * 0 = 0`) {
90  ⊢ semiring!(`add, zero, mul, one; a * 0 = 0`) from match s { Mk(am, mm, l, r, zl, zr) => zr(a) }
91}
92pub theorem mul_sum_left<A>(
93  add: (A, A) -> A,
94  zero: A,
95  mul: (A, A) -> A,
96  one: A,
97  s: Semiring(A, add, zero, mul, one),
98  a: A,
99  xs: List(A)
100): semiring!(`add, zero, mul, one; a * product(add, 0, xs)`) = product(
101  add,
102  zero,
103  map_list((x: A) => semiring!(`add, zero, mul, one; a * x`), xs)
104) {
105  map_product(
106    add,
107    zero,
108    add,
109    zero,
110    (x: A) => semiring!(`add, zero, mul, one; a * x`),
111    annihilate_right(add, zero, mul, one, s, a),
112    (x: A, y: A) => distribute_left(add, zero, mul, one, s, a, x, y),
113    xs
114  )
115}
116pub theorem mul_sum_right<A>(
117  add: (A, A) -> A,
118  zero: A,
119  mul: (A, A) -> A,
120  one: A,
121  s: Semiring(A, add, zero, mul, one),
122  a: A,
123  xs: List(A)
124): semiring!(`add, zero, mul, one; product(add, 0, xs) * a`) = product(
125  add,
126  zero,
127  map_list((x: A) => semiring!(`add, zero, mul, one; x * a`), xs)
128) {
129  map_product(
130    add,
131    zero,
132    add,
133    zero,
134    (x: A) => semiring!(`add, zero, mul, one; x * a`),
135    annihilate_left(add, zero, mul, one, s, a),
136    (x: A, y: A) => distribute_right(add, zero, mul, one, s, x, y, a),
137    xs
138  )
139}
140// Four terms in row order: valid even when multiplication is noncommutative.
141pub theorem expand_product<A>(
142  add: (A, A) -> A,
143  zero: A,
144  mul: (A, A) -> A,
145  one: A,
146  s: Semiring(A, add, zero, mul, one),
147  a: A,
148  b: A,
149  c: A,
150  d: A
151): semiring!(`add, zero, mul, one; (a + b) * (c + d) = a * c + a * d + (b * c + b * d)`) {
152  calc!(```
153    $semiring!(`add, zero, mul, one; (a + b) * (c + d)`)
154    = $semiring!(`add, zero, mul, one; a * (c + d) + b * (c + d)`) from $distribute_right(
155      add,
156      zero,
157      mul,
158      one,
159      s,
160      a,
161      b,
162      semiring!(`add, zero, mul, one; c + d`)
163    )
164    = $semiring!(`add, zero, mul, one; a * c + a * d + b * (c + d)`) from $Eq::congr(
165      (v: A) => semiring!(`add, zero, mul, one; v + b * (c + d)`),
166      distribute_left(add, zero, mul, one, s, a, c, d)
167    )
168    = $semiring!(`add, zero, mul, one; a * c + a * d + (b * c + b * d)`) from $Eq::congr(
169      (v: A) => semiring!(`add, zero, mul, one; a * c + a * d + v`),
170      distribute_left(add, zero, mul, one, s, b, c, d)
171    )
172    ```)
