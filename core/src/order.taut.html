4pub(open) inductive PartialOrder(A: Type, le: (A, A) -> Prop): Prop {
5  Mk(
6    reflexive: (a: A) -> logic!(`le(a, a)`),
7    transitive: (
8      a: A,
9      b: A,
10      c: A,
11      ab: logic!(`le(a, b)`),
12      bc: logic!(`le(b, c)`)
13    ) -> logic!(`le(a, c)`),
14    antisymmetric: (
15      a: A,
16      b: A,
17      ab: logic!(`le(a, b)`),
18      ba: logic!(`le(b, a)`)
19    ) -> logic!(`a = b`)
20  ): PartialOrder(A, le)
21}
22pub theorem order_refl<A>(le: (A, A) -> Prop, p: PartialOrder(A, le), a: A): logic!(`le(a, a)`) {
23  ⊢ logic!(`le(a, a)`) from match p { Mk(r, t, s) => r(a) }
24}
25pub theorem order_trans<A>(
26  le: (A, A) -> Prop,
27  p: PartialOrder(A, le),
28  a: A,
29  b: A,
30  c: A,
31  ab: logic!(`le(a, b)`),
32  bc: logic!(`le(b, c)`)
33): logic!(`le(a, c)`) {
34  ⊢ logic!(`le(a, c)`) from match p { Mk(r, t, s) => t(a, b, c, ab, bc) }
35}
36pub theorem order_antisym<A>(
37  le: (A, A) -> Prop,
38  p: PartialOrder(A, le),
39  a: A,
40  b: A,
41  ab: logic!(`le(a, b)`),
42  ba: logic!(`le(b, a)`)
43): logic!(`a = b`) {
44  ⊢ logic!(`a = b`) from match p { Mk(r, t, s) => s(a, b, ab, ba) }
45}
46// Uniqueness needs only antisymmetry; neither set nor membership is decidable.
47pub fn IsLeast<A>(le: (A, A) -> Prop, member: (A) -> Prop, a: A): Prop {
48  (b: A, in_set: member(b)) -> logic!(`le(a, b)`)
49}
50pub theorem least_unique<A>(
51  le: (A, A) -> Prop,
52  p: PartialOrder(A, le),
53  member: (A) -> Prop,
54  a: A,
55  b: A,
56  ha: member(a),
57  hb: member(b),
58  la: IsLeast(le, member, a),
59  lb: IsLeast(le, member, b)
60): logic!(`a = b`) {
61  order_antisym(le, p, a, b, la(b, hb), lb(a, ha))
62}
