5// Congruence for a binary operation, without choosing either argument first.
6pub theorem congr2<A, B, C>(
7  f: (A, B) -> C,
8  a: A,
9  b: A,
10  c: B,
11  d: B,
12  ab: logic!(`a = b`),
13  cd: logic!(`c = d`)
14): logic!(`f(a, c) = f(b, d)`) {
15  calc!(```
16    $logic!(`f(a, c)`)
17    = $logic!(`f(b, c)`) from $Eq::congr((x: A) => logic!(`f(x, c)`), ab)
18    = $logic!(`f(b, d)`) from $Eq::congr((y: B) => logic!(`f(b, y)`), cd)
19    ```)
20}
