1import std::Base
2import std::Rat
3import std::RatLaws

6// A real number is an inhabited, proper, downward closed rational cut
7// with no greatest element. Arithmetic on reals is not needed below.
8pub(open) inductive Real: Type {
9  Dedekind(
10    lower: (Rat) -> Prop,
11    inhabited: ∃ q: Rat. lower(q),
12    proper: ∃ q: Rat. ¬lower(q),
13    downward: (p: Rat, q: Rat, lt: RationalLt(p, q), h: lower(q)) -> lower(p),
14    rounded: (p: Rat, h: lower(p)) -> (∃ q: Rat. RationalLt(p, q) ∧ lower(q))
15  ): Real
16}
17
18pub fn below(x: Real, q: Rat): Prop {
19  ⊢ Prop from match x { Dedekind(lower, inhabited, proper, downward, rounded) => lower(q) }
20}
21
22pub theorem real_inhabited(x: Real): ∃ q: Rat. below(x, q) {
23  ⊢ ∃ q: Rat. below(x, q) from match x {
24    Dedekind(lower, inhabited, proper, downward, rounded) => inhabited
25  }
26}
27
28pub theorem real_proper(x: Real): ∃ q: Rat. ¬below(x, q) {
29  ⊢ ∃ q: Rat. ¬below(x, q) from match x {
30    Dedekind(lower, inhabited, proper, downward, rounded) => proper
31  }
32}
33
34pub theorem real_downward(
35  x: Real,
36  p: Rat,
37  q: Rat,
38  lt: RationalLt(p, q),
39  h: below(x, q)
40): below(x, p) {
41  ⊢ below(x, p) from match x {
42    Dedekind(lower, inhabited, proper, downward, rounded) => downward(p, q, lt, h)
43  }
44}
45
46pub theorem real_rounded(
47  x: Real,
48  p: Rat,
49  h: below(x, p)
50): ∃ q: Rat. RationalLt(p, q) ∧ below(x, q) {
51  ⊢ ∃ q: Rat. RationalLt(p, q) ∧ below(x, q) from match x {
52    Dedekind(lower, inhabited, proper, downward, rounded) => rounded(p, h)
53  }
54}
55
56// A cut depends on the rational number, not on the chosen fraction.
57pub theorem real_respects_rationals(
58  x: Real,
59  p: Rat,
60  q: Rat,
61  equal: Rat::equiv(p, q),
62  h: below(x, p)
63): below(x, q) {
64  ⊢ below(x, q) from match real_rounded(x, p, h) {
65    ExIntro(r, pair) => ⊢ below(x, q) from match pair {
66      AndIntro(pr, member) => real_downward(
67        x,
68        q,
69        r,
70        (rq: RatLe(r, q)) => pr(Rat::le_congr(r, q, p, Rat::equiv_symm(p, q, equal), rq)),
71        member
72      )
73    }
74  }
75}

77// Equality of real VALUES is equality of their cuts, not their proof fields.
78pub fn RealLe(x: Real, y: Real): Prop { (q: Rat, h: below(x, q)) -> below(y, q) }
79pub fn SameReal(x: Real, y: Real): Prop { RealLe(x, y) ∧ RealLe(y, x) }
80
81pub theorem same_real_refl(x: Real): SameReal(x, x) {
82  AndIntro((q: Rat, h: below(x, q)) => h, (q: Rat, h: below(x, q)) => h)
83}
84
85pub theorem same_real_symm(x: Real, y: Real, h: SameReal(x, y)): SameReal(y, x) {
86  ⊢ SameReal(y, x) from match h { AndIntro(xy, yx) => AndIntro(yx, xy) }
87}
88
89pub theorem same_real_trans(
90  x: Real,
91  y: Real,
92  z: Real,
93  h: SameReal(x, y),
94  k: SameReal(y, z)
95): SameReal(x, z) {
96  ⊢ SameReal(x, z) from match h {
97    AndIntro(xy, yx) => ⊢ SameReal(x, z) from match k {
98      AndIntro(yz, zy) => AndIntro(
99        (q: Rat, member: below(x, q)) => yz(q, xy(q, member)),
100        (q: Rat, member: below(z, q)) => yx(q, zy(q, member))
101      )
102    }
103  }
104}

106// The union of a nonempty family of cuts bounded above is its supremum.
107pub fn supremum_lower<I>(family: (I) -> Real, q: Rat): Prop {
108  ∃ i: I. below(family(i), q)
109}
110
111pub theorem supremum_inhabited<I>(family: (I) -> Real, i: I): ∃ q: Rat. supremum_lower(family, q) {
112  ⊢ ∃ q: Rat. supremum_lower(family, q) from match real_inhabited(family(i)) {
113    ExIntro(q, h) => ExIntro(q, ExIntro(i, h))
114  }
115}
116
117pub theorem supremum_proper<I>(
118  family: (I) -> Real,
119  upper: Real,
120  bound: (i: I) -> RealLe(family(i), upper)
121): ∃ q: Rat. ¬supremum_lower(family, q) {
122  ⊢ ∃ q: Rat. ¬supremum_lower(family, q) from match real_proper(upper) {
123    ExIntro(q, outside) => ExIntro(q, (member: supremum_lower(family, q)) => ⊢ False from match member {
124      ExIntro(i, h) => outside(bound(i, q, h))
125    })
126  }
127}
128
129pub theorem supremum_downward<I>(
130  family: (I) -> Real,
131  p: Rat,
132  q: Rat,
133  lt: RationalLt(p, q),
134  h: supremum_lower(family, q)
135): supremum_lower(family, p) {
136  ⊢ supremum_lower(family, p) from match h {
137    ExIntro(i, member) => ExIntro(i, real_downward(family(i), p, q, lt, member))
138  }
139}
140
141pub theorem supremum_rounded<I>(
142  family: (I) -> Real,
143  p: Rat,
144  h: supremum_lower(family, p)
145): ∃ q: Rat. RationalLt(p, q) ∧ supremum_lower(family, q) {
146  ⊢ ∃ q: Rat. RationalLt(p, q) ∧ supremum_lower(family, q) from match h {
147    ExIntro(i, member) => ⊢ ∃ q: Rat. RationalLt(p, q) ∧ supremum_lower(family, q) from match real_rounded(
148      family(i),
149      p,
150      member
151    ) {
152      ExIntro(q, pair) => ⊢ ∃ q: Rat. RationalLt(p, q) ∧ supremum_lower(family, q) from match pair {
153        AndIntro(lt, in_cut) => ExIntro(q, AndIntro(lt, ExIntro(i, in_cut)))
154      }
155    }
156  }
157}
158
159pub fn supremum<I>(
160  family: (I) -> Real,
161  i: I,
162  upper: Real,
163  bound: (j: I) -> RealLe(family(j), upper)
164): Real {
165  Dedekind(
166    (q: Rat) => supremum_lower(family, q),
167    supremum_inhabited(family, i),
168    supremum_proper(family, upper, bound),
169    (p: Rat, q: Rat, lt: RationalLt(p, q), h: supremum_lower(family, q)) => supremum_downward(
170      family,
171      p,
172      q,
173      lt,
174      h
175    ),
176    (p: Rat, h: supremum_lower(family, p)) => supremum_rounded(family, p, h)
177  )
178}

180pub theorem supremum_upper<I>(
181  family: (I) -> Real,
182  i: I,
183  upper: Real,
184  bound: (j: I) -> RealLe(family(j), upper),
185  j: I
186): RealLe(family(j), supremum(family, i, upper, bound)) {
187  (q: Rat, h: below(family(j), q)) => ExIntro(j, h)
188}
189
190pub theorem supremum_least<I>(
191  family: (I) -> Real,
192  i: I,
193  upper: Real,
194  bound: (j: I) -> RealLe(family(j), upper),
195  candidate: Real,
196  h: (j: I) -> RealLe(family(j), candidate)
197): RealLe(supremum(family, i, upper, bound), candidate) {
198  (q: Rat, member: supremum_lower(family, q)) => ⊢ below(candidate, q) from match member {
199    ExIntro(j, in_cut) => h(j, q, in_cut)
200  }
