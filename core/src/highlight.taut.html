3// Keep scanner offsets in UTF-8 bytes. Leave embedded Tauto expressions
4// unclassified here rather than painting their contents as one DSL token.
5pub fn math_token_kind(t: SynTok): Option(Text) ! {Div} {
6  if t.skipped || t.kind == "TautoExpr" { None } else {
7    if t.kind == "Diff" { Some("function") } else {
8      if t.text == "ε" || t.kind == "Identity" || t.kind == "ZeroLit" || t.kind == "OneLit" {
9        Some("enumMember")
10      } else {
11        if t.kind == "Name" { Some("variable") } else {
12          if t.kind == "Num" { Some("number") } else { Some("operator") }
13        }
14      }
15    }
16  }
17}
18pub fn next_is_call(ts: List(SynTok)): Bool ! {Div} {
19  match ts {
20    Nil => false
21    Cons(t, rest) => if t.skipped { next_is_call(rest) } else { t.text == "(" }
22  }
23}
24pub fn math_tokens(ts: List(SynTok)): List(SynHighlight) ! {Div} {
25  match ts {
26    Nil => Nil
27    Cons(t, rest) => match math_token_kind(t) {
28      None => math_tokens(rest)
29      Some(kind) => {
30        let semantic := if t.kind == "Name" && next_is_call(rest) { "function" } else { kind }
31        Cons(SynHighlight { start: t.start, end: t.end, kind: semantic }, math_tokens(rest))
32      }
33    }
34  }
35}
