12pub fn decision_flag(P: Prop, d: Decision(P)): Bool {
13  ⊢ Bool from match d {
14    Yes(h) => true
15    No(h) => false
16  }
17}

19// If the old real contains the middle, keep the left interval.
20// Otherwise move strictly to its right. Both choices leave a real gap.
21pub fn branch(box: Interval, take_left: Bool): Interval {
22  ⊢ Interval from match take_left {
23    true => Between(left(box), middle(box), left_middle(box))
24    false => Between(
25      rational_between(middle(box), right(box)),
26      right(box),
27      rational_between_right(middle(box), right(box), middle_right(box))
28    )
29  }
30}
31
32pub theorem branch_inside(box: Interval, take_left: Bool): Inside(branch(box, take_left), box) {
33  ⊢ Inside(branch(box, take_left), box) from match take_left {
34    true => AndIntro(
35      Rat::le_refl(left(box)),
36      rational_lt_le(middle(box), right(box), middle_right(box))
37    )
38    false => AndIntro(
39      rational_lt_le(
40        left(box),
41        rational_between(middle(box), right(box)),
42        rational_lt_trans(
43          left(box),
44          middle(box),
45          rational_between(middle(box), right(box)),
46          left_middle(box),
47          rational_between_left(middle(box), right(box), middle_right(box))
48        )
49      ),
50      Rat::le_refl(right(box))
51    )
52  }
53}

55pub fn Excludes(x: Real, box: Interval): Prop {
56  (
57    y: Real,
58    covers_lower: (q: Rat, lt: RationalLt(q, left(box))) -> below(y, q),
59    avoids_upper: ¬below(y, right(box))
60  ) -> ¬SameReal(x, y)
61}
62
63pub theorem branch_excludes(
64  x: Real,
65  box: Interval,
66  d: Decision(below(x, middle(box)))
67): Excludes(x, branch(box, decision_flag(below(x, middle(box)), d))) {
68  ⊢ Excludes(x, branch(box, decision_flag(below(x, middle(box)), d))) from match d {
69    Yes(member) => (y: Real, low: (q: Rat, lt: RationalLt(q, left(box))) -> below(y, q), high: ¬below(y, middle(box))) => (same: SameReal(x, y)) => ⊢ False from match same {
70      AndIntro(xy, yx) => high(xy(middle(box), member))
71    }
72    No(outside) => (y: Real, low: (q: Rat, lt: RationalLt(q, rational_between(middle(box), right(box)))) -> below(y, q), high: ¬below(y, right(box))) => (same: SameReal(x, y)) => ⊢ False from match same {
73      AndIntro(xy, yx) => outside(
74        yx(
75          middle(box),
76          low(middle(box), rational_between_left(middle(box), right(box), middle_right(box)))
77        )
78      )
79    }
80  }
81}

83pub fn unit_interval(): Interval {
84  Between(
85    Rat::of_nat(0),
86    Rat::of_nat(1),
87    (bad: RatLe(Rat::of_nat(1), Rat::of_nat(0))) => false_ne_true(bad)
88  )
89}

91pub fn diagonal_intervals(rows: (Nat) -> Real, decide: (P: Prop) -> Decision(P), n: Nat): Interval {
92  ⊢ Interval from match n {
93    Zero => unit_interval()
94    Succ(k) => branch(
95      diagonal_intervals(rows, decide, k),
96      decision_flag(
97        below(rows(k), middle(diagonal_intervals(rows, decide, k))),
98        decide(below(rows(k), middle(diagonal_intervals(rows, decide, k))))
99      )
100    )
101  }
102}

104pub theorem diagonal_nested(
105  rows: (Nat) -> Real,
106  decide: (P: Prop) -> Decision(P),
107  n: Nat
108): Inside(diagonal_intervals(rows, decide, Succ(n)), diagonal_intervals(rows, decide, n)) {
109  branch_inside(
110    diagonal_intervals(rows, decide, n),
111    decision_flag(
112      below(rows(n), middle(diagonal_intervals(rows, decide, n))),
113      decide(below(rows(n), middle(diagonal_intervals(rows, decide, n))))
114    )
115  )
116}

118pub theorem diagonal_inhabited(
119  rows: (Nat) -> Real,
120  decide: (P: Prop) -> Decision(P)
121): ∃ q: Rat. interval_lower((n: Nat) => diagonal_intervals(rows, decide, n), q) {
122  ExIntro(
123    Rat::of_int(IntNegSucc(0)),
124    ExIntro(0, (bad: RatLe(Rat::of_nat(0), Rat::of_int(IntNegSucc(0)))) => false_ne_true(bad))
125  )
126}

128pub fn real_diagonal(rows: (Nat) -> Real, decide: (P: Prop) -> Decision(P)): Real {
129  interval_real(
130    (n: Nat) => diagonal_intervals(rows, decide, n),
131    (n: Nat) => diagonal_nested(rows, decide, n),
132    diagonal_inhabited(rows, decide)
133  )
134}

136pub theorem real_diagonal_differs(
137  rows: (Nat) -> Real,
138  decide: (P: Prop) -> Decision(P),
139  n: Nat
140): ¬SameReal(rows(n), real_diagonal(rows, decide)) {
141  branch_excludes(
142    rows(n),
143    diagonal_intervals(rows, decide, n),
144    decide(below(rows(n), middle(diagonal_intervals(rows, decide, n)))),
145    real_diagonal(rows, decide),
146    (q: Rat, lt: RationalLt(q, left(diagonal_intervals(rows, decide, Succ(n))))) => ExIntro(
147      Succ(n),
148      lt
149    ),
150    interval_cut_bounded(
151      (k: Nat) => diagonal_intervals(rows, decide, k),
152      (k: Nat) => diagonal_nested(rows, decide, k),
153      Succ(n)
154    )
155  )
156}

158// Covers quantifies over real values, so alternative representations of
159// a cut cannot turn an incomplete enumeration into a complete one.
160pub fn CoversReals(rows: (Nat) -> Real): Prop {
161  (x: Real) -> (∃ n: Nat. SameReal(rows(n), x))
162}
163
164pub theorem reals_not_enumerable(
165  rows: (Nat) -> Real,
166  decide: (P: Prop) -> Decision(P)
167): ¬CoversReals(rows) {
168  (covers: CoversReals(rows)) => ⊢ False from match covers(real_diagonal(rows, decide)) {
169    ExIntro(n, same) => real_diagonal_differs(rows, decide, n, same)
170  }
171}

173pub theorem reals_uncountable(
174  rows: (Nat) -> Real,
175  decide: (P: Prop) -> Decision(P)
176): ¬Surjective(rows) {
177  (onto: Surjective(rows)) => ⊢ False from match onto(real_diagonal(rows, decide)) {
178    ImageOf(n, h) => real_diagonal_differs(
179      rows,
180      decide,
181      n,
182      Eq::subst((x: Real) => SameReal(rows(n), x), h, same_real_refl(rows(n)))
183    )
184  }
