9pub fn interval_lower(boxes: (Nat) -> Interval, q: Rat): Prop {
10  ∃ n: Nat. RationalLt(q, left(boxes(n)))
11}
12
13pub theorem interval_cut_downward(
14  boxes: (Nat) -> Interval,
15  p: Rat,
16  q: Rat,
17  pq: RationalLt(p, q),
18  h: interval_lower(boxes, q)
19): interval_lower(boxes, p) {
20  ⊢ interval_lower(boxes, p) from match h {
21    ExIntro(n, lt) => ExIntro(n, rational_lt_trans(p, q, left(boxes(n)), pq, lt))
22  }
23}
24pub theorem interval_cut_rounded(
25  boxes: (Nat) -> Interval,
26  p: Rat,
27  h: interval_lower(boxes, p)
28): ∃ q: Rat. RationalLt(p, q) ∧ interval_lower(boxes, q) {
29  ⊢ ∃ q: Rat. RationalLt(p, q) ∧ interval_lower(boxes, q) from match h {
30    ExIntro(n, lt) => ExIntro(
31      rational_between(p, left(boxes(n))),
32      AndIntro(
33        rational_between_left(p, left(boxes(n)), lt),
34        ExIntro(n, rational_between_right(p, left(boxes(n)), lt))
35      )
36    )
37  }
38}
39pub theorem interval_cut_bounded(
40  boxes: (Nat) -> Interval,
41  nested: (n: Nat) -> Inside(boxes(Succ(n)), boxes(n)),
42  n: Nat
43): ¬interval_lower(boxes, right(boxes(n))) {
44  (h: interval_lower(boxes, right(boxes(n)))) => ⊢ False from match h {
45    ExIntro(k, lt) => lt(
46      rational_lt_le(left(boxes(k)), right(boxes(n)), chain_cross(boxes, nested, k, n))
47    )
48  }
49}

51pub fn interval_real(
52  boxes: (Nat) -> Interval,
53  nested: (n: Nat) -> Inside(boxes(Succ(n)), boxes(n)),
54  inhabited: ∃ q: Rat. interval_lower(boxes, q)
55): Real {
56  Dedekind(
57    (q: Rat) => interval_lower(boxes, q),
58    inhabited,
59    ExIntro(right(boxes(0)), interval_cut_bounded(boxes, nested, 0)),
60    (p: Rat, q: Rat, pq: RationalLt(p, q), h: interval_lower(boxes, q)) => interval_cut_downward(
61      boxes,
62      p,
63      q,
64      pq,
65      h
66    ),
67    (p: Rat, h: interval_lower(boxes, p)) => interval_cut_rounded(boxes, p, h)
68  )
