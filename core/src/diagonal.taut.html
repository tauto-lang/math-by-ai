4// Cantor's construction needs neither an infinite list nor a limit:
5// a sequence is a function, and its diagonal is another function.
6pub fn diagonal(rows: (Nat) -> (Nat) -> Bool): (Nat) -> Bool {
7  (n: Nat) => not(rows(n)(n))
8}
9
10pub theorem not_ne_self(b: Bool): ¬(not(b) = b) {
11  ⊢ ¬(not(b) = b) from match b {
12    false => true_ne_false
13    true => false_ne_true
14  }
15}
16
17// The n-th coordinate alone separates the diagonal from the n-th row.
18pub theorem diagonal_differs(rows: (Nat) -> (Nat) -> Bool, n: Nat): ¬(rows(n) = diagonal(rows)) {
19  (h: rows(n) = diagonal(rows)) => not_ne_self(
20    rows(n)(n)
21  )(Eq::sym(Eq::congr((s: (Nat) -> Bool) => s(n), h)))
22}
23
24pub theorem binary_sequences_uncountable(rows: (Nat) -> (Nat) -> Bool): ¬Surjective(rows) {
25  (onto: Surjective(rows)) => ⊢ False from match onto(diagonal(rows)) {
26    ImageOf(n, h) => diagonal_differs(rows, n, h)
27  }
28}
