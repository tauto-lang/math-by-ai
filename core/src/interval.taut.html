8pub(open) inductive Interval: Type {
9  Between(left: Rat, right: Rat, positive: RationalLt(left, right)): Interval
10}
11pub fn left(i: Interval): Rat { ⊢ Rat from match i { Between(a, b, h) => a } }
12pub fn right(i: Interval): Rat { ⊢ Rat from match i { Between(a, b, h) => b } }
13pub theorem interval_positive(i: Interval): RationalLt(left(i), right(i)) {
14  ⊢ RationalLt(left(i), right(i)) from match i { Between(a, b, h) => h }
15}
16pub fn middle(i: Interval): Rat { rational_between(left(i), right(i)) }
17pub theorem left_middle(i: Interval): RationalLt(left(i), middle(i)) {
18  rational_between_left(left(i), right(i), interval_positive(i))
19}
20pub theorem middle_right(i: Interval): RationalLt(middle(i), right(i)) {
21  rational_between_right(left(i), right(i), interval_positive(i))
22}

24pub fn Inside(inner: Interval, outer: Interval): Prop {
25  RatLe(left(outer), left(inner)) ∧ RatLe(right(inner), right(outer))
26}
27pub theorem inside_left(
28  inner: Interval,
29  outer: Interval,
30  h: Inside(inner, outer)
31): RatLe(left(outer), left(inner)) {
32  ⊢ RatLe(left(outer), left(inner)) from match h { AndIntro(a, b) => a }
33}
34pub theorem inside_right(
35  inner: Interval,
36  outer: Interval,
37  h: Inside(inner, outer)
38): RatLe(right(inner), right(outer)) {
39  ⊢ RatLe(right(inner), right(outer)) from match h { AndIntro(a, b) => b }
40}
41
42pub theorem chain_left(
43  boxes: (Nat) -> Interval,
44  nested: (n: Nat) -> Inside(boxes(Succ(n)), boxes(n)),
45  n: Nat
46): RatLe(left(boxes(0)), left(boxes(n))) {
47  proof!(```
48    By induction on n.
49      [zero] Case Zero. By Rat::le_refl.
50      [succ] Case Succ(k). By Rat::le_trans.
51    %%
52    [zero] (left(boxes(0)))
53    [succ] (left(boxes(0)), left(boxes(k)), left(boxes(Succ(k))), ih, inside_left(boxes(Succ(k)), boxes(k), nested(k)))
54  ```)
55}
56pub theorem chain_right(
57  boxes: (Nat) -> Interval,
58  nested: (n: Nat) -> Inside(boxes(Succ(n)), boxes(n)),
59  n: Nat
60): RatLe(right(boxes(n)), right(boxes(0))) {
61  ⊢ RatLe(right(boxes(n)), right(boxes(0))) from match n {
62    Zero => Rat::le_refl(right(boxes(0)))
63    Succ(k) => Rat::le_trans(
64      right(boxes(Succ(k))),
65      right(boxes(k)),
66      right(boxes(0)),
67      inside_right(boxes(Succ(k)), boxes(k), nested(k)),
68      chain_right(boxes, nested, k)
69    )
70  }
71}

73// Every left endpoint is below every right endpoint. Dropping the first
74// interval from both indices avoids arithmetic on indices in this proof.
75pub theorem chain_cross(
76  boxes: (Nat) -> Interval,
77  nested: (n: Nat) -> Inside(boxes(Succ(n)), boxes(n)),
78  n: Nat,
79  m: Nat
80): RationalLt(left(boxes(n)), right(boxes(m))) {
81  ⊢ RationalLt(left(boxes(n)), right(boxes(m))) from match n {
82    Zero => rational_le_lt(
83      left(boxes(0)),
84      left(boxes(m)),
85      right(boxes(m)),
86      chain_left(boxes, nested, m),
87      interval_positive(boxes(m))
88    )
89    Succ(k) => ⊢ RationalLt(left(boxes(Succ(k))), right(boxes(m))) from match m {
90      Zero => rational_lt_le_trans(
91        left(boxes(Succ(k))),
92        right(boxes(Succ(k))),
93        right(boxes(0)),
94        interval_positive(boxes(Succ(k))),
95        chain_right(boxes, nested, Succ(k))
96      )
97      Succ(j) => chain_cross((i: Nat) => boxes(Succ(i)), (i: Nat) => nested(Succ(i)), k, j)
98    }
99  }
