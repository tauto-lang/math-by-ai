8// Strict order on rational VALUES, independent of fraction representations.
9pub fn RationalLt(a: Rat, b: Rat): Prop { ¬RatLe(b, a) }
10
11pub theorem rational_lt_irrefl(a: Rat): ¬RationalLt(a, a) {
12  (h: RationalLt(a, a)) => h(Rat::le_refl(a))
13}
14
15pub theorem rational_lt_le(a: Rat, b: Rat, h: RationalLt(a, b)): RatLe(a, b) {
16  ⊢ RatLe(a, b) from match Rat::le_total(a, b) {
17    OrInl(ab) => ab
18    OrInr(ba) => False::elim(h(ba))
19  }
20}

22pub theorem rational_lt_trans(
23  a: Rat,
24  b: Rat,
25  c: Rat,
26  ab: RationalLt(a, b),
27  bc: RationalLt(b, c)
28): RationalLt(a, c) {
29  (ca: RatLe(c, a)) => ab(Rat::le_trans(b, c, a, rational_lt_le(b, c, bc), ca))
30}

32pub theorem rational_le_lt(
33  a: Rat,
34  b: Rat,
35  c: Rat,
36  ab: RatLe(a, b),
37  bc: RationalLt(b, c)
38): RationalLt(a, c) {
39  (ca: RatLe(c, a)) => bc(Rat::le_trans(c, a, b, ca, ab))
40}

42pub theorem rational_lt_le_trans(
43  a: Rat,
44  b: Rat,

46  ab: RationalLt(a, b),

48): RationalLt(a, c) {

52pub theorem integer_le_add_cancel(
53  x: Int,
54  y: Int,
55  z: Int,
56  h: IntLe(Int::add(x, z), Int::add(y, z))
57): IntLe(x, y) {
58  let ex: Int::add(
59    Int::add(x, z),
60    Int::neg(z)
61  ) = x := by int_norm!(): Int::add(Int::add(x, z), Int::neg(z)) = x {}
62  let ey: Int::add(
63    Int::add(y, z),
64    Int::neg(z)
65  ) = y := by int_norm!(): Int::add(Int::add(y, z), Int::neg(z)) = y {}
66  Int::le_rw(
67    Int::add(Int::add(x, z), Int::neg(z)),
68    x,
69    Int::add(Int::add(y, z), Int::neg(z)),
70    y,
71    ex,
72    ey,
73    Int::le_add(Int::add(x, z), Int::add(y, z), Int::neg(z), h)
74  )
75}

77// The mediant (a+c)/(b+d) lies strictly between a/b and c/d.
78// Using it instead of a midpoint keeps the arithmetic proof small.
79pub fn rational_between(x: Rat, y: Rat): Rat {
80  ⊢ Rat from match x {
81    Mk(a, p) => ⊢ Rat from match y {
82      Mk(b, q) => Rat::Mk(Int::add(a, b), Succ(Nat::add(p, q)))
83    }
84  }
85}

87pub theorem rational_between_left(
88  x: Rat,
89  y: Rat,
90  h: RationalLt(x, y)
91): RationalLt(x, rational_between(x, y)) {
92  ⊢ RationalLt(x, rational_between(x, y)) from match x {
93    Mk(a, p) => ⊢ RationalLt(Rat::Mk(a, p), rational_between(Rat::Mk(a, p), y)) from match y {
94      Mk(b, q) => (bad: RatLe(rational_between(Rat::Mk(a, p), Rat::Mk(b, q)), Rat::Mk(a, p))) => h(
95        integer_le_add_cancel(
96          Int::mul(b, IntOfNat(Succ(p))),
97          Int::mul(a, IntOfNat(Succ(q))),
98          Int::mul(a, IntOfNat(Succ(p))),
99          Int::le_rw(
100            Int::mul(Int::add(a, b), IntOfNat(Succ(p))),
101            Int::add(Int::mul(b, IntOfNat(Succ(p))), Int::mul(a, IntOfNat(Succ(p)))),
102            Int::mul(a, IntOfNat(Succ(Succ(Nat::add(p, q))))),
103            Int::add(Int::mul(a, IntOfNat(Succ(q))), Int::mul(a, IntOfNat(Succ(p)))),
104            by int_norm!(): Int::mul(
105              Int::add(a, b),
106              IntOfNat(Succ(p))
107            ) = Int::add(Int::mul(b, IntOfNat(Succ(p))), Int::mul(a, IntOfNat(Succ(p)))) {},
108            by int_norm!(): Int::mul(
109              a,
110              IntOfNat(Succ(Succ(Nat::add(p, q))))
111            ) = Int::add(Int::mul(a, IntOfNat(Succ(q))), Int::mul(a, IntOfNat(Succ(p)))) {},
112            bad
113          )
114        )
115      )
116    }
117  }
118}

124): RationalLt(rational_between(x, y), y) {
