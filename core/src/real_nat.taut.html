12pub theorem negative_one_lt_nat(n: Nat): RationalLt(Rat::of_int(IntNegSucc(0)), Rat::of_nat(n)) {
13  (bad: RatLe(Rat::of_nat(n), Rat::of_int(IntNegSucc(0)))) => false_ne_true(
14    Int::le_rw(
15      Int::mul(IntOfNat(n), 1),
16      IntOfNat(n),
17      Int::mul(IntNegSucc(0), 1),
18      IntNegSucc(0),
19      Int::mul_one(IntOfNat(n)),
20      Int::mul_one(IntNegSucc(0)),
21      bad
22    )
23  )
24}

26pub fn natural_real(n: Nat): Real {
27  Dedekind(
28    (q: Rat) => RationalLt(q, Rat::of_nat(n)),
29    ExIntro(Rat::of_int(IntNegSucc(0)), negative_one_lt_nat(n)),
30    ExIntro(Rat::of_nat(n), rational_lt_irrefl(Rat::of_nat(n))),
31    (p: Rat, q: Rat, pq: RationalLt(p, q), qn: RationalLt(q, Rat::of_nat(n))) => rational_lt_trans(
32      p,
33      q,
34      Rat::of_nat(n),
35      pq,
36      qn
37    ),
38    (p: Rat, pn: RationalLt(p, Rat::of_nat(n))) => ExIntro(
39      rational_between(p, Rat::of_nat(n)),
40      AndIntro(
41        rational_between_left(p, Rat::of_nat(n), pn),
42        rational_between_right(p, Rat::of_nat(n), pn)
43      )
44    )
45  )
46}

48// An inclusion of the two cuts implies the corresponding rational inequality.
49pub theorem natural_real_le(
50  n: Nat,
51  m: Nat,
52  h: RealLe(natural_real(n), natural_real(m))
53): RatLe(Rat::of_nat(n), Rat::of_nat(m)) {
54  ⊢ RatLe(Rat::of_nat(n), Rat::of_nat(m)) from match Int::le(IntOfNat(n), IntOfNat(m)) eqn comparison {
55    true => Int::le_rw(
56      IntOfNat(n),
57      Int::mul(IntOfNat(n), 1),
58      IntOfNat(m),
59      Int::mul(IntOfNat(m), 1),
60      Eq::sym(Int::mul_one(IntOfNat(n))),
61      Eq::sym(Int::mul_one(IntOfNat(m))),
62      comparison
63    )
64    false => False::elim(
65      rational_lt_irrefl(
66        Rat::of_nat(m),
67        h(
68          Rat::of_nat(m),
69          (bad: RatLe(Rat::of_nat(n), Rat::of_nat(m))) => false_ne_true(
70            Eq::trans(
71              Eq::sym(comparison),
72              Int::le_rw(
73                Int::mul(IntOfNat(n), 1),
74                IntOfNat(n),
75                Int::mul(IntOfNat(m), 1),
76                IntOfNat(m),
77                Int::mul_one(IntOfNat(n)),
78                Int::mul_one(IntOfNat(m)),
79                bad
80              )
81            )
82          )
83        )
84      )
85    )
86  }
87}

89pub theorem natural_real_injective(
90  n: Nat,
91  m: Nat,
92  same: SameReal(natural_real(n), natural_real(m))
93): n = m {
94  ⊢ n = m from match same {
95    AndIntro(nm, mn) => Eq::congr(
96      Int::abs_nat,
97      Eq::trans(
98        Eq::sym(Int::mul_one(IntOfNat(n))),
99        Eq::trans(
100          Rat::le_antisymm(
101            Rat::of_nat(n),
102            Rat::of_nat(m),
103            natural_real_le(n, m, nm),
104            natural_real_le(m, n, mn)
105          ),
106          Int::mul_one(IntOfNat(m))
107        )
108      )
109    )
110  }
111}

113pub theorem natural_real_function_injective(): Injective(natural_real) {
114  (n: Nat, m: Nat, h: natural_real(n) = natural_real(m)) => natural_real_injective(
115    n,
116    m,
117    Eq::subst((x: Real) => SameReal(natural_real(n), x), h, same_real_refl(natural_real(n)))
118  )
