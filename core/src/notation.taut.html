38pub fn logic_args(xs: List(logic_notation::Arguments_Rep1)): List(SynExpr) ! {macro_error, Div} {
39  match xs { Nil => Nil
40    Cons(x, rest) => Cons(logic_expr(x.expr), logic_args(rest)) }
41}
42pub fn logic_atom(a: logic_notation::Atom): SynExpr ! {macro_error, Div} {
43  match a {
44    Num(t) => ExNat(t.text, t.span)
45    Embed(e) => e
46    Paren(p) => logic_expr(p.expr)
47    Negation(n) => { let p := logic_atom(n.atom)
48      quote { ¬($p) } }
49    Reference(r) => match r.arguments {
50      None => logic_name(r.name)
51      Some(args) => ExApp(
52        logic_name(r.name),
53        Cons(logic_expr(args.expr), logic_args(args.rep)),
54        r.span
55      )
56    }
57  }
58}
59pub fn logic_comparison(c: logic_notation::Comparison): SynExpr ! {macro_error, Div} {
60  let a := logic_atom(c.atom)
61  match c.opt { None => a
62    Some(r) => { let b := logic_atom(r.atom)
63      quote { $a = $b } }
64  }
65}
66pub fn logic_ands(
67  acc: SynExpr,
68  xs: List(logic_notation::Conjunction_Rep1)
69): SynExpr ! {macro_error, Div} {
70  match xs { Nil => acc
71    Cons(x, rest) => { let b := logic_comparison(x.comparison)

74}
75pub fn logic_conjunction(c: logic_notation::Conjunction): SynExpr ! {macro_error, Div} {
76  logic_ands(logic_comparison(c.comparison), c.rep)
77}
78pub fn logic_ors(
79  acc: SynExpr,
80  xs: List(logic_notation::Disjunction_Rep1)
81): SynExpr ! {macro_error, Div} {
82  match xs { Nil => acc
83    Cons(x, rest) => { let b := logic_conjunction(x.conjunction)

85  }

87pub fn logic_disjunction(d: logic_notation::Disjunction): SynExpr ! {macro_error, Div} {
88  logic_ors(logic_conjunction(d.conjunction), d.rep)
89}
90pub fn logic_expr(e: logic_notation::Expr): SynExpr ! {macro_error, Div} {
91  let a := logic_disjunction(e.disjunction)
92  match e.opt { None => a
93    Some(r) => { let b := logic_expr(r.expr)
94      quote { ($a) -> $b } }
95  }
96}
