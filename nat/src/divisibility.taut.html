10// A proof carries a witness. No quotient computation or Boolean decision is assumed.
11pub(open) inductive Divides(a: Nat, b: Nat): Prop {
12  By(k: Nat, equation: nat!(`a * k = b`)): Divides(a, b)
13}
14pub theorem divides_refl(a: Nat): Divides(a, a) { By(1, Nat::mul_one_right(a)) }
15pub theorem one_divides(a: Nat): Divides(1, a) { By(a, Nat::mul_one_left(a)) }
16pub theorem divides_zero(a: Nat): Divides(a, 0) { By(0, Nat::mul_zero_right(a)) }
17pub theorem divides_trans(
18  a: Nat,
19  b: Nat,
20  c: Nat,
21  ab: Divides(a, b),
22  bc: Divides(b, c)
23): Divides(a, c) {
24  proof!(```
25      Consider ab.
26        Case By(k, hk). Consider bc.
27          [step] Case By(l, hl). We have h: $nat!(`a * (k * l)`)
28            = $nat!(`(a * k) * l`) by Nat::mul_assoc(a, k, l)
29            = $nat!(`b * l`) by hk
30            = c by hl. By By.
31      %%
32      [step] (Nat::mul(k, l), h)
33    ```)
34}
35pub theorem divides_add(
36  a: Nat,
37  b: Nat,
38  c: Nat,
39  ab: Divides(a, b),
40  ac: Divides(a, c)
41): Divides(a, nat!(`b + c`)) {
42  ⊢ Divides(a, nat!(`b + c`)) from match ab {
43    By(k, hk) => ⊢ Divides(a, nat!(`b + c`)) from match ac {
44      By(l, hl) => By(
45        nat!(`k + l`),
46        calc!(```
47          $nat!(`a * (k + l)`)
48          = $nat!(`a * k + a * l`) from $Nat::mul_add_distrib(a, k, l)
49          = $nat!(`b + a * l`) from $Eq::congr((x: Nat) => nat!(`x + a * l`), hk)
50          = $nat!(`b + c`) from $Eq::congr((x: Nat) => nat!(`b + x`), hl)
51          ```)
52      )
53    }
54  }
55}
56pub theorem divides_mul_right(
57  a: Nat,
58  b: Nat,
59  c: Nat,
60  ab: Divides(a, b)
61): Divides(a, nat!(`b * c`)) {
62  ⊢ Divides(a, nat!(`b * c`)) from match ab {
63    By(k, hk) => By(
64      nat!(`k * c`),
65      calc!(```
66        $nat!(`a * (k * c)`)
67        = $nat!(`(a * k) * c`) from $Eq::sym(Nat::mul_assoc(a, k, c))
68        = $nat!(`b * c`) from $Eq::congr((x: Nat) => nat!(`x * c`), hk)
69        ```)
70    )
71  }
72}
73pub theorem divides_mul_left(a: Nat, b: Nat, c: Nat, ab: Divides(a, b)): Divides(a, nat!(`c * b`)) {
74  ⊢ Divides(a, nat!(`c * b`)) from match ab {
75    By(k, hk) => By(
76      nat!(`k * c`),
77      calc!(```
78        $nat!(`a * (k * c)`)
79        = $nat!(`(a * k) * c`) from $Eq::sym(Nat::mul_assoc(a, k, c))
80        = $nat!(`b * c`) from $Eq::congr((x: Nat) => nat!(`x * c`), hk)
81        = $nat!(`c * b`) from $Nat::mul_comm(b, c)
82        ```)
83    )
84  }
85}
86pub theorem zero_divides_only_zero(b: Nat, h: Divides(0, b)): nat!(`b = 0`) {
87  ⊢ nat!(`b = 0`) from match h {
88    By(k, hk) => calc!(```
89      $b
90      = $nat!(`0 * k`) from $Eq::sym(hk)
91      = $nat!(`0`) from $Nat::mul_zero_left(k)
92      ```)
93  }
94}

96pub(open) inductive AllDivisible(a: Nat): (List(Nat)) -> Prop {
97  AllNil: AllDivisible(a, Nil)
98  AllCons(
99    x: Nat,
100    xs: List(Nat),
101    head: Divides(a, x),
102    tail: AllDivisible(a, xs)
103  ): AllDivisible(a, Cons(x, xs))
104}

106pub theorem divides_bound(a: Nat, b: Nat, h: Divides(a, Succ(b))): nat!(`a ≤ Succ(b)`) {
107  ⊢ nat!(`a ≤ Succ(b)`) from match h {
108    By(k, hk) => ⊢ nat!(`a ≤ Succ(b)`) from match k {
109      Zero => False::elim(Nat::zero_ne_succ(b, Eq::trans(Eq::sym(Nat::mul_zero_right(a)), hk)))
110      Succ(j) => Eq::subst(
111        (x: Nat) => nat!(`a ≤ x`),
112        hk,
113        Eq::subst(
114          (x: Nat) => nat!(`a ≤ x`),
115          Eq::sym(Nat::mul_succ_right(a, j)),
116          Nat::le_add_left(nat!(`a * j`), a)
117        )
118      )
119    }
120  }
121}
122pub theorem divides_antisym(a: Nat, b: Nat, ab: Divides(a, b), ba: Divides(b, a)): nat!(`a = b`) {
123  ⊢ nat!(`a = b`) from match b {
124    Zero => zero_divides_only_zero(a, ba)
125    Succ(k) => ⊢ nat!(`a = Succ(k)`) from match a {
126      Zero => False::elim(Nat::zero_ne_succ(k, Eq::sym(zero_divides_only_zero(Succ(k), ab))))
127      Succ(j) => Nat::le_antisym(
128        Succ(j),
129        Succ(k),
130        divides_bound(Succ(j), k, ab),
131        divides_bound(Succ(k), j, ba)
132      )
133    }
134  }
