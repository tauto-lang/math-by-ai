13pub theorem divides_remainder_zero(
14  n: Nat,
15  k: Nat,
16  h: nat_rel!(`Succ(k) | n`)
17): nat!(`Nat::rem_nz(n, Succ(k), Nat::succ_ne_zero(k)) = 0`) {
18  ⊢ nat!(`Nat::rem_nz(n, Succ(k), Nat::succ_ne_zero(k)) = 0`) from match h {
19    By(q, eq) => Eq::subst(
20      (x: Nat) => nat!(`Nat::rem_nz(x, Succ(k), Nat::succ_ne_zero(k)) = 0`),
21      eq,
22      Nat::mod_of(q, 0, k, Refl)
23    )
24  }
25}
26pub fn bool_neg(b: Bool): Bool { ⊢ Bool from match b { false => true, true => false } }
27pub fn divisor_test(n: Nat, d: Nat): Bool {
28  ⊢ Bool from match d {
29    Zero => true
30    Succ(k) => ⊢ Bool from match k {
31      Zero => true
32      Succ(j) => Bool::or(
33        Nat::eq(d, n),
34        bool_neg(Nat::eq(Nat::rem_nz(n, d, Nat::succ_ne_zero(k)), 0))
35      )
36    }
37  }
38}
39pub fn trial(n: Nat, bound: Nat): Bool {
40  ⊢ Bool from match bound {
41    Zero => true
42    Succ(k) => Bool::and(trial(n, k), divisor_test(n, Succ(k)))
43  }
44}
45pub fn is_prime(n: Nat): Bool { Bool::and(Nat::lt(1, n), trial(n, n)) }

47pub theorem boolean_cases(b: Bool): logic!(`b = false ∨ b = true`) {
48  ⊢ logic!(`b = false ∨ b = true`) from match b { false => OrInl(Refl)
49    true => OrInr(Refl) }
50}
51theorem divisor_test_with_rem(
52  n: Nat,
53  k: Nat,
54  eq: Nat::eq(Succ(Succ(k)), n) = false,
55  rem: Nat::rem_nz(n, Succ(Succ(k)), Nat::succ_ne_zero(Succ(k))) = 0
56): divisor_test(n, Succ(Succ(k))) = false {
57  Eq::trans(
58    Eq::congr(
59      (e: Bool) => Bool::or(
60        e,
61        bool_neg(Nat::eq(Nat::rem_nz(n, Succ(Succ(k)), Nat::succ_ne_zero(Succ(k))), 0))
62      ),
63      eq
64    ),
65    Eq::congr((r: Nat) => bool_neg(Nat::eq(r, 0)), rem)
66  )
67}
68pub theorem divisor_test_sound(
69  n: Nat,
70  d: Nat,
71  pos: nat!(`1 < n`),
72  good: divisor_test(n, d) = true,
73  divides: nat_rel!(`d | n`)
74): logic!(`d = 1 ∨ d = n`) {
75  ⊢ logic!(`d = 1 ∨ d = n`) from match d {
76    Zero => False::elim(
77      false_ne_true(
78        Eq::trans(
79          Eq::sym(Eq::congr((x: Nat) => Nat::lt(1, x), zero_divides_only_zero(n, divides))),
80          pos
81        )
82      )
83    )
84    Succ(k) => ⊢ logic!(`Succ(k) = 1 ∨ Succ(k) = n`) from match k {
85      Zero => OrInl(Refl)
86      Succ(j) => Or::elim(
87        boolean_cases(Nat::eq(Succ(Succ(j)), n)),
88        (ne: Nat::eq(Succ(Succ(j)), n) = false) => False::elim(
89          false_ne_true(
90            Eq::trans(
91              Eq::sym(divisor_test_with_rem(n, j, ne, divides_remainder_zero(n, Succ(j), divides))),
92              good
93            )
94          )
95        ),
96        (eq: Nat::eq(Succ(Succ(j)), n) = true) => OrInr(Nat::eq_sound(Succ(Succ(j)), n, eq))
97      )
98    }
99  }
100}
101theorem range_sound(
102  n: Nat,
103  bound: Nat,
104  pos: nat!(`1 < n`),
105  good: trial(n, bound) = true,
106  d: Nat,
107  le: nat!(`d ≤ bound`),
108  divides: nat_rel!(`d | n`)
109): logic!(`d = 1 ∨ d = n`) {
110  ⊢ logic!(`d = 1 ∨ d = n`) from match bound {
111    Zero => ⊢ logic!(`d = 1 ∨ d = n`) from match d {
112      Zero => divisor_test_sound(n, 0, pos, Refl, divides)
113      Succ(k) => False::elim(false_ne_true(Eq::trans(Eq::sym(Nat::lt_zero_false(k)), le)))
114    }
115    Succ(k) => Or::elim(
116      boolean_cases(Nat::eq(d, Succ(k))),
117      (ne: Nat::eq(d, Succ(k)) = false) => range_sound(
118        n,
119        k,
120        pos,
121        Bool::and_true_left(trial(n, k), divisor_test(n, Succ(k)), good),
122        d,
123        Nat::lt_succ_ne(d, Succ(k), le, ne),
124        divides
125      ),
126      (eq: Nat::eq(d, Succ(k)) = true) => divisor_test_sound(
127        n,
128        d,
129        pos,
130        Eq::subst(
131          (x: Nat) => divisor_test(n, x) = true,
132          Eq::sym(Nat::eq_sound(d, Succ(k), eq)),
133          Bool::and_true_right(trial(n, k), divisor_test(n, Succ(k)), good)
134        ),
135        divides
136      )
137    )
138  }
139}
140pub theorem is_prime_sound(n: Nat, good: is_prime(n) = true): Prime(n) {
141  PrimeBy(Bool::and_true_left(Nat::lt(1, n), trial(n, n), good),
142    (d: Nat, h: nat_rel!(`d | n`)) => range_sound(n, n,
143      Bool::and_true_left(Nat::lt(1, n), trial(n, n), good),
144      Bool::and_true_right(Nat::lt(1, n), trial(n, n), good), d,
145      ⊢ nat!(`d ≤ n`) from match n {
146        Zero => False::elim(false_ne_true(Bool::and_true_left(Nat::lt(1, 0), trial(0, 0), good)))
147        Succ(k) => divides_bound(d, k, h)
148      }, h))
149}

151pub fn find_factor(n: Nat, bound: Nat): Nat {
152  ⊢ Nat from match bound {
153    Zero => 0
154    Succ(k) => if divisor_test(n, Succ(k)) { find_factor(n, k) } else { Succ(k) }
155  }
156}
157pub(open) inductive BadCandidate(n: Nat, bound: Nat, d: Nat): Prop {
158  Candidate(bad: divisor_test(n, d) = false, within: nat!(`d ≤ bound`)): BadCandidate(n, bound, d)
159}
160pub theorem candidate_bad(
161  n: Nat,
162  bound: Nat,
163  d: Nat,
164  h: BadCandidate(n, bound, d)
165): divisor_test(n, d) = false {
166  ⊢ divisor_test(n, d) = false from match h { Candidate(bad, within) => bad }
167}
168pub theorem candidate_bound(
169  n: Nat,
170  bound: Nat,
171  d: Nat,
172  h: BadCandidate(n, bound, d)
173): nat!(`d ≤ bound`) {
174  ⊢ nat!(`d ≤ bound`) from match h { Candidate(bad, within) => within }
175}
176theorem and_true_identity(a: Bool): Bool::and(a, true) = a {
177  ⊢ Bool::and(a, true) = a from match a { false => Refl, true => Refl }
178}
179theorem and_with_true(a: Bool, b: Bool, hb: b = true, h: Bool::and(a, b) = false): a = false {
180  calc!(```
181    $a
182    = $Bool::and(a, true) from $Eq::sym(and_true_identity(a))
183    = $Bool::and(a, b) from $Eq::sym(Eq::congr((x: Bool) => Bool::and(a, x), hb))
184    = $false from $h
185    ```)
186}
187theorem find_true(
188  n: Nat,
189  k: Nat,
190  good: divisor_test(n, Succ(k)) = true
191): find_factor(n, Succ(k)) = find_factor(n, k) {
192  Eq::congr((b: Bool) => if b { find_factor(n, k) } else { Succ(k) }, good)
193}
194theorem find_false(
195  n: Nat,
196  k: Nat,
197  bad: divisor_test(n, Succ(k)) = false
198): find_factor(n, Succ(k)) = Succ(k) {
199  Eq::congr((b: Bool) => if b { find_factor(n, k) } else { Succ(k) }, bad)
200}
201pub theorem find_factor_correct(
202  n: Nat,
203  bound: Nat,
204  bad: trial(n, bound) = false
205): BadCandidate(n, bound, find_factor(n, bound)) {
206  ⊢ BadCandidate(n, bound, find_factor(n, bound)) from match bound {
207    Zero => False::elim(true_ne_false(bad))
208    Succ(k) => Or::elim(
209      boolean_cases(divisor_test(n, Succ(k))),
210      (no: divisor_test(n, Succ(k)) = false) => Eq::subst(
211        (d: Nat) => BadCandidate(n, Succ(k), d),
212        Eq::sym(find_false(n, k, no)),
213        Candidate(no, Nat::le_refl(Succ(k)))
214      ),
215      (yes: divisor_test(n, Succ(k)) = true) => Eq::subst(
216        (d: Nat) => BadCandidate(n, Succ(k), d),
217        Eq::sym(find_true(n, k, yes)),
218        Candidate(
219          candidate_bad(
220            n,
221            k,
222            find_factor(n, k),
223            find_factor_correct(
224              n,
225              k,
226              and_with_true(trial(n, k), divisor_test(n, Succ(k)), yes, bad)
227            )
228          ),
229          Nat::le_trans(
230            find_factor(n, k),
231            k,
232            Succ(k),
233            candidate_bound(
234              n,
235              k,
236              find_factor(n, k),
237              find_factor_correct(
238                n,
239                k,
240                and_with_true(trial(n, k), divisor_test(n, Succ(k)), yes, bad)
241              )
242            ),
243            Nat::le_succ(k)
244          )
245        )
246      )
247    )
248  }
