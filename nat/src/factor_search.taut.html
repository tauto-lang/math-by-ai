10// Quotient projection used only after a nonzero factor has been established.
11pub fn quotient(n: Nat, d: Nat): Nat { Nat::dm_fst(Nat::divmod(n, d, 0, 0)) }
12pub(open) inductive FactorWitness(n: Nat, d: Nat): Prop {
13  Witness(
14    positive: nat!(`1 < d`),
15    proper: Nat::eq(d, n) = false,
16    product: nat!(`d * quotient(n, d) = n`)
17  ): FactorWitness(n, d)
18}
19pub theorem factor_product(n: Nat, d: Nat, h: FactorWitness(n, d)): nat!(`d * quotient(n, d) = n`) {
20  ⊢ nat!(`d * quotient(n, d) = n`) from match h { Witness(pos, ne, eq) => eq }
21}
22pub theorem factor_positive(n: Nat, d: Nat, h: FactorWitness(n, d)): nat!(`1 < d`) {
23  ⊢ nat!(`1 < d`) from match h { Witness(pos, ne, eq) => pos }
24}
25pub theorem factor_proper(n: Nat, d: Nat, h: FactorWitness(n, d)): Nat::eq(d, n) = false {
26  ⊢ Nat::eq(d, n) = false from match h { Witness(pos, ne, eq) => ne }
27}
28theorem test_equal(
29  n: Nat,
30  k: Nat,
31  eq: Nat::eq(Succ(Succ(k)), n) = true
32): divisor_test(n, Succ(Succ(k))) = true {
33  Eq::congr(
34    (b: Bool) => Bool::or(
35      b,
36      bool_neg(Nat::eq(Nat::rem_nz(n, Succ(Succ(k)), Nat::succ_ne_zero(Succ(k))), 0))
37    ),
38    eq
39  )
40}
41theorem test_nonzero(
42  n: Nat,
43  k: Nat,
44  eq: Nat::eq(Succ(Succ(k)), n) = false,
45  rem: Nat::eq(Nat::rem_nz(n, Succ(Succ(k)), Nat::succ_ne_zero(Succ(k))), 0) = false
46): divisor_test(n, Succ(Succ(k))) = true {
47  Eq::trans(
48    Eq::congr(
49      (b: Bool) => Bool::or(
50        b,
51        bool_neg(Nat::eq(Nat::rem_nz(n, Succ(Succ(k)), Nat::succ_ne_zero(Succ(k))), 0))
52      ),
53      eq
54    ),
55    Eq::congr(bool_neg, rem)
56  )
57}
58pub theorem bad_test_factor(n: Nat, d: Nat, bad: divisor_test(n, d) = false): FactorWitness(n, d) {
59  ⊢ FactorWitness(n, d) from match d {
60    Zero => False::elim(true_ne_false(bad))
61    Succ(k) => ⊢ FactorWitness(n, Succ(k)) from match k {
62      Zero => False::elim(true_ne_false(bad))
63      Succ(j) => Or::elim(
64        boolean_cases(Nat::eq(Succ(Succ(j)), n)),
65        (ne: Nat::eq(Succ(Succ(j)), n) = false) => Or::elim(
66          boolean_cases(Nat::eq(Nat::rem_nz(n, Succ(Succ(j)), Nat::succ_ne_zero(Succ(j))), 0)),
67          (nonzero: Nat::eq(Nat::rem_nz(n, Succ(Succ(j)), Nat::succ_ne_zero(Succ(j))), 0) = false) => False::elim(
68            true_ne_false(Eq::trans(Eq::sym(test_nonzero(n, j, ne, nonzero)), bad))
69          ),
70          (zero: Nat::eq(Nat::rem_nz(n, Succ(Succ(j)), Nat::succ_ne_zero(Succ(j))), 0) = true) => Witness(
71            Refl,
72            ne,
73            Eq::sym(Nat::mul_div_of_mod_zero(n, Succ(Succ(j)), Nat::succ_ne_zero(Succ(j)), zero))
74          )
75        ),
76        (eq: Nat::eq(Succ(Succ(j)), n) = true) => False::elim(
77          true_ne_false(Eq::trans(Eq::sym(test_equal(n, j, eq)), bad))
78        )
79      )
80    }
81  }
82}
83theorem trial_false(n: Nat, pos: nat!(`1 < n`), bad: is_prime(n) = false): trial(n, n) = false {
84  Eq::trans(Eq::sym(Eq::congr((b: Bool) => Bool::and(b, trial(n, n)), pos)), bad)
85}

98theorem quotient_positive_of(
99  n: Nat,
100  d: Nat,
101  q: Nat,
102  pos: nat!(`1 < n`),
103  product: nat!(`d * q = n`),
104  proper: Nat::eq(d, n) = false
105): nat!(`1 < q`) {
106  ⊢ nat!(`1 < q`) from match q {
107    Zero => False::elim(
108      false_ne_true(
109        Eq::trans(
110          Eq::congr((x: Nat) => Nat::lt(1, x), Eq::trans(Eq::sym(Nat::mul_zero_right(d)), product)),
111          pos
112        )
113      )
114    )
115    Succ(k) => ⊢ nat!(`1 < Succ(k)`) from match k {
116      Zero => False::elim(
117        true_ne_false(
118          Eq::trans(
119            Eq::sym(Nat::eq_refl(d)),
120            Eq::trans(
121              Eq::sym(
122                Eq::congr(
123                  (x: Nat) => Nat::eq(d, x),
124                  Eq::trans(Eq::sym(product), Nat::mul_one_right(d))
125                )
126              ),
127              proper
128            )
129          )
130        )
131      )
132      Succ(j) => Refl
133    }
134  }
135}
136pub theorem factor_quotient_positive(
137  n: Nat,
138  d: Nat,
139  pos: nat!(`1 < n`),
140  h: FactorWitness(n, d)
141): nat!(`1 < quotient(n, d)`) {
142  quotient_positive_of(n, d, quotient(n, d), pos, factor_product(n, d, h), factor_proper(n, d, h))
143}
144theorem add_positive(a: Nat, b: Nat, h: nat!(`0 < b`)): nat!(`a < a + b`) {
145  ⊢ nat!(`a < a + b`) from match a {
146    Zero => Eq::trans(Eq::congr((x: Nat) => Nat::lt(0, x), Nat::add_zero_left(b)), h)
147    Succ(k) => Eq::trans(
148      Eq::congr((x: Nat) => Nat::lt(Succ(k), x), Nat::add_succ_left(k, b)),
149      add_positive(k, b, h)
150    )
151  }
152}
153theorem mul_positive(a: Nat, b: Nat): nat!(`0 < Succ(a) * Succ(b)`) {
154  Eq::congr((x: Nat) => Nat::lt(0, x), Nat::mul_succ_left(a, Succ(b)))
155}
156theorem pair_smaller(a: Nat, b: Nat): nat!(`Succ(a) < Succ(a) * Succ(Succ(b))`) {
157  Eq::trans(
158    Eq::congr((x: Nat) => Nat::lt(Succ(a), x), Nat::mul_succ_right(Succ(a), Succ(b))),
159    Eq::trans(
160      Eq::congr((x: Nat) => Nat::lt(Succ(a), x), Nat::add_comm(nat!(`Succ(a) * Succ(b)`), Succ(a))),
161      add_positive(Succ(a), nat!(`Succ(a) * Succ(b)`), mul_positive(a, b))
162    )
163  )
164}
165theorem quotient_smaller_of(
166  n: Nat,
167  d: Nat,
168  q: Nat,
169  hd: nat!(`1 < d`),
170  hq: nat!(`1 < q`),
171  product: nat!(`d * q = n`)
172): nat!(`q < n`) {
173  ⊢ nat!(`q < n`) from match d {
174    Zero => False::elim(false_ne_true(hd))
175    Succ(k) => ⊢ nat!(`q < n`) from match k {
176      Zero => False::elim(false_ne_true(hd))
177      Succ(j) => ⊢ nat!(`q < n`) from match q {
178        Zero => False::elim(false_ne_true(hq))
179        Succ(i) => Eq::subst(
180          (x: Nat) => nat!(`Succ(i) < x`),
181          product,
182          Eq::subst(
183            (x: Nat) => nat!(`Succ(i) < x`),
184            Nat::mul_comm(Succ(i), Succ(Succ(j))),
185            pair_smaller(i, j)
186          )
187        )
188      }
189    }
190  }
191}
192pub theorem factor_quotient_smaller(
193  n: Nat,
194  d: Nat,
195  pos: nat!(`1 < n`),
196  h: FactorWitness(n, d)
197): nat!(`quotient(n, d) < n`) {
198  quotient_smaller_of(
199    n,
200    d,
201    quotient(n, d),
202    factor_positive(n, d, h),
203    factor_quotient_positive(n, d, pos, h),
204    factor_product(n, d, h)
205  )
206}
207pub theorem searched_smaller(
208  n: Nat,
209  pos: nat!(`1 < n`),
210  bad: is_prime(n) = false
211): nat!(`find_factor(n, n) < n`) {
212  Nat::lt_succ_ne(
213    find_factor(n, n),
214    n,
215    candidate_bound(n, n, find_factor(n, n), find_factor_correct(n, n, trial_false(n, pos, bad))),
216    factor_proper(n, find_factor(n, n), searched_factor(n, pos, bad))
217  )
