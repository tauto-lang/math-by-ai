8pub theorem is_prime_complete(n: Nat, hp: Prime(n)): is_prime(n) = true {
9  Or::elim(boolean_cases(is_prime(n)),
10    (bad: is_prime(n) = false) => {
11      let d := find_factor(n, n)
12      let w := searched_factor(n, prime_gt_one(n, hp), bad)
13      Or::elim(prime_divisor(n, hp, d, By(quotient(n, d), factor_product(n, d, w))),
14        (one: d = 1) => False::elim(false_ne_true(Eq::trans(
15          Eq::sym(Eq::congr((x: Nat) => Nat::lt(1, x), one)), factor_positive(n, d, w)))),
16        (equal: d = n) => False::elim(true_ne_false(Eq::trans(
17          Eq::sym(Nat::eq_refl(d)), Eq::trans(Eq::congr((x: Nat) => Nat::eq(d, x), equal), factor_proper(n, d, w))))))
18    },
19    (good: is_prime(n) = true) => good)
20}
21pub theorem is_prime_false(n: Nat, bad: is_prime(n) = false, hp: Prime(n)): False {
22  true_ne_false(Eq::trans(Eq::sym(is_prime_complete(n, hp)), bad))
23}
