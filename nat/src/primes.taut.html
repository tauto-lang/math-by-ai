10pub(open) inductive Prime(p: Nat): Prop {
11  PrimeBy(
12    positive: nat!(`1 < p`),
13    divisors: (d: Nat, h: nat_rel!(`d | p`)) -> logic!(`d = 1 ∨ d = p`)
14  ): Prime(p)
15}
16pub theorem prime_gt_one(p: Nat, hp: Prime(p)): nat!(`1 < p`) {
17  ⊢ nat!(`1 < p`) from match hp { PrimeBy(pos, all) => pos }
18}
19pub theorem prime_divisor(
20  p: Nat,
21  hp: Prime(p),
22  d: Nat,
23  h: nat_rel!(`d | p`)
24): logic!(`d = 1 ∨ d = p`) {
25  ⊢ logic!(`d = 1 ∨ d = p`) from match hp { PrimeBy(pos, all) => all(d, h) }
26}
27pub theorem prime_coprime_or_divides(
28  p: Nat,
29  a: Nat,
30  hp: Prime(p)
31): logic!(`Coprime(p, a) ∨ Divides(p, a)`) {
32  Or::elim(
33    prime_divisor(p, hp, gcd(p, a), gcd_left(p, a, gcd(p, a), gcd_correct(p, a))),
34    (h: nat!(`gcd(p, a) = 1`)) => OrInl(h),
35    (h: nat!(`gcd(p, a) = p`)) => OrInr(
36      Eq::subst((d: Nat) => nat_rel!(`d | a`), h, gcd_right(p, a, gcd(p, a), gcd_correct(p, a)))
37    )
38  )
39}
40// Euclid's prime lemma, derived from computed Bezout coefficients.
41pub theorem prime_divides_product(
42  p: Nat,
43  a: Nat,
44  b: Nat,
45  hp: Prime(p),
46  h: nat_rel!(`p | a * b`)
47): logic!(`Divides(p, a) ∨ Divides(p, b)`) {
48  Or::elim(
49    prime_coprime_or_divides(p, a, hp),
50    (cop: Coprime(p, a)) => OrInr(coprime_cancel(p, a, b, cop, h)),
51    (pa: nat_rel!(`p | a`)) => OrInl(pa)
52  )
53}
