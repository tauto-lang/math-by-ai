12pub(open) inductive AllPrime: (List(Nat)) -> Prop {
13  PrimesNil: AllPrime(Nil)
14  PrimesCons(p: Nat, xs: List(Nat), hp: Prime(p), tail: AllPrime(xs)): AllPrime(Cons(p, xs))
15}
16pub(open) inductive Factorization(n: Nat, xs: List(Nat)): Prop {
17  Factored(primes: AllPrime(xs), product: nat!(`prod(xs) = n`)): Factorization(n, xs)
18}
19pub theorem factorization_primes(n: Nat, xs: List(Nat), h: Factorization(n, xs)): AllPrime(xs) {
20  ⊢ AllPrime(xs) from match h { Factored(ps, product) => ps }
21}
22pub theorem factorization_product(
23  n: Nat,
24  xs: List(Nat),
25  h: Factorization(n, xs)
26): nat!(`prod(xs) = n`) {
27  ⊢ nat!(`prod(xs) = n`) from match h { Factored(ps, product) => product }
28}
29pub theorem primes_append(
30  xs: List(Nat),
31  ys: List(Nat),
32  hx: AllPrime(xs),
33  hy: AllPrime(ys)
34): AllPrime(List::append(xs, ys)) {
35  ⊢ AllPrime(List::append(xs, ys)) from match hx {
36    PrimesNil => hy
37    PrimesCons(p, rest, hp, hr) => PrimesCons(
38      p,
39      List::append(rest, ys),
40      hp,
41      primes_append(rest, ys, hr, hy)
42    )
43  }
44}
45pub theorem factorization_append(
46  a: Nat,
47  b: Nat,
48  xs: List(Nat),
49  ys: List(Nat),
50  hx: Factorization(a, xs),
51  hy: Factorization(b, ys)
52): Factorization(nat!(`a * b`), List::append(xs, ys)) {
53  Factored(
54    primes_append(xs, ys, factorization_primes(a, xs, hx), factorization_primes(b, ys, hy)),
55    calc!(```
56      $nat!(`prod(List::append(xs, ys))`)
57      = $nat!(`prod(xs) * prod(ys)`) from $prod_append(xs, ys)
58      = $nat!(`a * prod(ys)`) from $Eq::congr(
59        (x: Nat) => nat!(`x * prod(ys)`),
60        factorization_product(a, xs, hx)
61      )
62      = $nat!(`a * b`) from $Eq::congr((y: Nat) => nat!(`a * y`), factorization_product(b, ys, hy))
63      ```)
64  )
65}

67fn factor_loop(steps: Nat, n: Nat): List(Nat) decreasing_by steps {
68  ⊢ List(Nat) from match n {
69    Zero => Nil
70    Succ(k) => ⊢ List(Nat) from match k {
71      Zero => Nil
72      Succ(j) => ⊢ List(Nat) from match steps {
73        Zero => Nil
74        Succ(f) => if is_prime(n) { Cons(n, Nil) } else {
75          let d := find_factor(n, n)
76          List::append(factor_loop(f, d), factor_loop(f, quotient(n, d)))
77        }
78      }
79    }
80  }
81}
82pub fn factorize(n: Nat): List(Nat) { factor_loop(n, n) }

84theorem factor_loop_prime(
85  f: Nat,
86  j: Nat,
87  h: is_prime(Succ(Succ(j))) = true
88): factor_loop(Succ(f), Succ(Succ(j))) = Cons(Succ(Succ(j)), Nil) {
89  Eq::congr((b: Bool) => if b { Cons(Succ(Succ(j)), Nil) } else {
90    List::append(factor_loop(f, find_factor(Succ(Succ(j)), Succ(Succ(j)))),
91      factor_loop(f, quotient(Succ(Succ(j)), find_factor(Succ(Succ(j)), Succ(Succ(j))))))
92  }, h)
93}
94theorem factor_loop_composite(
95  f: Nat,
96  j: Nat,
97  h: is_prime(Succ(Succ(j))) = false
98): factor_loop(
99  Succ(f),
100  Succ(Succ(j))
101) = List::append(
102  factor_loop(f, find_factor(Succ(Succ(j)), Succ(Succ(j)))),
103  factor_loop(f, quotient(Succ(Succ(j)), find_factor(Succ(Succ(j)), Succ(Succ(j)))))
104) {
105  Eq::congr((b: Bool) => if b { Cons(Succ(Succ(j)), Nil) } else {
106    List::append(factor_loop(f, find_factor(Succ(Succ(j)), Succ(Succ(j)))),
107      factor_loop(f, quotient(Succ(Succ(j)), find_factor(Succ(Succ(j)), Succ(Succ(j))))))
108  }, h)
109}
110pub theorem descending_bound(
111  m: Nat,
112  n: Nat,
113  f: Nat,
114  lt: nat!(`m < n`),
115  bound: nat!(`n ≤ Succ(f)`)
116): nat!(`m ≤ f`) {
117  ⊢ nat!(`m ≤ f`) from match n {
118    Zero => False::elim(false_ne_true(Eq::trans(Eq::sym(Nat::lt_zero_false(m)), lt)))
119    Succ(k) => Nat::le_trans(m, k, f, lt, bound)
120  }
121}

123theorem factor_loop_correct(
124  steps: Nat,
125  n: Nat,
126  positive: nat!(`0 < n`),
127  bound: nat!(`n ≤ steps`)
128): Factorization(n, factor_loop(steps, n)) {
129  ⊢ Factorization(n, factor_loop(steps, n)) from match steps {
130    Zero => ⊢ Factorization(n, factor_loop(Zero, n)) from match n {
131      Zero => False::elim(false_ne_true(positive))
132      Succ(k) => False::elim(false_ne_true(Eq::trans(Eq::sym(Nat::lt_zero_false(k)), bound)))
133    }
134    Succ(f) => ⊢ Factorization(n, factor_loop(Succ(f), n)) from match n {
135      Zero => False::elim(false_ne_true(positive))
136      Succ(k) => ⊢ Factorization(Succ(k), factor_loop(Succ(f), Succ(k))) from match k {
137        Zero => Factored(PrimesNil, Refl)
138        Succ(j) => Or::elim(boolean_cases(is_prime(Succ(Succ(j)))),
139          (bad: is_prime(Succ(Succ(j))) = false) => {
140            let n := Succ(Succ(j))
141            let d := find_factor(n, n)
142            let w := searched_factor(n, Refl, bad)
143            Eq::subst((xs: List(Nat)) => Factorization(n, xs), Eq::sym(factor_loop_composite(f, j, bad)),
144              Eq::subst((x: Nat) => Factorization(x, List::append(factor_loop(f, d), factor_loop(f, quotient(n, d)))),
145                factor_product(n, d, w),
146                factorization_append(d, quotient(n, d), factor_loop(f, d), factor_loop(f, quotient(n, d)),
147                  factor_loop_correct(f, d, Nat::lt_of_succ_lt(0, d, factor_positive(n, d, w)),
148                    descending_bound(d, n, f, searched_smaller(n, Refl, bad), bound)),
149                  factor_loop_correct(f, quotient(n, d), Nat::lt_of_succ_lt(0, quotient(n, d), factor_quotient_positive(n, d, Refl, w)),
150                    descending_bound(quotient(n, d), n, f, factor_quotient_smaller(n, d, Refl, w), bound)))))
151          },
152          (good: is_prime(Succ(Succ(j))) = true) => Eq::subst((xs: List(Nat)) => Factorization(Succ(Succ(j)), xs),
153            Eq::sym(factor_loop_prime(f, j, good)),
154            Factored(PrimesCons(Succ(Succ(j)), Nil, is_prime_sound(Succ(Succ(j)), good), PrimesNil), Nat::mul_one_right(Succ(Succ(j))))))
155      }
156    }
157  }
158}
159pub theorem factorize_correct(n: Nat, positive: nat!(`0 < n`)): Factorization(n, factorize(n)) {
160  factor_loop_correct(n, n, positive, Nat::le_refl(n))
