17pub fn factorial(n: Nat): Nat {
18  ⊢ Nat from match n { Zero => 1
19    Succ(k) => nat!(`Succ(k) * factorial(k)`) }
20}
21theorem positive_mul(a: Nat, b: Nat, ha: nat!(`0 < a`), hb: nat!(`0 < b`)): nat!(`0 < a * b`) {
22  ⊢ nat!(`0 < a * b`) from match a {
23    Zero => False::elim(false_ne_true(ha))
24    Succ(k) => ⊢ nat!(`0 < Succ(k) * b`) from match b {
25      Zero => False::elim(false_ne_true(hb))
26      Succ(j) => Eq::congr((x: Nat) => Nat::lt(0, x), Nat::mul_succ_left(k, Succ(j)))
27    }
28  }
29}
30pub theorem factorial_positive(n: Nat): nat!(`0 < factorial(n)`) {
31  ⊢ nat!(`0 < factorial(n)`) from match n {
32    Zero => Refl
33    Succ(k) => positive_mul(Succ(k), factorial(k), Refl, factorial_positive(k))
34  }
35}
36pub theorem divides_factorial(
37  d: Nat,
38  n: Nat,
39  positive: nat!(`0 < d`),
40  bound: nat!(`d ≤ n`)
41): nat_rel!(`d | factorial(n)`) {
42  ⊢ nat_rel!(`d | factorial(n)`) from match n {
43    Zero => ⊢ nat_rel!(`d | factorial(0)`) from match d {
44      Zero => False::elim(false_ne_true(positive))
45      Succ(k) => False::elim(false_ne_true(Eq::trans(Eq::sym(Nat::lt_zero_false(k)), bound)))
46    }
47    Succ(k) => Or::elim(
48      boolean_cases(Nat::eq(d, Succ(k))),
49      (ne: Nat::eq(d, Succ(k)) = false) => divides_mul_left(
50        d,
51        factorial(k),
52        Succ(k),
53        divides_factorial(d, k, positive, Nat::lt_succ_ne(d, Succ(k), bound, ne))
54      ),
55      (eq: Nat::eq(d, Succ(k)) = true) => Eq::subst(
56        (x: Nat) => nat_rel!(`x | factorial(Succ(k))`),
57        Eq::sym(Nat::eq_sound(d, Succ(k), eq)),
58        By(factorial(k), Refl)
59      )
60    )
61  }
62}
63theorem le_false_lt(a: Nat, b: Nat, h: Nat::le(a, b) = false): nat!(`b < a`) {
64  ⊢ nat!(`b < a`) from match a {
65    Zero => False::elim(true_ne_false(Eq::trans(Eq::sym(Nat::le_zero(b)), h)))
66    Succ(k) => ⊢ nat!(`b < Succ(k)`) from match b {
67      Zero => Refl
68      Succ(j) => le_false_lt(k, j, h)
69    }
70  }
71}
72theorem prime_factor_above(
73  n: Nat,
74  p: Nat,
75  hp: Prime(p),
76  divides: nat_rel!(`p | Succ(factorial(n))`)
77): nat!(`n < p`) {
78  proof!(```
79      Consider whether $nat_bool!(`p ≤ n`) holds.
80        Case above: not($nat_bool!(`p ≤ n`)). By le_false_lt(p, n, above).
81        [below] Case below: $nat_bool!(`p ≤ n`). This contradicts prime_not_divides_one(p, hp).
82      %%
83      [below] Eq::subst($((x: Nat) => nat_rel!(`p | x`)), sub_sum_left(factorial(n), 1), divides_subtract(p, Succ(factorial(n)), factorial(n), divides, divides_factorial(p, n, Nat::lt_of_succ_lt(0, p, prime_gt_one(p, hp)), below)))
84    ```)
85}
86pub fn first_or_two(xs: List(Nat)): Nat {
87  ⊢ Nat from match xs { Nil => 2
88    Cons(x, rest) => x }
89}
90pub fn prime_above(n: Nat): Nat { first_or_two(factorize(Succ(factorial(n)))) }
91pub(open) inductive AbovePrime(n: Nat, p: Nat): Prop {
92  Above(prime: Prime(p), greater: nat!(`n < p`)): AbovePrime(n, p)
93}
94theorem head_prime(p: Nat, rest: List(Nat), h: AllPrime(Cons(p, rest))): Prime(p) {
95  ⊢ Prime(p) from match h { PrimesCons(x, xs, hp, hr) => hp }
96}
97theorem factor_head_above(
98  n: Nat,
99  xs: List(Nat),
100  h: Factorization(Succ(factorial(n)), xs)
101): AbovePrime(n, first_or_two(xs)) {
102  ⊢ AbovePrime(n, first_or_two(xs)) from match xs {
103    Nil => False::elim(
104      false_ne_true(
105        Eq::trans(
106          Eq::congr(
107            (x: Nat) => Nat::lt(0, x),
108            Eq::congr(Nat::pred, factorization_product(Succ(factorial(n)), Nil, h))
109          ),
110          factorial_positive(n)
111        )
112      )
113    )
114    Cons(p, rest) => Above(
115      head_prime(p, rest, factorization_primes(Succ(factorial(n)), Cons(p, rest), h)),
116      prime_factor_above(
117        n,
118        p,
119        head_prime(p, rest, factorization_primes(Succ(factorial(n)), Cons(p, rest), h)),
120        By(prod(rest), factorization_product(Succ(factorial(n)), Cons(p, rest), h))
121      )
122    )
123  }
124}
125// A constructive infinitude theorem: prime_above computes a witness for every bound.
126pub theorem prime_above_correct(n: Nat): AbovePrime(n, prime_above(n)) {
127  factor_head_above(n, factorize(Succ(factorial(n))), factorize_correct(Succ(factorial(n)), Refl))
