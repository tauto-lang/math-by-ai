13// Coefficients are signed pairs: (pa - na), (pb - nb).
14// All arithmetic inside the extended algorithm is exact natural arithmetic.
15pub(open) inductive Coefficients: Type {
16  Coeff(pa: Nat, na: Nat, pb: Nat, nb: Nat): Coefficients
17}
18pub fn balance(a: Nat, b: Nat, g: Nat, c: Coefficients): Prop {
19  ⊢ Prop from match c {
20    Coeff(pa, na, pb, nb) => nat!(`g + (a * na + b * nb) = a * pa + b * pb`)
21  }
22}
23pub fn lift_coeff(q: Nat, c: Coefficients): Coefficients {
24  ⊢ Coefficients from match c {
25    Coeff(pa, na, pb, nb) => Coeff(pb, nb, nat!(`pa + q * nb`), nat!(`na + q * pb`))
26  }
27}

29theorem lift_normalize_0(
30  b: Nat,
31  r: Nat,
32  q: Nat,
33  g: Nat,
34  pa: Nat,
35  na: Nat,
36  pb: Nat,
37  nb: Nat
38): nat!(`g + ((b * q + r) * nb + b * (na + q * pb)) = g + (b * na + r * nb) + (b * q * nb + b * q * pb)`) {
39  by nat_norm!(): nat!(`g + ((b * q + r) * nb + b * (na + q * pb)) = g + (b * na + r * nb) + (b * q * nb + b * q * pb)`) {}
40}
41theorem lift_normalize_1(
42  b: Nat,
43  r: Nat,
44  q: Nat,
45  g: Nat,
46  pa: Nat,
47  na: Nat,
48  pb: Nat,
49  nb: Nat
50): nat!(`b * pa + r * pb + (b * q * nb + b * q * pb) = (b * q + r) * pb + b * (pa + q * nb)`) {
51  by nat_norm!(): nat!(`b * pa + r * pb + (b * q * nb + b * q * pb) = (b * q + r) * pb + b * (pa + q * nb)`) {}
52}
53pub theorem lift_balance(
54  b: Nat,
55  r: Nat,
56  q: Nat,
57  g: Nat,
58  c: Coefficients,
59  h: balance(b, r, g, c)
60): balance(nat!(`b * q + r`), b, g, lift_coeff(q, c)) {
61  ⊢ balance(nat!(`b * q + r`), b, g, lift_coeff(q, c)) from match c {
62    Coeff(pa, na, pb, nb) => calc!(```
63      $nat!(`g + ((b * q + r) * nb + b * (na + q * pb))`)
64      = $nat!(`g + (b * na + r * nb) + (b * q * nb + b * q * pb)`) from $lift_normalize_0(
65        b,
66        r,
67        q,
68        g,
69        pa,
70        na,
71        pb,
72        nb
73      )
74      = $nat!(`b * pa + r * pb + (b * q * nb + b * q * pb)`) from $Eq::congr(
75        (x: Nat) => nat!(`x + (b * q * nb + b * q * pb)`),
76        h
77      )
78      = $nat!(`(b * q + r) * pb + b * (pa + q * nb)`) from $lift_normalize_1(
79        b,
80        r,
81        q,
82        g,
83        pa,
84        na,
85        pb,
86        nb
87      )
88      ```)
89  }
90}

92fn coefficient_loop(steps: Nat, a: Nat, b: Nat): Coefficients decreasing_by steps {
93  ⊢ Coefficients from match b {
94    Zero => Coeff(1, 0, 0, 0)
95    Succ(k) => ⊢ Coefficients from match steps {
96      Zero => Coeff(1, 0, 0, 0)
97      Succ(f) => lift_coeff(
98        Nat::div_nz(a, Succ(k), Nat::succ_ne_zero(k)),
99        coefficient_loop(f, Succ(k), Nat::rem_nz(a, Succ(k), Nat::succ_ne_zero(k)))
100      )
101    }
102  }
103}
104pub fn bezout_coefficients(a: Nat, b: Nat): Coefficients { coefficient_loop(b, a, b) }

106theorem base_balance(a: Nat): balance(a, 0, a, Coeff(1, 0, 0, 0)) {
107  by nat_norm!(): nat!(`a + (a * 0 + 0 * 0) = a * 1 + 0 * 0`) {}
108}
109theorem coefficient_correct(
110  steps: Nat,
111  a: Nat,
112  b: Nat,
113  bound: nat!(`b ≤ steps`)
114): balance(a, b, euclid_loop(steps, a, b), coefficient_loop(steps, a, b)) {
115  ⊢ balance(a, b, euclid_loop(steps, a, b), coefficient_loop(steps, a, b)) from match steps {
116    Zero => ⊢ balance(a, b, euclid_loop(Zero, a, b), coefficient_loop(Zero, a, b)) from match b {
117      Zero => base_balance(a)
118      Succ(k) => False::elim(false_ne_true(Eq::trans(Eq::sym(Nat::lt_zero_false(k)), bound)))
119    }
120    Succ(f) => ⊢ balance(a, b, euclid_loop(Succ(f), a, b), coefficient_loop(Succ(f), a, b)) from match b {
121      Zero => base_balance(a)
122      Succ(k) => Eq::subst(
123        (x: Nat) => balance(
124          x,
125          Succ(k),
126          euclid_loop(f, Succ(k), Nat::rem_nz(a, Succ(k), Nat::succ_ne_zero(k))),
127          coefficient_loop(Succ(f), a, Succ(k))
128        ),
129        Nat::div_mod_eq(a, Succ(k), Nat::succ_ne_zero(k)),
130        lift_balance(
131          Succ(k),
132          Nat::rem_nz(a, Succ(k), Nat::succ_ne_zero(k)),
133          Nat::div_nz(a, Succ(k), Nat::succ_ne_zero(k)),
134          euclid_loop(f, Succ(k), Nat::rem_nz(a, Succ(k), Nat::succ_ne_zero(k))),
135          coefficient_loop(f, Succ(k), Nat::rem_nz(a, Succ(k), Nat::succ_ne_zero(k))),
136          coefficient_correct(
137            f,
138            Succ(k),
139            Nat::rem_nz(a, Succ(k), Nat::succ_ne_zero(k)),
140            remainder_bound(a, k, f, bound)
141          )
142        )
143      )
144    }
145  }
146}
147pub theorem bezout_balance(a: Nat, b: Nat): balance(a, b, gcd(a, b), bezout_coefficients(a, b)) {
148  coefficient_correct(b, a, b, Nat::le_refl(b))
149}

151pub fn coefficient_a(c: Coefficients): Int {
152  ⊢ Int from match c { Coeff(pa, na, pb, nb) => int!(`diff(pa, na)`) }
153}
154pub fn coefficient_b(c: Coefficients): Int {
155  ⊢ Int from match c { Coeff(pa, na, pb, nb) => int!(`diff(pb, nb)`) }
156}
157theorem zero_term(a: Nat, b: Nat): nat!(`a + 0 * b = a`) {
158  Eq::congr((x: Nat) => nat!(`a + x`), Nat::mul_zero_left(b))
159}
160theorem mul_difference(a: Nat, p: Nat, n: Nat): int!(`↑a * diff(p, n) = diff(a * p, a * n)`) {
161  calc!(```
162    $int!(`↑a * diff(p, n)`)
163    = $int!(`diff(a, 0) * diff(p, n)`) from $Eq::congr(
164      (x: Int) => int!(`x * diff(p, n)`),
165      Eq::sym(Int::sub_nat_zero(a))
166    )
167    = $int!(`diff(a * p + 0 * n, a * n + 0 * p)`) from $Int::mul_sub(a, 0, p, n)
168    = $int!(`diff(a * p, a * n)`) from $Int::sub_congr(
169      nat!(`a * p + 0 * n`),
170      nat!(`a * n + 0 * p`),
171      nat!(`a * p`),
172      nat!(`a * n`),
173      zero_term(nat!(`a * p`), n),
174      zero_term(nat!(`a * n`), p)
175    )
176    ```)
177}
178pub theorem signed_balance(
179  a: Nat,
180  b: Nat,
181  g: Nat,
182  c: Coefficients,
183  h: balance(a, b, g, c)
184): int!(`↑a * coefficient_a(c) + ↑b * coefficient_b(c) = ↑g`) {
185  ⊢ int!(`↑a * coefficient_a(c) + ↑b * coefficient_b(c) = ↑g`) from match c {
186    Coeff(pa, na, pb, nb) => calc!(```
187      $int!(`↑a * diff(pa, na) + ↑b * diff(pb, nb)`)
188      = $int!(`diff(a * pa, a * na) + ↑b * diff(pb, nb)`) from $Eq::congr(
189        (x: Int) => int!(`x + ↑b * diff(pb, nb)`),
190        mul_difference(a, pa, na)
191      )
192      = $int!(`diff(a * pa, a * na) + diff(b * pb, b * nb)`) from $Eq::congr(
193        (x: Int) => int!(`diff(a * pa, a * na) + x`),
194        mul_difference(b, pb, nb)
195      )
196      = $int!(`diff(a * pa + b * pb, a * na + b * nb)`) from $Int::add_sub(
197        nat!(`a * pa`),
198        nat!(`a * na`),
199        nat!(`b * pb`),
200        nat!(`b * nb`)
201      )
202      = $int!(`diff(g, 0)`) from $Int::sub_eq(
203        nat!(`a * pa + b * pb`),
204        nat!(`a * na + b * nb`),
205        g,
206        0,
207        Eq::sym(h)
208      )
209      = $int!(`↑g`) from $Int::sub_nat_zero(g)
210      ```)
211  }
212}
213pub theorem bezout_identity(
214  a: Nat,
215  b: Nat
216): int!(`↑a * coefficient_a(bezout_coefficients(a, b)) + ↑b * coefficient_b(bezout_coefficients(a, b)) = ↑gcd(a, b)`) {
217  signed_balance(a, b, gcd(a, b), bezout_coefficients(a, b), bezout_balance(a, b))
