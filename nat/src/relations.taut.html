42pub fn relation_arguments(
43  first: nat_relations::Sum,
44  rest: List(nat_relations::Arguments_Rep1)
45): List(SynExpr) ! {macro_error, Div} {
46  Cons(relation_sum(first), relation_more_arguments(rest))
47}
48pub fn relation_more_arguments(
49  rest: List(nat_relations::Arguments_Rep1)
50): List(SynExpr) ! {macro_error, Div} {
51  match rest { Nil => Nil
52    Cons(r, more) => Cons(relation_sum(r.sum), relation_more_arguments(more)) }
53}
54pub fn relation_atom(t: nat_relations::Atom): SynExpr ! {macro_error, Div} {
55  match t {
56    Num(tok) => ExNat(tok.text, tok.span)
57    Reference(r) => match r.arguments {
58      None => relation_name(r.name)
59      Some(args) => ExApp(relation_name(r.name), relation_arguments(args.sum, args.rep), r.span)
60    }
61    Paren(p) => relation_sum(p.sum)
62    Embed(e) => e
63  }
64}
65pub fn relation_factors(
66  acc: SynExpr,
67  rest: List(nat_relations::Prod_Rep1)
68): SynExpr ! {macro_error, Div} {
69  match rest { Nil => acc
70    Cons(r, more) => {
71      let t := relation_atom(r.atom)
72      relation_factors(quote { Nat::mul($acc, $t) }, more)
73    }
74  }
75}
76pub fn relation_product(p: nat_relations::Prod): SynExpr ! {macro_error, Div} {
77  relation_factors(relation_atom(p.atom), p.rep)
78}
79pub fn relation_summands(
80  acc: SynExpr,
81  rest: List(nat_relations::Sum_Rep1)
82): SynExpr ! {macro_error, Div} {
83  match rest { Nil => acc
84    Cons(r, more) => {
85      let t := relation_product(r.prod)
86      let next := match r.add_op {
87        Plus(tok) => quote { Nat::add($acc, $t) }
88        Minus(tok) => quote { Nat::sub($acc, $t) }
89      }
90      relation_summands(next, more)
91    }
92  }
93}
94pub fn relation_sum(s: nat_relations::Sum): SynExpr ! {macro_error, Div} {
95  relation_summands(relation_product(s.prod), s.rep)
96}
97pub fn relation_expression(e: nat_relations::Expr): SynExpr ! {macro_error, Div} {
98  let l := relation_sum(e.sum)
99  let b := relation_sum(e.sum_2)
100  match e.relation {
101    EqOp(tok) => quote { Eq(Nat, $l, $b) }
102    LeOp(tok) => quote { nat_below($l, $b) }
103    LtOp(tok) => quote { Eq(Bool, Nat::lt($l, $b), true) }
104    DvdOp(tok) => quote { Divides($l, $b) }
105  }
106}
107pub macro fn nat_rel(
108  e: nat_relations::Expr
109): SynExpr ! {macro_error, Div} { relation_expression(e) }
