8pub theorem additive_monoid(): Monoid(Nat, Nat::add, 0) {
9  Monoid::Mk(
10    (a: Nat, b: Nat, c: Nat) => Nat::add_assoc(a, b, c),
11    (a: Nat) => Nat::add_zero_left(a),
12    (a: Nat) => Refl
13  )
14}
15
16pub theorem multiplicative_monoid(): Monoid(Nat, Nat::mul, 1) {
17  Monoid::Mk(
18    (a: Nat, b: Nat, c: Nat) => Nat::mul_assoc(a, b, c),
19    (a: Nat) => Nat::mul_one_left(a),
20    (a: Nat) => Nat::mul_one_right(a)
21  )
22}

24pub fn sum(xs: List(Nat)): Nat { product(Nat::add, 0, xs) }
25pub fn prod(xs: List(Nat)): Nat { product(Nat::mul, 1, xs) }

27pub theorem sum_append(
28  xs: List(Nat),
29  ys: List(Nat)
30): nat!(`sum(List::append(xs, ys)) = sum(xs) + sum(ys)`) {
31  product_append(Nat::add, 0, additive_monoid(), xs, ys)

40pub theorem additive_comm_monoid(): CommMonoid(Nat, Nat::add, 0) {
41  CommMonoid::Mk(additive_monoid(), (a: Nat, b: Nat) => Nat::add_comm(a, b))
42}
43pub theorem multiplicative_comm_monoid(): CommMonoid(Nat, Nat::mul, 1) {
44  CommMonoid::Mk(multiplicative_monoid(), (a: Nat, b: Nat) => Nat::mul_comm(a, b))
45}
46pub theorem natural_semiring(): Semiring(Nat, Nat::add, 0, Nat::mul, 1) {
47  Semiring::Mk(
48    additive_comm_monoid(),
49    multiplicative_monoid(),
50    (a: Nat, b: Nat, c: Nat) => Nat::mul_add_distrib(a, b, c),
51    (a: Nat, b: Nat, c: Nat) => Nat::mul_add_distrib_right(a, b, c),
52    (a: Nat) => Nat::mul_zero_left(a),
53    (a: Nat) => Nat::mul_zero_right(a)
54  )
55}
