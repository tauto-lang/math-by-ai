12pub(open) inductive IsGcd(a: Nat, b: Nat, g: Nat): Prop {
13  Greatest(
14    left: nat_rel!(`g | a`),
15    right: nat_rel!(`g | b`),
16    greatest: (d: Nat, da: nat_rel!(`d | a`), db: nat_rel!(`d | b`)) -> nat_rel!(`d | g`)
17  ): IsGcd(a, b, g)
18}
19pub theorem gcd_left(a: Nat, b: Nat, g: Nat, h: IsGcd(a, b, g)): nat_rel!(`g | a`) {
20  ⊢ nat_rel!(`g | a`) from match h { Greatest(l, r, all) => l }
21}
22pub theorem gcd_right(a: Nat, b: Nat, g: Nat, h: IsGcd(a, b, g)): nat_rel!(`g | b`) {
23  ⊢ nat_rel!(`g | b`) from match h { Greatest(l, r, all) => r }
24}
25pub theorem gcd_greatest(
26  a: Nat,
27  b: Nat,
28  g: Nat,
29  h: IsGcd(a, b, g),
30  d: Nat,
31  da: nat_rel!(`d | a`),
32  db: nat_rel!(`d | b`)
33): nat_rel!(`d | g`) {
34  ⊢ nat_rel!(`d | g`) from match h { Greatest(l, r, all) => all(d, da, db) }
35}
36pub theorem remainder_sub(
37  a: Nat,
38  b: Nat,
39  nz: logic!(`¬(b = 0)`)
40): nat!(`Nat::rem_nz(a, b, nz) = a - b * Nat::div_nz(a, b, nz)`) {
41  calc!(```
42    $nat!(`Nat::rem_nz(a, b, nz)`)
43    = $nat!(
44      `(b * Nat::div_nz(a, b, nz) + Nat::rem_nz(a, b, nz)) - b * Nat::div_nz(a, b, nz)`
45    ) from $Eq::sym(sub_sum_left(nat!(`b * Nat::div_nz(a, b, nz)`), Nat::rem_nz(a, b, nz)))
46    = $nat!(`a - b * Nat::div_nz(a, b, nz)`) from $Eq::congr(
47      (x: Nat) => nat!(`x - b * Nat::div_nz(a, b, nz)`),
48      Nat::div_mod_eq(a, b, nz)
49    )
50    ```)
51}
52pub theorem divides_remainder(
53  d: Nat,
54  a: Nat,
55  b: Nat,
56  nz: logic!(`¬(b = 0)`),
57  da: nat_rel!(`d | a`),
58  db: nat_rel!(`d | b`)
59): nat_rel!(`d | Nat::rem_nz(a, b, nz)`) {
60  Eq::subst(
61    (x: Nat) => nat_rel!(`d | x`),
62    Eq::sym(remainder_sub(a, b, nz)),
63    divides_subtract(
64      d,
65      a,
66      nat!(`b * Nat::div_nz(a, b, nz)`),
67      da,
68      divides_mul_right(d, b, Nat::div_nz(a, b, nz), db)
69    )
70  )
71}
72pub theorem gcd_step(
73  a: Nat,
74  b: Nat,
75  nz: logic!(`¬(b = 0)`),
76  g: Nat,
77  h: IsGcd(b, Nat::rem_nz(a, b, nz), g)
78): IsGcd(a, b, g) {
79  Greatest(
80    Eq::subst(
81      (x: Nat) => nat_rel!(`g | x`),
82      Nat::div_mod_eq(a, b, nz),
83      divides_add(
84        g,
85        nat!(`b * Nat::div_nz(a, b, nz)`),
86        Nat::rem_nz(a, b, nz),
87        divides_mul_right(g, b, Nat::div_nz(a, b, nz), gcd_left(b, Nat::rem_nz(a, b, nz), g, h)),
88        gcd_right(b, Nat::rem_nz(a, b, nz), g, h)
89      )
90    ),
91    gcd_left(b, Nat::rem_nz(a, b, nz), g, h),
92    (d: Nat, da: nat_rel!(`d | a`), db: nat_rel!(`d | b`)) => gcd_greatest(
93      b,
94      Nat::rem_nz(a, b, nz),
95      g,
96      h,
97      d,
98      db,
99      divides_remainder(d, a, b, nz, da, db)
100    )
101  )
102}

104// The public function supplies b steps. Correctness proves these always suffice.
105pub fn euclid_loop(steps: Nat, a: Nat, b: Nat): Nat decreasing_by steps {
106  ⊢ Nat from match b {
107    Zero => a
108    Succ(k) => ⊢ Nat from match steps {
109      Zero => a
110      Succ(f) => euclid_loop(f, Succ(k), Nat::rem_nz(a, Succ(k), Nat::succ_ne_zero(k)))
111    }
112  }
113}
114pub fn gcd(a: Nat, b: Nat): Nat { euclid_loop(b, a, b) }

116pub theorem remainder_bound(
117  a: Nat,
118  b: Nat,
119  f: Nat,
120  h: nat!(`Succ(b) ≤ Succ(f)`)
121): nat!(`Nat::rem_nz(a, Succ(b), Nat::succ_ne_zero(b)) ≤ f`) {
122  Nat::le_trans(Nat::rem_nz(a, Succ(b), Nat::succ_ne_zero(b)), b, f, Nat::mod_succ_lt(a, b), h)
123}
124theorem loop_correct(
125  steps: Nat,
126  a: Nat,
127  b: Nat,
128  bound: nat!(`b ≤ steps`)
129): IsGcd(a, b, euclid_loop(steps, a, b)) {
130  ⊢ IsGcd(a, b, euclid_loop(steps, a, b)) from match steps {
131    Zero => ⊢ IsGcd(a, b, euclid_loop(Zero, a, b)) from match b {
132      Zero => Greatest(
133        divides_refl(a),
134        divides_zero(a),
135        (d: Nat, da: nat_rel!(`d | a`), db: nat_rel!(`d | 0`)) => da
136      )
137      Succ(k) => False::elim(false_ne_true(Eq::trans(Eq::sym(Nat::lt_zero_false(k)), bound)))
138    }
139    Succ(f) => ⊢ IsGcd(a, b, euclid_loop(Succ(f), a, b)) from match b {
140      Zero => Greatest(
141        divides_refl(a),
142        divides_zero(a),
143        (d: Nat, da: nat_rel!(`d | a`), db: nat_rel!(`d | 0`)) => da
144      )
145      Succ(k) => gcd_step(
146        a,
147        Succ(k),
148        Nat::succ_ne_zero(k),
149        euclid_loop(f, Succ(k), Nat::rem_nz(a, Succ(k), Nat::succ_ne_zero(k))),
150        loop_correct(
151          f,
152          Succ(k),
153          Nat::rem_nz(a, Succ(k), Nat::succ_ne_zero(k)),
154          remainder_bound(a, k, f, bound)
155        )
156      )
157    }
158  }
159}
160pub theorem gcd_correct(a: Nat, b: Nat): IsGcd(a, b, gcd(a, b)) {
161  loop_correct(b, a, b, Nat::le_refl(b))
162}

164pub theorem gcd_unique(
165  a: Nat,
166  b: Nat,
167  g: Nat,
168  h: Nat,
169  hg: IsGcd(a, b, g),
170  hh: IsGcd(a, b, h)
171): nat!(`g = h`) {
172  divides_antisym(
173    g,
174    h,
175    gcd_greatest(a, b, h, hh, g, gcd_left(a, b, g, hg), gcd_right(a, b, g, hg)),
176    gcd_greatest(a, b, g, hg, h, gcd_left(a, b, h, hh), gcd_right(a, b, h, hh))
177  )
178}
179pub theorem gcd_comm(a: Nat, b: Nat): nat!(`gcd(a, b) = gcd(b, a)`) {
180  let h := gcd_correct(b, a)
181  gcd_unique(
182    a,
183    b,
184    gcd(a, b),
185    gcd(b, a),
186    gcd_correct(a, b),
187    Greatest(
188      gcd_right(b, a, gcd(b, a), h),
189      gcd_left(b, a, gcd(b, a), h),
190      (d: Nat, da: nat_rel!(`d | a`), db: nat_rel!(`d | b`)) => gcd_greatest(
191        b,
192        a,
193        gcd(b, a),
194        h,
195        d,
196        db,
197        da
198      )
199    )
200  )
201}
202pub theorem gcd_self(a: Nat): nat!(`gcd(a, a) = a`) {
203  gcd_unique(
204    a,
205    a,
206    gcd(a, a),
207    a,
208    gcd_correct(a, a),
209    Greatest(
210      divides_refl(a),
211      divides_refl(a),
212      (d: Nat, da: nat_rel!(`d | a`), db: nat_rel!(`d | a`)) => da
213    )
214  )
215}
216pub theorem gcd_zero_right(a: Nat): nat!(`gcd(a, 0) = a`) { Refl }
217pub theorem gcd_zero_left(a: Nat): nat!(`gcd(0, a) = a`) {
218  calc!(```
219    $nat!(`gcd(0, a)`)
220    = $nat!(`gcd(a, 0)`) from $gcd_comm(0, a)
221    = $a from $gcd_zero_right(a)
222    ```)
223}
224pub theorem gcd_euclid(
225  a: Nat,
226  b: Nat,
227  nz: logic!(`¬(b = 0)`)
228): nat!(`gcd(a, b) = gcd(b, Nat::rem_nz(a, b, nz))`) {
229  gcd_unique(
230    a,
231    b,
232    gcd(a, b),
233    gcd(b, Nat::rem_nz(a, b, nz)),
234    gcd_correct(a, b),
235    gcd_step(a, b, nz, gcd(b, Nat::rem_nz(a, b, nz)), gcd_correct(b, Nat::rem_nz(a, b, nz)))
236  )
