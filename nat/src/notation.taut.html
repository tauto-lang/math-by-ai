39pub fn nat_arguments(
40  first: nat_notation::Expr,
41  rest: List(nat_notation::Arguments_Rep1)
42): List(SynExpr) ! {macro_error, Div} {
43  Cons(nat_expression(first), nat_more_arguments(rest))
44}
45pub fn nat_more_arguments(
46  rest: List(nat_notation::Arguments_Rep1)
47): List(SynExpr) ! {macro_error, Div} {
48  match rest { Nil => Nil
49    Cons(r, more) => Cons(nat_expression(r.expr), nat_more_arguments(more)) }
50}
51pub fn nat_atom(t: nat_notation::Atom): SynExpr ! {macro_error, Div} {
52  match t {
53    Num(tok) => ExNat(tok.text, tok.span)
54    Reference(r) => match r.arguments {
55      None => nat_name(r.name)
56      Some(args) => ExApp(nat_name(r.name), nat_arguments(args.expr, args.rep), r.span)
57    }
58    Paren(p) => nat_expression(p.expr)
59    Embed(e) => e
60  }
61}
62pub fn nat_factors(
63  acc: SynExpr,
64  rest: List(nat_notation::Prod_Rep1)
65): SynExpr ! {macro_error, Div} {
66  match rest { Nil => acc
67    Cons(r, more) => {
68      let t := nat_atom(r.atom)
69      nat_factors(quote { Nat::mul($acc, $t) }, more)
70    }
71  }
72}
73pub fn nat_product(p: nat_notation::Prod): SynExpr ! {macro_error, Div} {
74  nat_factors(nat_atom(p.atom), p.rep)
75}
76pub fn nat_summands(
77  acc: SynExpr,
78  rest: List(nat_notation::Sum_Rep1)
79): SynExpr ! {macro_error, Div} {
80  match rest { Nil => acc
81    Cons(r, more) => {
82      let t := nat_product(r.prod)
83      let next := match r.add_op {
84        Plus(tok) => quote { Nat::add($acc, $t) }
85        Minus(tok) => quote { Nat::sub($acc, $t) }
86      }
87      nat_summands(next, more)
88    }
89  }
90}
91pub fn nat_sum(s: nat_notation::Sum): SynExpr ! {macro_error, Div} {
92  nat_summands(nat_product(s.prod), s.rep)
93}
94pub fn nat_expression(e: nat_notation::Expr): SynExpr ! {macro_error, Div} {
95  let l := nat_sum(e.sum)
96  match e.opt {
97    None => l
98    Some(r) => {
99      let b := nat_sum(r.sum)
100      match r.relation {
101        EqOp(tok) => quote { Eq(Nat, $l, $b) }
102        LeOp(tok) => quote { Eq(Bool, Nat::le($l, $b), true) }
103        LtOp(tok) => quote { Eq(Bool, Nat::lt($l, $b), true) }
104      }
105    }
106  }
107}
108// The Bool test behind a relation, for `Consider whether $nat_bool!(`p ≤ n`) holds.`
109pub fn nat_bool_expression(e: nat_notation::Expr): SynExpr ! {macro_error, Div} {
110  let l := nat_sum(e.sum)
111  match e.opt {
112    None => l
113    Some(r) => {
114      let b := nat_sum(r.sum)
115      match r.relation {
116        EqOp(tok) => quote { Nat::eq($l, $b) }
117        LeOp(tok) => quote { Nat::le($l, $b) }
118        LtOp(tok) => quote { Nat::lt($l, $b) }
119      }
120    }
121  }
122}
123pub macro fn nat_bool(e: nat_notation::Expr): SynExpr ! {macro_error, Div} {
124  nat_bool_expression(e)
125}
126pub macro fn nat(e: nat_notation::Expr): SynExpr ! {macro_error, Div} { nat_expression(e) }

128pub macro fn nat_eq(e: nat_notation::Expr): SynExpr ! {macro_error, Div} { nat_expression(e) }
