9pub theorem sub_common_right(a: Nat, b: Nat, c: Nat): nat!(`a + c - (b + c) = a - b`) {
10  ⊢ nat!(`a + c - (b + c) = a - b`) from match c {
11    Zero => Refl
12    Succ(k) => calc!(```
13      $nat!(`(a + Succ(k)) - (b + Succ(k))`)
14      = $nat!(`(a + k) - (b + k)`) from $Nat::sub_succ_succ(nat!(`a + k`), nat!(`b + k`))
15      = $nat!(`a - b`) from $sub_common_right(a, b, k)
16      ```)
17  }
18}
19pub theorem mul_subtract(a: Nat, k: Nat, l: Nat): nat!(`a * (k - l) = a * k - a * l`) {
20  ⊢ nat!(`a * (k - l) = a * k - a * l`) from match k {
21    Zero => Eq::trans(
22      Eq::congr((x: Nat) => nat!(`a * x`), Nat::sub_zero_left(l)),
23      Eq::trans(
24        Nat::mul_zero_right(a),
25        Eq::sym(
26          Eq::trans(
27            Eq::congr((x: Nat) => nat!(`x - a * l`), Nat::mul_zero_right(a)),
28            Nat::sub_zero_left(nat!(`a * l`))
29          )
30        )
31      )
32    )
33    Succ(j) => ⊢ nat!(`a * (Succ(j) - l) = a * Succ(j) - a * l`) from match l {
34      Zero => Eq::sym(Eq::congr((x: Nat) => nat!(`a * Succ(j) - x`), Nat::mul_zero_right(a)))
35      Succ(i) => Eq::trans(
36        Eq::congr((x: Nat) => nat!(`a * x`), Nat::sub_succ_succ(j, i)),
37        Eq::trans(
38          mul_subtract(a, j, i),
39          Eq::sym(
40            Eq::trans(
41              Eq::congr((x: Nat) => nat!(`x - a * Succ(i)`), Nat::mul_succ_right(a, j)),
42              Eq::trans(
43                Eq::congr((x: Nat) => nat!(`a * j + a - x`), Nat::mul_succ_right(a, i)),
44                sub_common_right(nat!(`a * j`), nat!(`a * i`), a)
45              )
46            )
47          )
48        )
49      )
50    }
51  }
52}
53pub theorem divides_subtract(
54  a: Nat,
55  b: Nat,
56  c: Nat,
57  ab: nat_rel!(`a | b`),
58  ac: nat_rel!(`a | c`)
59): nat_rel!(`a | b - c`) {
60  ⊢ nat_rel!(`a | b - c`) from match ab {
61    By(k, hk) => ⊢ nat_rel!(`a | b - c`) from match ac {
62      By(l, hl) => By(
63        nat!(`k - l`),
64        calc!(```
65          $nat!(`a * (k - l)`)
66          = $nat!(`a * k - a * l`) from $mul_subtract(a, k, l)
67          = $nat!(`b - a * l`) from $Eq::congr((x: Nat) => nat!(`x - a * l`), hk)
68          = $nat!(`b - c`) from $Eq::congr((x: Nat) => nat!(`b - x`), hl)
69          ```)
70      )
71    }
72  }

74pub theorem sub_sum_left(a: Nat, b: Nat): nat!(`a + b - a = b`) {
75  ⊢ nat!(`a + b - a = b`) from match a {
76    Zero => Nat::add_zero_left(b)
77    Succ(k) => calc!(```
78      $nat!(`Succ(k) + b - Succ(k)`)
79      = $nat!(`Succ(k + b) - Succ(k)`) from $Eq::congr(
80        (x: Nat) => nat!(`x - Succ(k)`),
81        Nat::add_succ_left(k, b)
82      )
83      = $nat!(`k + b - k`) from $Nat::sub_succ_succ(nat!(`k + b`), k)
84      = $b from $sub_sum_left(k, b)
85      ```)
86  }
