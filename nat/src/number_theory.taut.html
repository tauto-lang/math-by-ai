9pub theorem divides_sum(a: Nat, xs: List(Nat), h: AllDivisible(a, xs)): nat_rel!(`a | sum(xs)`) {
10  ⊢ nat_rel!(`a | sum(xs)`) from match h {
11    AllNil => divides_zero(a)
12    AllCons(x, rest, hx, hr) => divides_add(a, x, sum(rest), hx, divides_sum(a, rest, hr))
13  }
14}
15// Every positive power of a number is a multiple of that number.
16pub theorem divides_positive_power(a: Nat, n: Nat): nat_rel!(`a | power(Nat::mul, 1, a, Succ(n))`) {
17  By(power(Nat::mul, 1, a, n), Nat::mul_comm(a, power(Nat::mul, 1, a, n)))
18}
19// The same divisor divides every natural linear combination of two multiples.
20pub theorem divides_linear_combination(
21  a: Nat,
22  b: Nat,
23  c: Nat,
24  x: Nat,
25  y: Nat,
26  ab: nat_rel!(`a | b`),
27  ac: nat_rel!(`a | c`)
28): nat_rel!(`a | b * x + c * y`) {
29  divides_add(
30    a,
31    nat!(`b * x`),
32    nat!(`c * y`),
33    divides_mul_right(a, b, x, ab),
34    divides_mul_right(a, c, y, ac)
35  )
36}
