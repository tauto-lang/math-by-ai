17pub theorem prime_ne_one(p: Nat, hp: Prime(p), eq: nat!(`p = 1`)): False {
18  false_ne_true(Eq::trans(Eq::sym(Eq::congr((x: Nat) => Nat::lt(1, x), eq)), prime_gt_one(p, hp)))
19}
20pub theorem prime_not_divides_one(p: Nat, hp: Prime(p), h: nat_rel!(`p | 1`)): False {
21  false_ne_true(
22    Eq::trans(
23      Eq::sym(Nat::lt_irrefl(p)),
24      Nat::le_lt_trans(p, 1, p, divides_bound(p, 0, h), prime_gt_one(p, hp))
25    )
26  )
27}
28pub theorem prime_divides_prime(
29  p: Nat,
30  q: Nat,
31  hp: Prime(p),
32  hq: Prime(q),
33  h: nat_rel!(`p | q`)
34): nat!(`p = q`) {
35  Or::elim(
36    prime_divisor(q, hq, p, h),
37    (eq: p = 1) => False::elim(prime_ne_one(p, hp, eq)),
38    (eq: p = q) => eq
39  )
40}

42pub(open) inductive ExtractPrime(p: Nat, xs: List(Nat)): Prop {
43  Extracted(
44    rest: List(Nat),
45    primes: AllPrime(rest),
46    product: nat!(`prod(xs) = p * prod(rest)`),
47    permutation: Permutation(Nat, xs, Cons(p, rest))
48  ): ExtractPrime(p, xs)
49}
50pub theorem extract_prime(
51  p: Nat,
52  xs: List(Nat),
53  hp: Prime(p),
54  all: AllPrime(xs),
55  divides: nat_rel!(`p | prod(xs)`)
56): ExtractPrime(p, xs) {
57  ⊢ ExtractPrime(p, xs) from match all {
58    PrimesNil => False::elim(prime_not_divides_one(p, hp, divides))
59    PrimesCons(q, rest, hq, hr) => Or::elim(prime_divides_product(p, q, prod(rest), hp, divides),
60      (head: nat_rel!(`p | q`)) => {
61        let eq := prime_divides_prime(p, q, hp, hq, head)
62        Extracted(rest, hr, Eq::congr((x: Nat) => nat!(`x * prod(rest)`), Eq::sym(eq)),
63          Eq::subst((x: Nat) => Permutation(Nat, Cons(q, rest), Cons(x, rest)), Eq::sym(eq), permutation_refl(Cons(q, rest))))
64      },
65      (tail: nat_rel!(`p | prod(rest)`)) => ⊢ ExtractPrime(p, Cons(q, rest)) from match extract_prime(p, rest, hp, hr, tail) {
66        Extracted(remaining, primes, product, perm) => Extracted(Cons(q, remaining), PrimesCons(q, remaining, hq, primes),
67          calc!(```
68            $nat!(`q * prod(rest)`)
69            = $nat!(`q * (p * prod(remaining))`) from $Eq::congr((x: Nat) => nat!(`q * x`), product)
70            = $nat!(`p * (q * prod(remaining))`) from $left_commute(
71              Nat::mul,
72              1,
73              multiplicative_comm_monoid(),
74              q,
75              p,
76              prod(remaining)
77            )
78            ```),
79          Chain(Cons(q, rest), Cons(q, Cons(p, remaining)), Cons(p, Cons(q, remaining)),
80            Keep(q, rest, Cons(p, remaining), perm), Swap(q, p, remaining)))
81      })
82  }
83}
84theorem cancel_prime(
85  p: Nat,
86  a: Nat,
87  b: Nat,
88  hp: Prime(p),
89  h: nat!(`p * a = p * b`)
90): nat!(`a = b`) {
91  ⊢ nat!(`a = b`) from match p {
92    Zero => False::elim(false_ne_true(prime_gt_one(0, hp)))
93    Succ(k) => Nat::mul_cancel(
94      a,
95      b,
96      k,
97      calc!(```
98        $nat!(`a * Succ(k)`)
99        = $nat!(`Succ(k) * a`) from $Eq::sym(Nat::mul_comm(Succ(k), a))
100        = $nat!(`Succ(k) * b`) from $h
101        = $nat!(`b * Succ(k)`) from $Nat::mul_comm(Succ(k), b)
102        ```)
103    )
104  }
105}

107pub theorem prime_products_unique(
108  xs: List(Nat),
109  ys: List(Nat),
110  hx: AllPrime(xs),
111  hy: AllPrime(ys),
112  product: nat!(`prod(xs) = prod(ys)`)
113): Permutation(Nat, xs, ys) {
114  ⊢ Permutation(Nat, xs, ys) from match hx {
115    PrimesNil => ⊢ Permutation(Nat, Nil, ys) from match hy {
116      PrimesNil => Empty
117      PrimesCons(p, rest, hp, hr) => False::elim(
118        prime_not_divides_one(
119          p,
120          hp,
121          Eq::subst((n: Nat) => nat_rel!(`p | n`), Eq::sym(product), By(prod(rest), Refl))
122        )
123      )
124    }
125    PrimesCons(p, rest, hp, hr) => ⊢ Permutation(Nat, Cons(p, rest), ys) from match extract_prime(
126      p,
127      ys,
128      hp,
129      hy,
130      Eq::subst((n: Nat) => nat_rel!(`p | n`), product, By(prod(rest), Refl))
131    ) {
132      Extracted(remaining, primes, eq, perm) => Chain(
133        Cons(p, rest),
134        Cons(p, remaining),
135        ys,
136        Keep(
137          p,
138          rest,
139          remaining,
140          prime_products_unique(
141            rest,
142            remaining,
143            hr,
144            primes,
145            cancel_prime(p, prod(rest), prod(remaining), hp, Eq::trans(product, eq))
146          )
147        ),
148        permutation_sym(ys, Cons(p, remaining), perm)
149      )
150    }
151  }
152}
153pub theorem factorization_unique(
154  n: Nat,
155  xs: List(Nat),
156  ys: List(Nat),
157  hx: Factorization(n, xs),
158  hy: Factorization(n, ys)
159): Permutation(Nat, xs, ys) {
160  prime_products_unique(
161    xs,
162    ys,
163    factorization_primes(n, xs, hx),
164    factorization_primes(n, ys, hy),
165    calc!(```
166      $nat!(`prod(xs)`)
167      = $n from $factorization_product(n, xs, hx)
168      = $nat!(`prod(ys)`) from $Eq::sym(factorization_product(n, ys, hy))
169      ```)
170  )
