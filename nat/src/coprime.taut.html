13pub fn Coprime(a: Nat, b: Nat): Prop { nat!(`gcd(a, b) = 1`) }

15theorem scale_left(
16  a: Nat,
17  b: Nat,
18  g: Nat,
19  c: Nat,
20  na: Nat,
21  nb: Nat
22): nat!(`(g + (a * na + b * nb)) * c = g * c + (a * c * na + b * c * nb)`) {
23  by nat_norm!(): nat!(`(g + (a * na + b * nb)) * c = g * c + (a * c * na + b * c * nb)`) {}
24}
25theorem scale_right(
26  a: Nat,
27  b: Nat,
28  c: Nat,
29  pa: Nat,
30  pb: Nat
31): nat!(`(a * pa + b * pb) * c = a * c * pa + b * c * pb`) {
32  by nat_norm!(): nat!(`(a * pa + b * pb) * c = a * c * pa + b * c * pb`) {}
33}
34pub theorem balance_scaled(
35  a: Nat,
36  b: Nat,
37  g: Nat,
38  c: Nat,
39  coeff: Coefficients,
40  h: balance(a, b, g, coeff)
41): balance(nat!(`a * c`), nat!(`b * c`), nat!(`g * c`), coeff) {
42  ⊢ balance(nat!(`a * c`), nat!(`b * c`), nat!(`g * c`), coeff) from match coeff {
43    Coeff(pa, na, pb, nb) => calc!(```
44      $nat!(`g * c + (a * c * na + b * c * nb)`)
45      = $nat!(`(g + (a * na + b * nb)) * c`) from $Eq::sym(scale_left(a, b, g, c, na, nb))
46      = $nat!(`(a * pa + b * pb) * c`) from $Eq::congr((x: Nat) => nat!(`x * c`), h)
47      = $nat!(`a * c * pa + b * c * pb`) from $scale_right(a, b, c, pa, pb)
48      ```)
49  }
50}
51// A common divisor of a and b divides the balanced value g.
52pub theorem divides_balance(
53  a: Nat,
54  b: Nat,
55  g: Nat,
56  coeff: Coefficients,
57  h: balance(a, b, g, coeff),
58  d: Nat,
59  da: nat_rel!(`d | a`),
60  db: nat_rel!(`d | b`)
61): nat_rel!(`d | g`) {
62  ⊢ nat_rel!(`d | g`) from match coeff {
63    Coeff(pa, na, pb, nb) => Eq::subst(
64      (x: Nat) => nat_rel!(`d | x`),
65      calc!(```
66        $nat!(`(a * pa + b * pb) - (a * na + b * nb)`)
67        = $nat!(`(g + (a * na + b * nb)) - (a * na + b * nb)`) from $Eq::congr(
68          (x: Nat) => nat!(`x - (a * na + b * nb)`),
69          Eq::sym(h)
70        )
71        = $nat!(`((a * na + b * nb) + g) - (a * na + b * nb)`) from $Eq::congr(
72          (x: Nat) => nat!(`x - (a * na + b * nb)`),
73          Nat::add_comm(g, nat!(`a * na + b * nb`))
74        )
75        = $g from $sub_sum_left(nat!(`a * na + b * nb`), g)
76        ```),
77      divides_subtract(
78        d,
79        nat!(`a * pa + b * pb`),
80        nat!(`a * na + b * nb`),
81        divides_add(
82          d,
83          nat!(`a * pa`),
84          nat!(`b * pb`),
85          divides_mul_right(d, a, pa, da),
86          divides_mul_right(d, b, pb, db)
87        ),
88        divides_add(
89          d,
90          nat!(`a * na`),
91          nat!(`b * nb`),
92          divides_mul_right(d, a, na, da),
93          divides_mul_right(d, b, nb, db)
94        )
95      )
96    )
97  }
98}
99pub theorem coprime_common_multiple(
100  a: Nat,
101  b: Nat,
102  c: Nat,
103  d: Nat,
104  h: Coprime(a, b),
105  da: nat_rel!(`d | a * c`),
106  db: nat_rel!(`d | b * c`)
107): nat_rel!(`d | c`) {
108  Eq::subst(
109    (x: Nat) => nat_rel!(`d | x`),
110    calc!(```
111      $nat!(`gcd(a, b) * c`)
112      = $nat!(`1 * c`) from $Eq::congr((x: Nat) => nat!(`x * c`), h)
113      = $c from $Nat::mul_one_left(c)
114      ```),
115    divides_balance(
116      nat!(`a * c`),
117      nat!(`b * c`),
118      nat!(`gcd(a, b) * c`),
119      bezout_coefficients(a, b),
120      balance_scaled(a, b, gcd(a, b), c, bezout_coefficients(a, b), bezout_balance(a, b)),
121      d,
122      da,
123      db
124    )
125  )
126}
127// Euclid's lemma for arbitrary coprime natural numbers.
128pub theorem coprime_cancel(
129  a: Nat,
130  b: Nat,
131  c: Nat,
132  h: Coprime(a, b),
133  divides_product: nat_rel!(`a | b * c`)
134): nat_rel!(`a | c`) {
135  coprime_common_multiple(
136    a,
137    b,
138    c,
139    a,
140    h,
141    divides_mul_right(a, a, c, divides_refl(a)),
142    divides_product
143  )
